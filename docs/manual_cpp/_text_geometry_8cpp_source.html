3 * @brief Texto vectorial SVG/DXF convertido a polígonos con la fuente del sistema.
4 *
5 * Interpreta las transformaciones de SVG (`translate`, `scale`, `rotate`,
6 * `matrix`, `skewX`, `skewY`), compone la matriz del elemento de texto con la
7 * pila de grupos y `<svg>` contenedores, y convierte los glifos de
8 * `QPainterPath` en geometría GEOS. Los contadores de las letras (A, O, R…)
9 * se conservan como agujeros aplicando una XOR sucesiva de subcaminos. Para
10 * DXF se reutiliza el mismo enfoque con los datos ya transformados por el
11 * parser (posición, altura, rotación y alineación).

34/**
35 * @brief Interpreta una lista de transformaciones SVG como un `QTransform`.
36 *
37 * Extrae cada `nombre(args)` con una expresión regular, convierte los
38 * argumentos a números (separados por comas o espacios, ignorando los tramos no
39 * numéricos) y compone cada matriz por la izquierda, que es la semántica de
40 * SVG: en `translate(10,0) scale(2)` primero se escala y luego se traslada.
41 *
42 * Se soportan `translate(tx[,ty])`, `scale(sx[,sy])`, `rotate(a[,cx,cy])`
43 * (rotación alrededor de un centro si se dan 3 argumentos), `matrix(a,b,c,d,e,f)`
44 * y `skewX`/`skewY` en grados (convertidos con `tan`).
45 *
46 * @param text Valor del atributo `transform`; puede estar vacío.
47 * @return Transformación equivalente; la identidad si no hay funciones
48 * reconocibles o los argumentos son insuficientes.
49 * @note Las funciones desconocidas se ignoran silenciosamente y el orden de
50 * composición asume la convención de vectores fila de Qt.
51 */
52QTransform parseSvgTransformQt(const QString& text) {
53 QTransform result;
54 static const QRegularExpression fn(
55 R"(([A-Za-z]+)\s*\‍(([^)]*)\))");
56 auto it = fn.globalMatch(text);
57 while (it.hasNext()) {
58 const QRegularExpressionMatch m = it.next();
59 const QString name = m.captured(1).toLower();
60 const QStringList raw =
61 m.captured(2).split(QRegularExpression("[\\s,]+"), Qt::SkipEmptyParts);
62 QVector<double> v;
63 for (const QString& part : raw) {
64 bool ok = false;
65 const double value = part.toDouble(&ok);
66 if (ok) v.push_back(value);
67 }
68 QTransform t;
69 if (name == "translate" && v.size() >= 1)
70 t.translate(v[0], v.size() > 1 ? v[1] : 0.0);
71 else if (name == "scale" && v.size() >= 1)
72 t.scale(v[0], v.size() > 1 ? v[1] : v[0]);
73 else if (name == "rotate" && v.size() >= 1) {
74 if (v.size() >= 3) {
75 t.translate(v[1], v[2]);
76 t.rotate(v[0]);
77 t.translate(-v[1], -v[2]);
78 } else {
79 t.rotate(v[0]);
80 }
81 } else if (name == "matrix" && v.size() >= 6) {
82 t = QTransform(v[0], v[1], v[2], v[3], v[4], v[5]);
83 } else if (name == "skewx" && v.size() >= 1) {
84 t.shear(std::tan(v[0] * kPi / 180.0), 0.0);
85 } else if (name == "skewy" && v.size() >= 1) {
86 t.shear(0.0, std::tan(v[0] * kPi / 180.0));
87 }
88 result = t * result;
89 }
90 return result;
91}

93/**
94 * @brief Extrae el primer número de un atributo SVG.
95 *
96 * Acepta el formato de la especificación (signo, parte entera y/o decimal,
97 * exponente) y descarta unidades y texto adicional, de modo que `"12.5mm"`,
98 * `"10"` o `" 3e-2 px"` producen valores utilizables.
99 *
100 * @param value Cadena del atributo.
101 * @param fallback Valor a devolver si la cadena está vacía o no contiene
102 * ningún número.
103 * @return Número detectado o `fallback`.
104 */
105static double svgNumber(const QString& value, double fallback) {
106 if (value.isEmpty()) return fallback;
107 static const QRegularExpression number(
108 R"re([-+]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][-+]?\d+)?)re");
109 const auto m = number.match(value);
110 return m.hasMatch() ? m.captured(0).toDouble() : fallback;
111}

113/**
114 * @brief Convierte todos los `<text>` de un SVG en polígonos rellenos.
115 *
116 * Lee el archivo en streaming con `QXmlStreamReader`, manteniendo una pila de
117 * transformaciones: al entrar en `<svg>` o `<g>` apila su `transform` compuesta
118 * con la del padre y al salir la descarta. Para cada `<text>` toma el contenido
119 * (incluidos `tspan`), arma la fuente con familia (`Helvetica` por defecto),
120 * peso negrita y un tamaño de referencia de 100 px, y genera el contorno con
121 * `QPainterPath::addText`. Luego aplica: desplazamiento por `text-anchor`
122 * (start/middle/end, midiendo con `QFontMetricsF`), traslación a `(x, y)`,
123 * escala `fontSize/100` con inversión de Y (SVG crece hacia abajo), la
124 * transformación propia del elemento y la heredada de los grupos.
125 *
126 * Los subcaminos resultantes se convierten en polígonos GEOS reparados y se
127 * combinan con XOR sucesiva (`A ∪ B − A ∩ B`) para que los contadores queden
128 * huecos en lugar de rellenos.
129 *
130 * @param path Ruta del archivo SVG.
131 * @return Un polígono por elemento de texto no vacío; vector vacío si el
132 * archivo no se puede abrir, no es XML válido o no tiene texto.
133 * @note El `font-size` se interpreta en unidades de usuario del SVG, sin
134 * resolver `em`, `%` ni hojas de estilo CSS. El tamaño de 100 px sólo es
135 * un pivote de escala para conservar precisión.
136 */
137std::vector<fc::Geom> svgTextGeometry(const QString& path) {
138 std::vector<Geom> out;
139 QFile file(path);
140 if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) return out;
141
142 QXmlStreamReader xml(&file);
143 QVector<QTransform> stack{QTransform()};
144 while (!xml.atEnd()) {
145 xml.readNext();
146 if (xml.isStartElement()) {
147 const QString name = xml.name().toString().toLower();
148 const QXmlStreamAttributes attrs = xml.attributes();
149 if (name == "g" || name == "svg") {
150 stack.push_back(parseSvgTransformQt(attrs.value("transform").toString()) *
151 stack.last());
152 continue;
153 }
154 if (name != "text") continue;
155
156 const QString text = xml.readElementText(QXmlStreamReader::IncludeChildElements);
157 if (text.trimmed().isEmpty()) continue;
158
159 QString family = attrs.value("font-family").toString();
160 if (family.isEmpty()) family = QStringLiteral("Helvetica");
161 QFont font(family);
162 font.setPixelSize(100);
163 font.setBold(attrs.value("font-weight").toString() == "bold");
164
165 QPainterPath pathText;
166 pathText.addText(0.0, 0.0, font, text);
167 if (pathText.isEmpty()) continue;
168
169 const double fontSize = svgNumber(attrs.value("font-size").toString(), 16.0);
170 const double scale = fontSize / 100.0;
171 const double x = svgNumber(attrs.value("x").toString(), 0.0);
172 const double y = svgNumber(attrs.value("y").toString(), 0.0);
173 const QString anchor = attrs.value("text-anchor").toString().toLower();
174 double dx = x;
175 if (anchor == "middle" || anchor == "end") {
176 const double width =
177 QFontMetricsF(font).horizontalAdvance(text) * scale;
178 dx -= anchor == "middle" ? width / 2.0 : width;
179 }
180
181 QTransform local;
182 local.translate(dx, y);
183 local.scale(scale, -scale);
184 const QTransform total =
185 local * parseSvgTransformQt(attrs.value("transform").toString()) * stack.last();
186 const QPainterPath mapped = total.map(pathText);
187
188 Geom filled = Geom::emptyPolygon();
189 bool any = false;
190 for (const QPolygonF& poly : mapped.toSubpathPolygons()) {
191 if (poly.size() < 3) continue;
192 Ring ring;
193 for (const QPointF& p : poly) ring.push_back({p.x(), p.y()});
194 const Geom part = Geom::polygon(ring).fixed();
195 if (!part.valid() || part.empty()) continue;
196 if (!any) {
197 filled = part;
198 any = true;
199 } else {
200 filled = filled.unite(part).difference(filled.intersection(part));
201 }
202 }
203 if (any) out.push_back(filled);
204 } else if (xml.isEndElement()) {
205 const QString name = xml.name().toString().toLower();
206 if ((name == "g" || name == "svg") && stack.size() > 1) stack.pop_back();
207 }
208 }
209 return out;
210}

216/**
217 * @brief Convierte una ruta Qt en un único polígono GEOS con agujeros.
218 *
219 * Toma cada subcamino cerrado con al menos 3 puntos, lo convierte en polígono,
220 * lo repara (`fixed()`) y lo acumula con XOR sucesiva para que los contornos
221 * internos (contadores de letras, huecos de glifos) resten área en vez de
222 * sumarla.
223 *
224 * @param path Ruta con el texto ya transformado.
225 * @return Vector con un polígono, o vacío si ningún subcamino es utilizable.
226 * @note La XOR es conmutativa respecto del orden de los subcaminos, lo que
227 * evita depender del orden en que Qt los devuelve.
228 */
229std::vector<Geom> pathToGeometry(const QPainterPath& path) {
230 std::vector<Geom> out;
231 Geom filled = Geom::emptyPolygon();
232 bool any = false;
233 for (const QPolygonF& poly : path.toSubpathPolygons()) {
234 if (poly.size() < 3) continue;
235 Ring ring;
236 for (const QPointF& p : poly) ring.push_back({p.x(), p.y()});
237 const Geom part = Geom::polygon(ring).fixed();
238 if (!part.valid() || part.empty()) continue;
239 if (!any) {
240 filled = part;
241 any = true;
242 } else {
243 filled = filled.unite(part).difference(filled.intersection(part));
244 }
245 }
246 if (any) out.push_back(filled);
247 return out;
248}

252/**
253 * @brief Convierte los TEXT/MTEXT de un DXF en polígonos rellenos.
254 *
255 * El parser DXF entrega `fc::DxfText` con posición, altura, rotación y
256 * alineación ya transformadas (incluidos INSERT y bloques). Para cada texto se
257 * separa el contenido por saltos de línea, se mide la altura de mayúscula con
258 * `QFontMetricsF` para derivar la escala (`altura / capHeight`, con 72 px como
259 * respaldo) y se avanza 1.4 veces la altura entre líneas. La alineación
260 * horizontal (0 izquierda, 1 centro, 2 derecha) y la vertical (0 línea base,
261 * 1 abajo, 2 medio, 3 arriba) se traducen en desplazamientos previos a la
262 * rotación y la traslación final.
263 *
264 * @param path Ruta del archivo DXF.
265 * @return Un polígono por línea de texto no vacía; vector vacío si la lectura
266 * falla o no hay textos.
267 * @note Siempre usa Helvetica: el DXF no siempre trae una fuente disponible y
268 * el original de FlatCAM tampoco la respeta. Si la altura es 0 se usa una
269 * escala de 0.01 para no degenerar.
270 * @warning Un fallo del parser se traga con `catch (const std::exception&)` y se
271 * informa como "sin textos", sin mensaje al usuario.
272 */
273std::vector<fc::Geom> dxfTextGeometry(const QString& path) {
274 std::vector<fc::Geom> out;
275 std::vector<fc::DxfText> texts;
276 try {
277 texts = fc::importDxfTexts(path.toStdString());
278 } catch (const std::exception&) {
279 return out;
280 }
281 for (const fc::DxfText& info : texts) {
282 const QStringList lines = QString::fromUtf8(info.text.c_str()).split('\n');
283 if (lines.isEmpty()) continue;
284 QFont font(QStringLiteral("Helvetica"));
285 font.setPixelSize(100);
286 const QFontMetricsF metrics(font);
287 const double capHeight = metrics.capHeight() > 0.0 ? metrics.capHeight() : 72.0;
288 const double scale = info.height > 0.0 ? info.height / capHeight : 0.01;
289 const double lineStep = info.height * 1.4;
290 for (int line = 0; line < lines.size(); ++line) {
291 if (lines[line].isEmpty()) continue;
292 QPainterPath pathText;
293 pathText.addText(0.0, 0.0, font, lines[line]);
294 if (pathText.isEmpty()) continue;
295
296 double dx = 0.0;
297 if (info.hAlign == 1)
298 dx = -metrics.horizontalAdvance(lines[line]) * scale / 2.0;
299 else if (info.hAlign == 2)
300 dx = -metrics.horizontalAdvance(lines[line]) * scale;
301
302 double dy = -line * lineStep;
303 if (info.vAlign == 1) dy += metrics.descent() * scale;
304 else if (info.vAlign == 2)
305 dy -= (capHeight - metrics.descent()) * scale / 2.0;
306 else if (info.vAlign == 3) dy -= capHeight * scale;
307
308 QTransform glyphScale;
309 glyphScale.scale(scale, -scale);
310 QTransform offset;
311 offset.translate(dx, dy);
312 QTransform textRotate;
313 textRotate.rotate(info.rotation);
314 QTransform anchor;
315 anchor.translate(info.x, info.y);
316 const QTransform total = glyphScale * offset * textRotate * anchor;
317 const std::vector<Geom> parts = pathToGeometry(total.map(pathText));
318 out.insert(out.end(), parts.begin(), parts.end());
319 }
320 }
321 return out;
322}
