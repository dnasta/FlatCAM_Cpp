114/**
115 * @brief Decide si una ruta corresponde a un archivo de taladrado Excellon.
116 *
117 * Compara la extensión en minúsculas contra `drl`, `xln`, `exc` y `ncd`. Se
118 * usa para separar Gerber de Excellon en `film`, `project --create` y `gcode`.
119 *
120 * @param path Ruta a evaluar.
121 * @return `true` si la extensión es una de las reconocidas.
122 *
123 * @note Una ruta sin punto se considera Gerber.
124 * @note Depende de `std::tolower` sin incluir `<cctype>`; hoy funciona por
125 * inclusión transitiva de la biblioteca estándar.
126 */
127bool isExcellonPath(const std::string& path) {
128 const std::size_t dot = path.find_last_of('.');
129 if (dot == std::string::npos) return false;
130 std::string ext = path.substr(dot + 1);
131 for (char& c : ext) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
132 return ext == "drl" || ext == "xln" || ext == "exc" || ext == "ncd";
133}

176/**
177 * @brief Interpreta los argumentos de la línea de comandos.
178 *
179 * El primer argumento es el subcomando y el resto son opciones. Acepta tanto
180 * `--opcion valor` como `--opcion=valor`. Los argumentos que no empiezan con
181 * `-` se acumulan como entradas (`inputs`); el primero queda además como
182 * `file` para los comandos de archivo único (isolate, cutout, ncc, ...).
183 *
184 * Tras el bucle se fija la unidad de los programas G-code y se traduce el nombre
185 * del preprocesador a `GCodeFlavor`, con estos alias:
186 * - `grbl` / `grbl_11` → `Grbl`;
187 * - `grbl_no_m6` / `grbl_11_no_m6` → `Grbl` (sin `M6`, por la tabla de perfiles);
188 * - `grbl_laser` → `GrblLaser`;
189 * - `marlin` / `marlin_laser` → `Marlin`;
190 * - cualquier otro → `Default`.
191 *
192 * @param argc Cantidad de argumentos.
193 * @param argv Vector de argumentos del sistema.
194 * @param args Estructura a completar.
195 * @return `true` si la línea es válida; `false` si falta el subcomando, una
196 * opción es desconocida o el comando no es `tcl` y no hay entradas.
197 *
198 * @note Los porcentajes se convierten a fracción en el acto: `--overlap 20`
199 * produce `0.2` y `--superposition 80` produce `0.2` de paso.
200 * @note `--power` copia el valor también a `spindleSpeed`, porque el láser usa
201 * ese campo como potencia en los perfiles láser.
202 * @note `--area` se guarda en `cutout.area`, pero `runNcc` también lo lee.
203 * @warning `--area` exige exactamente cuatro números separados por comas; una
204 * cantidad distinta hace que `std::stod` lance `invalid_argument`, que
205 * sólo se captura en `main` y termina con un mensaje genérico.
206 * @warning Las opciones numéricas usan `std::stod`/`std::stoi` sin validar el
207 * texto; un valor no numérico también aborta con el mensaje genérico.
208 * @warning Los alias `grbl_no_m6` y `grbl_11_no_m6` se traducen al mismo
209 * `Grbl` que `grbl`, de modo que son redundantes: el perfil GRBL nunca
210 * emite `M6` (ver `usesM6` en `gcode.cpp`), por lo que la variante "sin
211 * M6" no cambia el programa generado.
212 */
213bool parseArgs(int argc, char** argv, Args& args) {
214 if (argc < 2) return false;
215 args.kind = argv[1];
216
217 for (int i = 2; i < argc; ++i) {
218 std::string arg = argv[i];
219 std::string inlineValue;
220 bool hasInlineValue = false;
221 if (arg.rfind("--", 0) == 0) {
222 const std::size_t eq = arg.find('=');
223 if (eq != std::string::npos) {
224 inlineValue = arg.substr(eq + 1);
225 arg = arg.substr(0, eq);
226 hasInlineValue = true;
227 }
228 }
229 auto value = [&](const char* name) -> std::string {
230 if (hasInlineValue) return inlineValue;
231 if (i + 1 >= argc) {
232 std::cerr << "missing value for " << name << "\n";
233 return {};
234 }
235 return argv[++i];
236 };
237
238 if (arg == "--units") {
239 const std::string u = value("--units");
240 args.units = (u == "in" || u == "inch" || u == "IN") ? fc::Units::Inch : fc::Units::Mm;
241 } else if (arg == "--json") {
242 args.json = true;
243 } else if (arg == "--flatcam-compat") {
244 args.compat = true;
245 } else if (arg == "-o" || arg == "--output") {
246 args.output = value("-o");
247 } else if (arg == "--preprocessor") {
248 args.preprocessor = value("--preprocessor");
249 } else if (arg == "--z-cut") {
250 args.gcode.zCut = std::stod(value("--z-cut"));
251 } else if (arg == "--z-move") {
252 args.gcode.zMove = std::stod(value("--z-move"));
253 } else if (arg == "--z-start") {
254 args.gcode.zStart = std::stod(value("--z-start"));
255 } else if (arg == "--z-end") {
256 args.gcode.zEnd = std::stod(value("--z-end"));
257 } else if (arg == "--z-toolchange") {
258 args.gcode.zToolchange = std::stod(value("--z-toolchange"));
259 } else if (arg == "--feed-z") {
260 args.gcode.feedZ = std::stod(value("--feed-z"));
261 } else if (arg == "--feed-xy") {
262 args.gcode.feedXY = std::stod(value("--feed-xy"));
263 } else if (arg == "--spindle") {
264 args.gcode.spindleSpeed = std::stod(value("--spindle"));
265 } else if (arg == "--power") {
266 args.gcode.laserPower = std::stod(value("--power"));
267 args.gcode.spindleSpeed = args.gcode.laserPower;
268 } else if (arg == "--z-focus") {
269 args.gcode.zFocus = std::stod(value("--z-focus"));
270 } else if (arg == "--decimals") {
271 args.gcode.decimals = std::stoi(value("--decimals"));
272 } else if (arg == "--dwell") {
273 args.gcode.dwellTime = std::stod(value("--dwell"));
274 } else if (arg == "--depth-per-pass") {
275 args.gcode.depthPerPass = std::stod(value("--depth-per-pass"));
276 } else if (arg == "--margin") {
277 args.cutout.margin = std::stod(value("--margin"));
278 args.ncc.margin = args.cutout.margin;
279 args.paint.margin = args.cutout.margin;
280 args.thieving.margin = args.cutout.margin;
281 } else if (arg == "--internal") {
282 args.cutout.external = false;
283 } else if (arg == "--tabs") {
284 args.cutout.tabCount = std::stoi(value("--tabs"));
285 } else if (arg == "--tab-width") {

383 args.gcode.flavor = fc::GCodeFlavor::GrblNoM6;
384 } else {
385 args.gcode.flavor = fc::GCodeFlavor::Default;
386 }
387 return true;
388}

390/**
391 * @brief Parsea un Gerber e informa aperturas, flashes, segmentos y área.
392 *
393 * Con `--json` emite un objeto JSON de una línea; sin ella imprime un informe
394 * legible con unidades del archivo, conteos, área y límites, y lista cada aviso
395 * del parser.
396 *
397 * @param args Opciones ya interpretadas (`file`, `units`, `json`, `compat`).
398 * @return 0 si el parseo se completa (aunque haya avisos); propaga la excepción
399 * del parser en caso de archivo ilegible.
400 *
401 * @note `--flatcam-compat` activa `legacyRectStroke`, que reproduce el trazo
402 * rectangular del FlatCAM original en lugar de la suma de Minkowski.
403 */
404int runGerber(const Args& args) {
405 fc::GerberOptions opt;
406 opt.appUnits = args.units;
407 opt.legacyRectStroke = args.compat;
408 const fc::GerberResult r = fc::GerberParser(opt).parseFile(args.file);
409
410 if (args.json) {
411 std::cout << "{\"type\":\"gerber\",\"file\":\"" << args.file << "\",\"units\":\""
412 << fc::toString(args.units) << "\",\"file_units\":\"" << fc::toString(r.fileUnits)
413 << "\",\"apertures\":" << r.apertures.size() << ",\"flashes\":" << r.flashes
414 << ",\"segments\":" << r.segments << ",\"regions\":" << r.regions << ",\"area\":";
415 printNumber(r.solid.area());
416 std::cout << ",\"bounds\":";
417 printBBox(r.bounds);
418 std::cout << ",\"warnings\":" << r.warnings.size() << "}\n";
419 return 0;
420 }
421
422 std::cout << "Gerber: " << args.file << "\n"
423 << " file units: " << fc::toString(r.fileUnits) << "\n"
424 << " apertures : " << r.apertures.size() << "\n"
425 << " flashes : " << r.flashes << "\n"
426 << " segments : " << r.segments << "\n"
427 << " regions : " << r.regions << "\n"
428 << " area : " << r.solid.area() << " " << fc::toString(args.units) << "^2\n";
429 if (r.bounds.valid) {

431 << r.bounds.xmax << ", " << r.bounds.ymax << "\n";
432 }
433 for (const std::string& w : r.warnings) std::cout << " warning : " << w << "\n";
434 return 0;
435}

481/**
482 * @brief Lee, inspecciona, crea o exporta proyectos `.FlatPrj`.
483 *
484 * El comportamiento depende de las banderas y sigue este orden de prioridad:
485 *
486 * 1. `--create`: el primer `input` es el archivo de salida y el resto son
487 * Gerber/Excellon que se parsean y empaquetan en un proyecto nuevo.
488 * 2. `--summarize` o `--export-all DIR`: recorrido en streaming
489 * (`forEachProjectObject`) que informa un resumen por objeto y, si se pidió,
490 * escribe un `.wkt` por objeto con sus sólidos y recorridos.
491 * 3. `--object N`: carga sólo ese objeto del proyecto (sin materializar el
492 * resto), informa su área y límites y, si se pide, exporta el WKT.
493 * 4. `--info`: lee únicamente los metadatos con `inspectProject`, apto para
494 * proyectos comprimidos de varios cientos de MB.

701/**
702 * @brief Calcula el aislamiento de una capa de cobre y exporta su G-code.
703 *
704 * Parsea el Gerber y llama a `isolateRest` si se pidió `--rest` con un diámetro
705 * previo positivo; en caso contrario a `isolateCopper`. Sin `-o` imprime
706 * estadísticas; con `-o` convierte cada recorrido en una polilínea, aplica
707 * opcionalmente `optimizePathOrder` y escribe el programa con `pathsGCode`.
708 *
709 * @param args Opciones de aislamiento, G-code y salida.
710 * @return 0 en éxito; 1 si no puede escribir el archivo de salida.
711 *
712 * @note Los recorridos se obtienen con `Geom::coords()`; los que tienen menos
713 * de 2 puntos se descartan antes de generar el programa.
714 * @note El JSON informa también `not_isolated`, útil para diagnosticar rest
715 * machining.
716 * @warning Las unidades del G-code dependen de `--units`, pero la geometría ya
717 * viene escalada por el parser, así que no hay conversión adicional al
718 * escribir: si el Gerber está en pulgadas y se pide `--units mm`, el
719 * programa queda en mm.
720 */
721int runIsolate(const Args& args) {
722 fc::GerberOptions gopt;
723 gopt.appUnits = args.units;
724 gopt.legacyRectStroke = args.compat;
725 const fc::GerberResult gerber = fc::GerberParser(gopt).parseFile(args.file);
726
727 fc::IsolationResult iso;
728 if (args.rest && args.previousToolDia > 0.0) {
729 iso = fc::isolateRest(gerber.solid, args.previousToolDia, args.isolation);
730 } else {
731 iso = fc::isolateCopper(gerber.solid, args.isolation);
732 }
733
734 if (args.json) {
735 std::cout << "{\"type\":\"isolation\",\"file\":\"" << args.file << "\",\"paths\":"
736 << iso.paths.size() << ",\"not_isolated\":" << iso.notIsolated.size()
737 << ",\"passes\":" << args.isolation.passes
738 << ",\"tool_dia\":" << args.isolation.toolDiameter << ",\"length\":";
739 printNumber(iso.totalLength);
740 std::cout << ",\"bounds\":";
741 printBBox(iso.bounds);
742 std::cout << "}\n";
743 return 0;
744 }
745
746 if (args.output.empty() || args.output == "-") {
747 std::cout << "Isolation: " << args.file << "\n"
748 << " tool : Ø" << args.isolation.toolDiameter << " "
749 << fc::toString(args.units) << "\n"
750 << " passes : " << args.isolation.passes << "\n"
751 << " paths : " << iso.paths.size() << "\n"
752 << " length : " << iso.totalLength << " " << fc::toString(args.units) << "\n";
753 if (iso.bounds.valid) {
754 std::cout << " bounds : " << iso.bounds.xmin << ", " << iso.bounds.ymin << " .. "
755 << iso.bounds.xmax << ", " << iso.bounds.ymax << "\n";
756 }
757 return 0;
758 }
759
760 fc::PathJob job;
761 job.bounds = gerber.bounds;
762 job.toolDiameter = args.isolation.toolDiameter;
763 for (const fc::Geom& path : iso.paths) {
764 fc::Ring ring = path.coords();
765 if (ring.size() < 2) continue;
766 std::vector<fc::Pt> points;
767 points.reserve(ring.size());
768 for (const fc::Pt& p : ring) points.push_back(p);
769 job.paths.push_back(std::move(points));
770 }
771
772 if (args.optimize) fc::optimizePathOrder(job);
773 const std::string code = fc::pathsGCode(job, args.gcode);
774 std::ofstream out(args.output, std::ios::binary);
775 if (!out) {
776 std::cerr << "cannot write " << args.output << "\n";
777 return 1;
778 }
779 out << code;
780 std::cerr << "wrote " << args.output << " (" << job.paths.size() << " paths, "
781 << iso.totalLength << " " << fc::toString(args.units) << ")\n";
782 return 0;
783}

787/**
788 * @brief Aplana geometría de líneas en una lista de polilíneas.
789 *
790 * Acepta `LineString`/`LinearRing` directamente y recurre sobre
791 * `MultiLineString` y `GeometryCollection`. Es el equivalente local del helper
792 * de `ncc.cpp`, usado aquí para los recorridos "follow" del Gerber.
793 *
794 * @param geom Geometría a recorrer.
795 * @param out Lista de polilíneas de salida.
796 *
797 * @note Se ignoran las geometrías vacías, inválidas y los tipos que no son de
798 * línea (los polígonos no se convierten a su contorno).
799 */
800void collectLineStrings(const fc::Geom& geom, std::vector<std::vector<fc::Pt>>& out) {
801 if (!geom.valid() || geom.empty()) return;
802 const int type = geom.typeId();
803 if (type == GEOS_LINESTRING || type == GEOS_LINEARRING) {
804 fc::Ring ring = geom.coords();
805 if (ring.size() >= 2) out.emplace_back(ring.begin(), ring.end());
806 return;
807 }
808 if (type == GEOS_MULTILINESTRING || type == GEOS_GEOMETRYCOLLECTION) {
809 const int n = geom.numGeometries();
810 for (int i = 0; i < n; ++i) collectLineStrings(geom.geometryN(i), out);
811 }
812}

918 * @brief Compone varias capas en un SVG tipo "film" con marco opcional.
919 *
920 * Parsea cada entrada como Gerber o Excellon (según la extensión), une todos los
921 * sólidos, calcula un marco expandido por el margen y exporta el SVG. Con
922 * `--negative` el contenido es `marco − cobre` (película negativa); sin esa
923 * bandera se dibujan el cobre unido y el rectángulo del marco.
924 *
925 * @param args Entradas, `--margin`, `--negative` y `-o out.svg`.
926 * @return 0 en éxito; 1 si la geometría resultante está vacía; 2 si faltan
927 * entradas o la salida.
928 *
929 * @note Si el margen es 0 se usa 2 mm por defecto para que el marco no quede
930 * pegado al cobre.
931 * @note Los recorridos "follow" se acumulan y se pasan a `exportSvg` aparte de
932 * los sólidos.
933 * @warning En modo negativo los recorridos "follow" se siguen dibujando, de
934 * modo que pueden aparecer trazos sobre la zona que debería quedar
935 * transparente.
936 */
937int runFilm(const Args& args) {
938 if (args.inputs.empty()) {
939 std::cerr << "film: at least one input is required\n";
940 return 2;
941 }
942
943 std::vector<fc::Geom> solids;
944 std::vector<fc::Geom> follows;
945 for (const std::string& input : args.inputs) {
946 if (isExcellonPath(input)) {
947 fc::ExcellonOptions eopt;
948 eopt.appUnits = args.units;
949 const fc::ExcellonResult drl = fc::ExcellonParser(eopt).parseFile(input);
950 solids.insert(solids.end(), drl.geometry.begin(), drl.geometry.end());
951 } else {
952 fc::GerberOptions gopt;
953 gopt.appUnits = args.units;
954 const fc::GerberResult gerber = fc::GerberParser(gopt).parseFile(input);
955 solids.insert(solids.end(), gerber.solids.begin(), gerber.solids.end());
956 follows.insert(follows.end(), gerber.follows.begin(), gerber.follows.end());
957 }
958 }
959
960 const fc::Geom merged = fc::Geom::unionOf(solids);
961 const fc::BBox b = merged.bounds();
962 if (!b.valid) {
963 std::cerr << "film: empty geometry\n";
964 return 1;
965 }
966 const double margin = args.cutout.margin == 0.0 ? 2.0 : args.cutout.margin;
967 const fc::BBox frame = b.expanded(margin);
968 const fc::Geom box = fc::Geom::box(frame.xmin, frame.ymin, frame.xmax, frame.ymax);
969
970 std::vector<fc::Geom> outSolids;
971 if (args.negative) {
972 outSolids.push_back(box.difference(merged));
973 } else {
974 outSolids.push_back(merged);
975 outSolids.push_back(fc::Geom::line({
976 {frame.xmin, frame.ymin}, {frame.xmax, frame.ymin},
977 {frame.xmax, frame.ymax}, {frame.xmin, frame.ymax}, {frame.xmin, frame.ymin}}));
978 }
979
980 if (args.output.empty()) {
981 std::cerr << "film: -o out.svg is required\n";
982 return 2;
983 }
984 fc::exportSvg(args.output, outSolids, follows, frame);
985 std::cerr << "wrote " << args.output << "\n";
986 return 0;
987}
