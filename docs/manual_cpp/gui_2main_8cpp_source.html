26// Every form label shows the tooltip of its buddy (the field it describes),
27// including labels created by QFormLayout without an explicit tooltip.
28/**
29 * @brief Filtro global que propaga el tooltip del campo a su etiqueta.
30 *
31 * Instalado sobre la `QApplication`, revisa los eventos de tooltip de todas las
32 * etiquetas del programa. Si la etiqueta no tiene tooltip propio pero su
33 * *buddy* sí, muestra el del campo: así los textos de ayuda definidos una sola
34 * vez (en el control) aparecen también sobre el rótulo, incluso en los
35 * formularios generados por `QFormLayout`.
36 */
37class LabelTooltipFilter : public QObject {
38public:
39 using QObject::QObject;
40
41protected:
42 /**
43 * @brief Muestra el tooltip del campo asociado a una etiqueta.
44 *
45 * Intercepta `QEvent::ToolTip` sobre `QLabel`: si la etiqueta no tiene
46 * tooltip propio pero sí un *buddy* (el campo que describe) con tooltip, lo
47 * muestra en la posición del cursor y consume el evento. Así los formularios
48 * creados con `QFormLayout` heredan la ayuda del campo sin duplicar textos.
49 *
50 * @param obj Objeto que recibe el evento.
51 * @param event Evento a filtrar.
52 * @return `true` si el evento quedó manejado; en cualquier otro caso delega en
53 * `QObject::eventFilter`.
54 */
55 bool eventFilter(QObject* obj, QEvent* event) override {
56 if (event->type() == QEvent::ToolTip) {
57 auto* label = qobject_cast<QLabel*>(obj);
58 if (label && label->toolTip().isEmpty() && label->buddy()) {
59 const QString tip = label->buddy()->toolTip();
60 if (!tip.isEmpty()) {
61 auto* help = static_cast<QHelpEvent*>(event);
62 QToolTip::showText(help->globalPos(), tip, label);
63 return true;
64 }
65 }
66 }
67 return QObject::eventFilter(obj, event);
68 }
69};

89 * @param argc Número de argumentos.
90 * @param argv Vector de argumentos del sistema.

92 * @note Los valores numéricos inválidos se convierten a 0 y la acción
