5 * Renderiza cada página del documento con Core Graphics sobre un búfer

20/**
21 * @brief Renderiza todas las páginas de un PDF a imágenes Qt.
22 *
23 * Abre el documento con `PDFDocument`, calcula la escala `dpi/72` y, por cada
24 * página, deriva el tamaño en píxeles del `MediaBox`, crea la imagen, obtiene un
25 * `CGContext` sobre sus bits (`kCGImageAlphaPremultipliedFirst` con orden de
26 * bytes little-endian, equivalente a `QImage::Format_ARGB32_Premultiplied`),
27 * pinta el fondo blanco, aplica `scale·translate(0, alto)·scale(1,-1)` y dibuja
28 * la página con `drawWithBox:`. Al finalizar libera contexto y espacio de color.
29 *
30 * @param path Ruta del archivo PDF.
31 * @param dpi Resolución de rasterizado; 200 ppp es el valor usado por la
32 * importación.
33 * @param pageSizesMm Si no es nulo, recibe el tamaño de cada página en mm
34 * (`puntos · 25.4 / 72`) en el mismo orden que las imágenes.
35 * @return Imágenes de las páginas en orden; vector vacío si el documento no se
36 * puede abrir o no tiene páginas.
37 * @note El tamaño de píxeles se fuerza a un mínimo de 1 para no crear imágenes
38 * inválidas con páginas de área cero.
39 * @note Usa `CropBox` (con fallback lógico a `MediaBox`) y valida `dpi > 0`.
40 * Bajo MRC el `PDFDocument` se libera explícitamente al terminar.
41 */
42QVector<QImage> renderPdfPages(const QString& path, int dpi, QVector<QSizeF>* pageSizesMm) {
43 QVector<QImage> pages;
44 if (dpi <= 0) return pages;
45 @autoreleasepool {
46 NSString* nsPath = [NSString stringWithUTF8String:path.toUtf8().constData()];
47 NSURL* url = [NSURL fileURLWithPath:nsPath];
48 PDFDocument* document = [[PDFDocument alloc] initWithURL:url];
49 if (!document) return pages;
50 const int count = static_cast<int>([document pageCount]);
51 const double scale = dpi / 72.0;
52 for (int i = 0; i < count; ++i) {
53 PDFPage* page = [document pageAtIndex:i];
54 if (!page) continue;
55 NSRect box = [page boundsForBox:kPDFDisplayBoxCropBox];
56 if (box.size.width <= 0.0 || box.size.height <= 0.0)
57 box = [page boundsForBox:kPDFDisplayBoxMediaBox];
58 const int width = qMax(1, static_cast<int>(std::lround(box.size.width * scale)));
59 const int height = qMax(1, static_cast<int>(std::lround(box.size.height * scale)));
60 QImage image(width, height, QImage::Format_ARGB32_Premultiplied);
61 image.fill(Qt::white);
62 CGColorSpaceRef colorSpace = CGColorSpaceCreateDeviceRGB();
63 CGContextRef context = CGBitmapContextCreate(
64 image.bits(), width, height, 8, image.bytesPerLine(), colorSpace,
65 kCGImageAlphaPremultipliedFirst | kCGBitmapByteOrder32Little);
66 if (!context) {
67 CGColorSpaceRelease(colorSpace);
68 continue;
69 }
70 CGContextSetRGBFillColor(context, 1, 1, 1, 1);
71 CGContextFillRect(context, CGRectMake(0, 0, width, height));
72 CGContextScaleCTM(context, scale, scale);
73 CGContextTranslateCTM(context, 0, box.size.height);
74 CGContextScaleCTM(context, 1, -1);
75 [page drawWithBox:kPDFDisplayBoxCropBox toContext:context];
76 CGContextRelease(context);
77 CGColorSpaceRelease(colorSpace);
78 // PDFKit dibuja con el origen en la esquina inferior izquierda;
79 // QImage usa la esquina superior izquierda. El espejo vertical
80 // deja el texto y la geometría en la orientación correcta.
81 image = image.flipped(Qt::Vertical);
82 pages.push_back(image);
83 if (pageSizesMm)
84 pageSizesMm->push_back(QSizeF(box.size.width * 25.4 / 72.0,
85 box.size.height * 25.4 / 72.0));
86 }
87 [document release];
88 }
89 return pages;
90}
