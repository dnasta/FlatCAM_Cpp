18/**
19 * @brief Interpreta un campo de coordenada Gerber según formato y ceros.
20 *
21 * Acepta un signo opcional, valida que el resto sean sólo dígitos y arma el
22 * valor en tres modos:
23 * - 'L' o 'D': el texto ya trae todos los dígitos, se divide por 10^fracDigits.
24 * - 'T': los ceros finales se omitieron, así que se escala por
25 * 10^(intDigits + fracDigits - len - fracDigits).
26 * Cualquier otro modo es error.
27 * @param text Campo tal como aparece en el archivo (sin el carácter de eje).
28 * @param intDigits Dígitos enteros declarados en FS (3 en X34).
29 * @param fracDigits Dígitos fraccionarios declarados en FS (4 en X34).
30 * @param zeros Modo de ceros: 'L', 'T' o 'D'.
31 * @return Valor con signo, en las unidades del archivo.
32 * @throw std::invalid_argument si el campo está vacío, contiene caracteres no
33 * numéricos o el modo de ceros es desconocido.
34 * @note Un campo con sólo "+" o "-" es válido y produce 0.
35 */
36double parseGerberNumber(std::string_view text, int intDigits, int fracDigits, char zeros) {
37 if (text.empty()) throw std::invalid_argument("empty gerber number");
38
39 bool negative = false;
40 std::size_t i = 0;
41 if (text[i] == '+' || text[i] == '-') {
42 negative = text[i] == '-';
43 ++i;
44 }
45 const std::string_view digits = text.substr(i);
46 for (char c : digits) {
47 if (c < '0' || c > '9') throw std::invalid_argument("invalid gerber number: " + std::string(text));
48 }
49 const long long raw = std::strtoll(std::string(digits).c_str(), nullptr, 10);
50
51 double value = 0.0;
52 if (zeros == 'L' || zeros == 'D') {
53 value = static_cast<double>(raw) * std::pow(10.0, -fracDigits);
54 } else if (zeros == 'T') {
55 const int total = intDigits + fracDigits;
56 const int len = static_cast<int>(digits.size());
57 value = static_cast<double>(raw) * std::pow(10.0, total - len - fracDigits);
58 } else {
59 throw std::invalid_argument("unknown zero suppression mode");
60 }
61 return negative ? -value : value;
62}
