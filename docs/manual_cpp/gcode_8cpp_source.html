40 return flavor == GCodeFlavor::GrblLaser || flavor == GCodeFlavor::MarlinLaserFan ||
41 flavor == GCodeFlavor::MarlinLaserSpindle || flavor == GCodeFlavor::ZLaser;

44/**
45 * @brief Indica si el perfil emite `M6` (cambio de herramienta real).
46 *
47 * Los perfiles "sin M6" (`Grbl`, `GrblNoM6`, `DefaultNoM6`, etc.) dejan el
48 * cambio de herramienta como un comentario `MSG`, pensado para routers con
49 * cambio manual asistido.
50 *
51 * @param flavor Perfil de postprocesador.
52 * @return `true` para MACH3, Marlin, Repetier, Berta, ISEL, Roland, cambio
53 * manual y sonda; `false` para GRBL, láser, HPGL y línea XYZ.
54 */
55bool usesM6(GCodeFlavor flavor) {
56 switch (flavor) {
57 case GCodeFlavor::Default:
58 case GCodeFlavor::Marlin:
59 case GCodeFlavor::Repetier:
60 case GCodeFlavor::Berta:
61 case GCodeFlavor::Isel:
62 case GCodeFlavor::IselIcp:
63 case GCodeFlavor::Roland:
64 case GCodeFlavor::ToolchangeManual:
65 case GCodeFlavor::ToolchangeProbe:
66 return true;
67 default:
68 return false;
69 }
70}

146/**
147 * @brief Reemplaza todas las apariciones de `from` por `to` en `text`.
148 *
149 * Sustitución literal (sin expresiones regulares) que avanza `pos` después del
150 * texto insertado, de modo que no se re-procesa lo reemplazado.
151 *
152 * @param text Cadena a modificar in situ.
153 * @param from Texto buscado; si está vacío la función no hace nada.
154 * @param to Texto de reemplazo.
155 */
156void replaceAll(std::string& text, const std::string& from, const std::string& to) {
157 if (from.empty()) return;
158 std::size_t pos = 0;
159 while ((pos = text.find(from, pos)) != std::string::npos) {
160 text.replace(pos, from.size(), to);
161 pos += to.size();
162 }
163}

165/**
166 * @brief Expande la plantilla de cambio de herramienta personalizada.
167 *
168 * Reemplaza los marcadores `{tool}`, `{diameter}`, `{z}`, `{x}` e `{y}` con los
169 * valores de la herramienta y de las opciones de máquina. Si no hay posición de
170 * cambio definida (`toolchangeXY`), `{x}` e `{y}` se sustituyen por `0`.
171 *
172 * @param opt Opciones que contienen la plantilla y los decimales de formato.
173 * @param tool Número de herramienta.
174 * @param diameter Diámetro de la herramienta.
175 * @return Código resultante, sin salto de línea final.
176 *
177 * @note No valida la plantilla: un marcador mal escrito queda tal cual en el
178 * G-code y puede provocar un error de sintaxis en el controlador.
179 */
180std::string customToolchangeCode(const GCodeOptions& opt, int tool, double diameter) {
181 std::string code = opt.customToolchange;
182 replaceAll(code, "{tool}", std::to_string(tool));
183 replaceAll(code, "{diameter}", number(diameter, opt.decimals));
184 replaceAll(code, "{z}", number(opt.zToolchange, opt.decimals));
185 replaceAll(code, "{x}", number(opt.toolchangeXY ? opt.toolchangeXY->x : 0.0, opt.decimals));
186 replaceAll(code, "{y}", number(opt.toolchangeXY ? opt.toolchangeXY->y : 0.0, opt.decimals));
187 return code;
188}

192 char buffer[64];
193 std::snprintf(buffer, sizeof(buffer), "%.*f", decimals, value);
194 return std::string(buffer);

223/// @brief Escribe el bloque de parámetros usados para construir el programa.
224void writeParameterComments(std::ostringstream& out, const GCodeOptions& opt) {
225 const char* unit = opt.units == Units::Inch ? "in" : "mm";
226 const bool laser = isLaserFlavor(opt.flavor);
227 out << "(Postprocessor: " << flavorName(opt.flavor) << ")\n";
228 out << "(Units: " << unit << ")\n";
229 out << "(Tool diameter: " << number(opt.toolDiameter, opt.decimals) << " " << unit << ")\n";
230 out << "(Feedrate XY: " << number(opt.feedXY, opt.feedDecimals) << " mm/min)\n";
231 out << "(Feedrate Z: " << number(opt.feedZ, opt.feedDecimals) << " mm/min)\n";
232 if (laser) {
233 out << "(Laser power: " << number(opt.laserPower, 1) << ")\n";
234 out << "(Z focus: " << number(opt.zFocus, opt.decimals) << " mm)\n";
235 } else {
236 out << "(Spindle: " << number(opt.spindleSpeed, 0) << " RPM)\n";
237 out << "(Z travel: " << number(opt.zMove, opt.decimals) << " mm)\n";
238 out << "(Z cut: " << number(opt.zCut, opt.decimals) << " mm)\n";
239 out << "(Z start: " << number(opt.zStart, opt.decimals) << " mm)\n";
240 out << "(Z end: " << number(opt.zEnd, opt.decimals) << " mm)\n";
241 out << "(Z tool change: " << number(opt.zToolchange, opt.decimals) << " mm)\n";
242 }
243 out << "(Depth per pass: " << number(opt.depthPerPass, opt.decimals) << " mm)\n";
244 out << "(Dwell: " << number(opt.dwellTime, opt.feedDecimals) << " s)\n";
245 out << "(Tool changes: " << (opt.includeToolChange ? "yes" : "no") << ")\n";
246 out << "(Connect paths: " << (opt.connectPaths ? "yes" : "no") << ")\n";
247 for (const std::string& line : opt.headerComments) out << "(" << line << ")\n";
248 out << "\n";
249}

266/**
267 * @brief Escribe la cabecera de un programa de taladrado.
268 *
269 * La cabecera documenta el preprocesador, la tabla de herramientas (diámetro,
270 * Z de corte y Z de viaje), los rangos X/Y, las RPM, las unidades (`G20`/`G21`),
271 * el modo absoluto (`G90`) y el avance por minuto (`G94`). Después enciende el
272 * husillo si `spindleSpeed > 0` y eleva a `zStart`.
273 *
274 * @param out Flujo de salida del programa.
275 * @param job Trabajo de taladrado (herramientas y límites).
276 * @param opt Parámetros de máquina y formato.
277 *
278 * @note El `M03`/`M04` inicial respeta `opt.spindleCw`; las herramientas con
279 * override de RPM vuelven a emitir su propio `M03` en `drillingGCode`.
280 * @note Los comentarios usan paréntesis, válidos tanto en MACH3 como en GRBL.
281 */
282void writeHeader(std::ostringstream& out, const DrillJob& job, const GCodeOptions& opt) {
283 out << flavorHeader(opt.flavor);
284 out << "(Generated by FlatCAM C++.)\n\n";
285 writeParameterComments(out, opt);
286
287 out << "(TOOLS DIAMETER: )\n";
288 for (const DrillTool& tool : job.tools) {
289 out << "(Tool: " << tool.number << " -> Dia: " << number(tool.diameter, opt.decimals) << ")\n";
290 }
291 out << "\n(Z_CUT: )\n";
292 for (const DrillTool& tool : job.tools) {
293 out << "(Tool: " << tool.number << " -> Z_Cut: " << number(toolZCut(tool, opt), opt.decimals) << ")\n";
294 }
295 out << "\n(Z_MOVE: )\n";
296 for (const DrillTool& tool : job.tools) {
297 out << "(Tool: " << tool.number << " -> Z_Move: " << number(opt.zMove, opt.decimals) << ")\n";
298 }
299 if (job.bounds.valid) {
300 out << "\n(X range: " << number(job.bounds.xmin, opt.decimals) << " ... "
301 << number(job.bounds.xmax, opt.decimals) << ")\n";
302 out << "(Y range: " << number(job.bounds.ymin, opt.decimals) << " ... "
303 << number(job.bounds.ymax, opt.decimals) << ")\n";
304 }
305 out << "\n(Spindle Speed: " << number(opt.spindleSpeed, 0) << " RPM)\n";
306 out << (opt.units == Units::Inch ? "G20\n" : "G21\n");
307 out << "G90\n";
308 out << "G94\n";
309 if (opt.spindleSpeed > 0.0) {
310 out << (opt.spindleCw ? "M03" : "M04");
311 out << " S" << number(opt.spindleSpeed, 0) << "\n";
312 }
313 out << "G00 Z" << number(opt.zStart, opt.decimals) << "\n";
314}

316/**
317 * @brief Emite el cambio de herramienta para una herramienta de taladrado.
318 *
319 * Tres caminos posibles:
320 * - `ToolchangeCustom` con plantilla no vacía: expande la plantilla y baja a
321 * `zMove`, sin `M6`.
322 * - `includeToolChange` activo: apaga el husillo (`M5`), sube a `zToolchange`,
323 * viaja a la posición de cambio si está definida, selecciona con `Tn` y, si
324 * el perfil usa `M6`, emite `M6` + comentario con diámetro y cantidad de
325 * taladros + `M0` (pausa). Sin `M6` deja sólo el comentario `MSG`.
326 * - En cualquier otro caso no escribe nada (el programa sigue con la
327 * herramienta anterior).
328 *
329 * @param out Flujo de salida del programa.
330 * @param tool Herramienta que se va a montar.
331 * @param opt Parámetros de máquina y postprocesador.
332 *
333 * @note La cantidad informada en el mensaje suma taladros y ranuras, útil para
334 * que el operador verifique la herramienta antes de continuar.
335 * @note Termina siempre en `zMove` para que el primer rápido se haga en altura
336 * segura.
337 */
338void writeToolChange(std::ostringstream& out, const DrillTool& tool, const GCodeOptions& opt) {
339 if (opt.flavor == GCodeFlavor::ToolchangeCustom && !opt.customToolchange.empty()) {
340 out << customToolchangeCode(opt, tool.number, tool.diameter) << "\n";
341 out << "G00 Z" << number(opt.zMove, opt.decimals) << "\n";
342 return;
343 }
344 if (opt.includeToolChange) {
345 out << "M5\n";
346 out << "G00 Z" << number(opt.zToolchange, opt.decimals) << "\n";
347 if (opt.toolchangeXY) {
348 out << "G00 " << xy(*opt.toolchangeXY, opt) << "\n";
349 }
350 out << "T" << tool.number << "\n";
351 if (usesM6(opt.flavor)) {
352 out << "M6\n";
353 out << "(MSG, Change to Tool Dia = " << number(tool.diameter, opt.decimals)
354 << " ||| Total drills for tool T" << tool.number << " = "
355 << (tool.holes.size() + tool.slotSegments.size()) << ")\n";
356 out << "M0\n";
357 } else {
358 out << "(MSG, Change to Tool Dia = " << number(tool.diameter, opt.decimals) << ")\n";
359 }
360 out << "G00 Z" << number(opt.zToolchange, opt.decimals) << "\n";
361 out << "G00 Z" << number(opt.zMove, opt.decimals) << "\n";
362 }
363}

370/**
371 * @brief Escribe el ciclo de un taladro simple (centro + penetración).
372 *
373 * Secuencia: rápido al centro (aplicando `offsetX/offsetY` de la herramienta),
374 * penetración a `zCut` con avance `feedZ` —o varias penetraciones parciales de
375 * `depthPerPass` con retracción a `zMove` entre cada una (peck drilling)—,
376 * pausa `G04` si corresponde y retracción final a `zMove`.
377 *
378 * @param out Flujo de salida del programa.
379 * @param holeIn Centro del taladro en coordenadas originales.
380 * @param tool Herramienta (diámetro, overrides y offsets).
381 * @param opt Parámetros globales (Z, avances, dwell y profundidad por pasada).
382 *
383 * @note El peck drilling sólo se activa si `depthPerPass > 0`, la Z de corte es
384 * negativa y su magnitud supera la profundidad por pasada.
385 * @note La pausa usa `tool.dwell` si existe; si no, `opt.dwellTime`.
386 * @warning No se emite `G98`/`G99`: el retorno siempre es a `zMove`, no a la
387 * altura inicial del ciclo.
388 */
389void writeHole(std::ostringstream& out, const Pt& holeIn, const DrillTool& tool, const GCodeOptions& opt) {
390 const Pt hole{holeIn.x + tool.offsetX, holeIn.y + tool.offsetY};
391 const double zCut = toolZCut(tool, opt);
392 const double zMove = toolZMove(tool, opt);
393 const double feedZ = toolFeedZ(tool, opt);
394 const double peck = opt.depthPerPass;
395 out << "G00 " << xy(hole, opt) << "\n";
396 if (peck > 0.0 && zCut < 0.0 && std::fabs(zCut) > peck + 1e-9) {
397 double z = 0.0;
398 while (z > zCut + 1e-9) {
399 z = std::max(zCut, z - peck);
400 out << "G01 Z" << number(z, opt.decimals) << " F"
401 << number(feedZ, opt.feedDecimals) << "\n";
402 if (z > zCut + 1e-9) {
403 out << "G00 Z" << number(zMove, opt.decimals) << "\n";
404 out << "G00 " << xy(hole, opt) << "\n";
405 }
406 }
407 } else {
408 out << "G01 Z" << number(zCut, opt.decimals) << " F"
409 << number(feedZ, opt.feedDecimals) << "\n";
410 }
411 const double dwell = tool.dwell.value_or(opt.dwellTime);
412 if (dwell > 0.0) out << "G04 P" << number(dwell, opt.feedDecimals) << "\n";
413 out << "G00 Z" << number(zMove, opt.decimals) << "\n";
414}

416/**
417 * @brief Escribe el fresado de una ranura (de `start` a `stop` en una pasada).
418 *
419 * Repite el esquema de penetración de `writeHole` (con peck opcional) en el
420 * punto inicial y luego avanza en línea recta hasta el extremo con el avance
421 * XY de la herramienta o el global, para terminar retrayendo a `zMove`.
422 *
423 * @param out Flujo de salida del programa.
424 * @param startIn Extremo inicial de la ranura.
425 * @param stopIn Extremo final de la ranura.
426 * @param tool Herramienta (offsets y overrides de avance).
427 * @param opt Parámetros globales.
428 *
429 * @note A diferencia de `writeHole`, no emite `G04`: una pausa en una ranura no
430 * tiene sentido práctico y se omite para no alargar el ciclo.
431 * @note El peck drilling se aplica sólo en el extremo inicial; la pasada
432 * lateral se hace a la profundidad final ya alcanzada.
433 */
434void writeSlot(std::ostringstream& out, const Pt& startIn, const Pt& stopIn, const DrillTool& tool,
435 const GCodeOptions& opt) {
436 const Pt start{startIn.x + tool.offsetX, startIn.y + tool.offsetY};
437 const Pt stop{stopIn.x + tool.offsetX, stopIn.y + tool.offsetY};
438 const double zCut = toolZCut(tool, opt);
439 const double zMove = toolZMove(tool, opt);
440 const double feedZ = toolFeedZ(tool, opt);
441 const double peck = opt.depthPerPass;
442 out << "G00 " << xy(start, opt) << "\n";
443 if (peck > 0.0 && zCut < 0.0 && std::fabs(zCut) > peck + 1e-9) {
444 double z = 0.0;
445 while (z > zCut + 1e-9) {
446 z = std::max(zCut, z - peck);
447 out << "G01 Z" << number(z, opt.decimals) << " F"
448 << number(feedZ, opt.feedDecimals) << "\n";
449 if (z > zCut + 1e-9) {
450 out << "G00 Z" << number(zMove, opt.decimals) << "\n";
451 out << "G00 " << xy(start, opt) << "\n";
452 }
453 }
454 } else {
455 out << "G01 Z" << number(zCut, opt.decimals) << " F"
456 << number(feedZ, opt.feedDecimals) << "\n";
457 }
458 out << "G01 " << xy(stop, opt) << " F"
459 << number(tool.feedXY.value_or(opt.feedXY), opt.feedDecimals) << "\n";
460 out << "G00 Z" << number(zMove, opt.decimals) << "\n";
461}

465/**
466 * @brief Genera el programa G-code completo de un trabajo de taladrado.
467 *
468 * Escribe la cabecera y, por cada herramienta con contenido, el cambio de
469 * herramienta, el override de RPM si corresponde y todos sus taladros y
470 * ranuras. Las ranuras degeneradas (`start == stop`) se emiten como taladro
471 * simple. Al final apaga el husillo, sube a `zEnd` y, si se pidió, se desplaza
472 * a `endXY`.
473 *
474 * @param job Herramientas con taladros y ranuras.
475 * @param opt Parámetros de máquina, unidades y postprocesador.
476 * @return Programa G-code como cadena.
477 *
478 * @note Las herramientas sin taladros ni ranuras se omiten por completo, de
479 * modo que no se emite un cambio de herramienta inútil.
480 * @note `M05` sólo se emite si la cabecera encendió el husillo
481 * (`spindleSpeed > 0`), evitando comandos huérfanos en programas sin
482 * husillo configurado.
483 * @warning Los overrides de RPM por herramienta emiten `M03` sin apagar antes
484 * el husillo (`M05`); en MACH3 puede ser aceptable, pero no todas las
485 * controladoras aceptan un cambio de sentido en caliente.
486 */
487std::string drillingGCode(const DrillJob& job, const GCodeOptions& opt) {
488 std::ostringstream out;
489 writeHeader(out, job, opt);
490
491 for (const DrillTool& tool : job.tools) {
492 if (tool.holes.empty() && tool.slotSegments.empty()) continue;
493 writeToolChange(out, tool, opt);
494 if (tool.spindleSpeed && *tool.spindleSpeed > 0.0)
495 out << "M03 S" << number(*tool.spindleSpeed, 0) << "\n";
496 for (const Pt& hole : tool.holes) {
497 writeHole(out, hole, tool, opt);
498 }
499 for (const auto& [start, stop] : tool.slotSegments) {
500 if (start.x == stop.x && start.y == stop.y) {
501 writeHole(out, start, tool, opt);
502 } else {
503 writeSlot(out, start, stop, tool, opt);
504 }
505 }
506 }
507
508 if (opt.spindleSpeed > 0.0) out << "M05\n";
509 out << "G00 Z" << number(opt.zEnd, opt.decimals) << "\n";
510 if (opt.endXY) out << "G00 " << xy(*opt.endXY, opt) << "\n";
511 return out.str();
512}

516/// @brief Distancia euclidiana entre dos puntos.
517double dist(const Pt& a, const Pt& b) { return std::hypot(b.x - a.x, b.y - a.y); }

520 * @brief Longitud del recorrido que visita `holes` en orden partiendo de `start`.
521 * @param holes Taladros en su orden actual.
522 * @param start Punto de partida.
523 * @return Distancia acumulada en mm.

525double holesDistance(const std::vector<Pt>& holes, const Pt& start) {
526 double total = 0.0;
527 Pt current = start;
528 for (const Pt& hole : holes) {
529 total += dist(current, hole);
530 current = hole;
531 }
532 return total;

535/**
536 * @brief Ordena `points` por vecino más cercano partiendo de `start`.
537 *
538 * En cada iteración busca el punto no visitado de menor distancia al actual.
539 * Es O(n²) y se usa como solución inicial de las heurísticas de mejora
540 * (2-opt y Or-opt) cuando hay más taladros que el límite del TSP exacto.
541 *
542 * @param points Puntos a reordenar (se modifican in situ).
543 * @param start Punto de partida.
544 */
545void sortNearest(std::vector<Pt>& points, const Pt& start) {
546 std::vector<Pt> remaining = points;
547 std::vector<Pt> ordered;
548 ordered.reserve(points.size());
549 Pt current = start;
550 while (!remaining.empty()) {
551 std::size_t best = 0;
552 double bestDistance = dist(current, remaining[0]);
553 for (std::size_t i = 1; i < remaining.size(); ++i) {
554 const double d = dist(current, remaining[i]);
555 if (d < bestDistance) {
556 bestDistance = d;
557 best = i;
558 }
559 }
560 current = remaining[best];
561 ordered.push_back(current);
562 remaining.erase(remaining.begin() + static_cast<std::ptrdiff_t>(best));
563 }
564 points = std::move(ordered);
565}

567/**
568 * @brief Mejora el orden con la heurística 2-opt (intercambio de aristas).
569 *
570 * Para cada par `(i, j)` evalúa el efecto de invertir el subrecorrido
571 * `[i, j]`: compara las aristas afectadas antes y después (incluida la arista
572 * que llega desde `start` cuando `i == 0`) y acepta el cambio si reduce la
573 * longitud en más de 1e-9 mm. Repite hasta que no haya mejoras o hasta 200
574 * rondas.
575 *
576 * @param points Orden de puntos a mejorar (in situ).
577 * @param start Punto de partida, tratado como predecesor de `points[0]`.
578 *
579 * @note El límite de 200 rondas actúa como guarda contra ciclos por errores de
580 * redondeo en conjuntos muy grandes.
581 * @note El delta ignora la arista de salida cuando `j` es el último punto, que
582 * es correcto porque no hay arista posterior que invertir.
583 */
584void twoOpt(std::vector<Pt>& points, const Pt& start) {
585 if (points.size() < 3) return;
586 bool improved = true;
587 int guard = 0;
588 while (improved && guard++ < 200) {
589 improved = false;
590 for (std::size_t i = 0; i + 1 < points.size(); ++i) {
591 for (std::size_t j = i + 1; j < points.size(); ++j) {
592 const Pt& a = i == 0 ? start : points[i - 1];
593 const Pt& b = points[i];
594 const Pt& c = points[j];
595 const double before = dist(a, b);
596 const double after = dist(a, c);
597 double delta = after - before;
598 if (j + 1 < points.size()) {
599 const Pt& d = points[j + 1];
600 delta += dist(b, d) - dist(c, d);
601 }
602 if (delta < -1e-9) {
603 std::reverse(points.begin() + static_cast<std::ptrdiff_t>(i),
604 points.begin() + static_cast<std::ptrdiff_t>(j) + 1);
605 improved = true;
606 }
607 }
608 }
609 }

614 * @param points Orden de visita.
615 * @param start Punto de partida.

618double pathLengthWithStart(const std::vector<Pt>& points, const Pt& start) {
619 double total = 0.0;

621 for (const Pt& p : points) {
622 total += dist(current, p);

628/**
629 * @brief Resuelve el TSP abierto de forma exacta con programación dinámica.
630 *
631 * Implementa Held-Karp sobre subconjuntos: `dp[mask][i]` es el costo mínimo de
632 * visitar `mask` terminando en `i`, partiendo de `start`. Reconstruye el orden
633 * óptimo siguiendo la matriz `parent`.
634 *
635 * @param points Taladros a ordenar.
636 * @param start Punto de partida.
637 * @return Orden óptimo, o `points` sin cambios si hay más de 18 elementos.
638 *
639 * @note El límite de 18 puntos acota la tabla a 2^18 × 18 entradas
640 * (≈ 4,7 M celdas por matriz); con más puntos se usan las heurísticas.
641 * @note Para `n == 0` devuelve un vector vacío sin tocar las matrices.
642 * @warning Con `n == 18` el consumo de memoria (dos matrices de `double`/`int`)
643 * ronda las decenas de MB; es el motivo del límite.
644 */
645std::vector<Pt> exactTsp(const std::vector<Pt>& points, const Pt& start) {
646 const std::size_t n = points.size();
647 if (n > 18) return points;
648
649 const std::size_t states = std::size_t{1} << n;
650 const double inf = std::numeric_limits<double>::infinity();
651 std::vector<std::vector<double>> dp(states, std::vector<double>(n, inf));
652 std::vector<std::vector<int>> parent(states, std::vector<int>(n, -1));
653
654 for (std::size_t i = 0; i < n; ++i) {
655 dp[std::size_t{1} << i][i] = dist(start, points[i]);
656 }
657
658 for (std::size_t mask = 1; mask < states; ++mask) {
659 for (std::size_t i = 0; i < n; ++i) {
660 if (!(mask & (std::size_t{1} << i)) || !std::isfinite(dp[mask][i])) continue;
661 for (std::size_t j = 0; j < n; ++j) {
662 if (mask & (std::size_t{1} << j)) continue;
663 const std::size_t next = mask | (std::size_t{1} << j);
664 const double cost = dp[mask][i] + dist(points[i], points[j]);
665 if (cost < dp[next][j]) {
666 dp[next][j] = cost;
667 parent[next][j] = static_cast<int>(i);
668 }
669 }
670 }
671 }
672
673 std::size_t last = 0;
674 double best = inf;
675 const std::size_t full = states - 1;
676 for (std::size_t i = 0; i < n; ++i) {
677 if (dp[full][i] < best) {
678 best = dp[full][i];
679 last = i;
680 }
681 }
682
683 std::vector<Pt> ordered(n);
684 std::size_t mask = full;
685 std::size_t current = last;
686 for (std::size_t k = n; k > 0; --k) {
687 ordered[k - 1] = points[current];
688 const int previous = parent[mask][current];
689 mask &= ~(std::size_t{1} << current);
690 if (previous < 0) break;
691 current = static_cast<std::size_t>(previous);
692 }
693 return ordered;
694}

696/**
697 * @brief Reubica puntos uno a uno con la heurística Or-opt.
698 *
699 * Para cada punto prueba a extraerlo e insertarlo en todas las posiciones
700 * posibles, conservando el cambio cuando la longitud total disminuye. Repite
701 * hasta que no haya mejoras o hasta 50 rondas.
702 *
703 * @param points Orden a mejorar (in situ).
704 * @param start Punto de partida para medir la longitud.
705 *
706 * @note Se usa como refinamiento final tras el 2-opt en conjuntos grandes.
707 * @warning La implementación reconstruye el vector completo en cada intento
708 * (O(n³) por ronda); con miles de taladros puede ser costosa.
709 */
710void orOpt(std::vector<Pt>& points, const Pt& start) {
711 if (points.size() < 3) return;
712 bool improved = true;
713 int guard = 0;
714 while (improved && guard++ < 50) {
715 improved = false;
716 for (std::size_t i = 0; i < points.size(); ++i) {
717 std::vector<Pt> candidate = points;
718 const Pt point = candidate[i];
719 candidate.erase(candidate.begin() + static_cast<std::ptrdiff_t>(i));
720
721 double bestLength = pathLengthWithStart(points, start);
722 std::size_t bestPos = i;
723 for (std::size_t pos = 0; pos <= candidate.size(); ++pos) {
724 std::vector<Pt> trial = candidate;
725 trial.insert(trial.begin() + static_cast<std::ptrdiff_t>(pos), point);
726 const double length = pathLengthWithStart(trial, start);
727 if (length < bestLength - 1e-9) {
728 bestLength = length;
729 bestPos = pos;
730 }
731 }
732 if (bestPos != i) {
733 candidate.insert(candidate.begin() + static_cast<std::ptrdiff_t>(bestPos), point);
734 points = std::move(candidate);
735 improved = true;
736 }
737 }
738 }
739}

743/**
744 * @brief Calcula el recorrido XY total de un trabajo de taladrado.
745 *
746 * Suma los viajes entre taladros consecutivos de cada herramienta, el salto
747 * entre herramientas y los desplazamientos de las ranuras (inicio y fin). No
748 * incluye movimientos en Z ni retracciones.
749 *
750 * @param job Trabajo a medir.
751 * @param start Punto de partida (por defecto el origen de máquina).
752 * @return Distancia total en mm.
753 *
754 * @note Se usa para comparar el efecto de las optimizaciones de orden.
755 * @note El punto de continuación tras una herramienta es su último taladro;
756 * si la herramienta no tiene taladros se mantiene el punto anterior.
757 */
758double drillTravelDistance(const DrillJob& job, const Pt& start) {
759 double total = 0.0;
760 Pt current = start;
761 for (const DrillTool& tool : job.tools) {
762 total += holesDistance(tool.holes, current);
763 if (!tool.holes.empty()) current = tool.holes.back();
764 for (const auto& [slotStart, slotEnd] : tool.slotSegments) {
765 total += dist(current, slotStart);
766 total += dist(slotStart, slotEnd);
767 current = slotEnd;
768 }
769 }
770 return total;
771}

773/**
774 * @brief Reordena taladros y ranuras del trabajo para reducir los rápidos.
775 *
776 * Con `DrillOrder::AsIs` no hace nada. En otro caso, para cada herramienta:
777 * - hasta 18 taladros usa el TSP exacto (Held-Karp);
778 * - con más, aplica vecino más cercano + 2-opt + Or-opt.
779 *
780 * Después ordena las ranuras por cercanía al punto actual y actualiza el punto
781 * de partida con el extremo final de cada elemento para que la herramienta
782 * siguiente continúe cerca.
783 *
784 * @param job Trabajo a modificar in situ.
785 * @param order Estrategia de orden.
786 * @param start Punto de partida (origen o posición real de la máquina).
787 *
788 * @note El orden se optimiza herramienta por herramienta, no globalmente: el
789 * orden de las herramientas lo define el archivo original.
790 * @note Las ranuras se ordenan con comparador que captura `current` por
791 * referencia; el valor no cambia durante el `std::sort`, por lo que el
792 * criterio es estable.
793 * @warning El comparador no es un orden débil estricto ante distancias
794 * exactamente iguales, aunque al comparar sólo con `<` no dispara la
795 * mayoría de las aserciones de las bibliotecas estándar.
796 */
797void optimizeDrillOrder(DrillJob& job, DrillOrder order, const Pt& start) {
798 if (order == DrillOrder::AsIs) return;
799 Pt current = start;
800 for (DrillTool& tool : job.tools) {
801 if (tool.holes.size() <= 18) {
802 tool.holes = exactTsp(tool.holes, current);
803 } else {
804 sortNearest(tool.holes, current);
805 twoOpt(tool.holes, current);
806 orOpt(tool.holes, current);
807 }
808 if (!tool.holes.empty()) current = tool.holes.back();
809
810 std::sort(tool.slotSegments.begin(), tool.slotSegments.end(),
811 [&current](const auto& a, const auto& b) {
812 return dist(current, a.first) < dist(current, b.first);
813 });
814 for (const auto& [slotStart, slotEnd] : tool.slotSegments) {
815 (void)slotStart;
816 current = slotEnd;
817 }
818 }
819}

821/**
822 * @brief Reordena los recorridos de geometría eligiendo el extremo más cercano.
823 *
824 * Greedy por vecino más cercano sobre los extremos: para cada paso compara la
825 * distancia al inicio y al final de cada recorrido pendiente, elige el menor e
826 * invierte el recorrido si conviene empezar por su extremo final. No aplica
827 * 2-opt porque los recorridos no son puntos intercambiables.
828 *
829 * @param job Trabajo a modificar in situ (recorridos de una herramienta).
830 * @param start Punto de partida (normalmente el origen).
831 *
832 * @note Los recorridos con menos de 2 puntos se descartan del resultado, ya que
833 * no se pueden mecanizar.
834 * @note Es una heurística: no garantiza el mínimo global del recorrido en vacío.
835 */
836void optimizePathOrder(PathJob& job, const Pt& start) {
837 if (job.paths.size() < 2) return;
838
839 std::vector<std::vector<Pt>> remaining = job.paths;
840 std::vector<std::vector<Pt>> ordered;
841 ordered.reserve(job.paths.size());
842 Pt current = start;
843
844 while (!remaining.empty()) {
845 std::size_t best = 0;
846 bool reverse = false;
847 double bestDistance = std::numeric_limits<double>::infinity();
848 for (std::size_t i = 0; i < remaining.size(); ++i) {
849 if (remaining[i].size() < 2) continue;
850 const double dStart = dist(current, remaining[i].front());
851 const double dEnd = dist(current, remaining[i].back());
852 if (dStart < bestDistance) {
853 bestDistance = dStart;
854 best = i;
855 reverse = false;
856 }
857 if (dEnd < bestDistance) {
858 bestDistance = dEnd;
859 best = i;
860 reverse = true;
861 }
862 }
863 std::vector<Pt> path = remaining[best];
864 if (reverse) std::reverse(path.begin(), path.end());
865 current = path.back();
866 ordered.push_back(std::move(path));
867 remaining.erase(remaining.begin() + static_cast<std::ptrdiff_t>(best));
868 }
869 job.paths = std::move(ordered);
870}

934/**
935 * @brief Genera el programa de dosificación de pasta de soldar.
936 *
937 * Cabecera con boquilla por defecto y límites, unidades y modo absoluto; luego,
938 * por cada punto (en orden directo o invertido si `job.reverse`), si hay tabla
939 * de boquillas completa compara el índice con el anterior y, al cambiar, emite
940 * un comentario con el diámetro y —si `includeToolChange`— una pausa `M0` para
941 * el cambio manual. Cada punto se ejecuta como: rápido XY, rápido a `zMove`,
942 * bajada a `zCut` con `feedZ`, pausa `G04` según boquilla y retracción a
943 * `zEnd`. Cierra con `G00 Z zMove`, `M05` y `M02`.
944 *
945 * @param job Puntos, tabla de boquillas, pausas y límites.
946 * @param opt Parámetros de máquina y formato.
947 * @return Programa G-code de dosificación.
948 *
949 * @note Si `nozzles` o `nozzleDwells` no tienen exactamente el mismo tamaño que
950 * `points` se ignoran y se usa la boquilla/pausa por defecto, de modo que
951 * un script incompleto no desalinea los datos.
952 * @note El orden invertido se aplica indexando desde el final en lugar de
953 * invertir el vector, así las tablas paralelas siguen alineadas.
954 * @warning `M05` y `M02` se emiten siempre, incluso si el equipo no tiene
955 * husillo; algunos firmwares aceptan ambos comandos sin efecto.
956 * @warning `zCut` se usa como altura de dosificación: si viene con el valor por
957 * defecto de corte (negativo) la boquilla penetrará en la placa.
958 */
959std::string pasteGCode(const PasteJob& job, const GCodeOptions& opt) {
960 std::ostringstream out;
961 out << "(Solder paste dispensing program generated by FlatCAM C++.)\n";
962 writeParameterComments(out, opt);
963 out << "(Nozzle diameter: " << number(job.nozzleDiameter, opt.decimals) << ")\n";
964 if (job.bounds.valid) {
965 out << "(X range: " << number(job.bounds.xmin, opt.decimals) << " ... "
966 << number(job.bounds.xmax, opt.decimals) << ")\n";
967 out << "(Y range: " << number(job.bounds.ymin, opt.decimals) << " ... "
968 << number(job.bounds.ymax, opt.decimals) << ")\n";
969 }
970 out << (opt.units == Units::Inch ? "G20\n" : "G21\n");
971 out << "G90\nG94\n";
972 out << "G00 Z" << number(opt.zStart, opt.decimals) << "\n";
973 int currentNozzle = -1;
974 for (std::size_t step = 0; step < job.points.size(); ++step) {
975 const std::size_t index = job.reverse ? job.points.size() - 1 - step : step;
976 const Pt& p = job.points[index];
977 if (job.nozzles.size() == job.points.size() &&
978 job.nozzles[index] != currentNozzle) {
979 currentNozzle = job.nozzles[index];
980 const double dia = currentNozzle >= 0 &&
981 currentNozzle < static_cast<int>(job.nozzleDiameters.size())
982 ? job.nozzleDiameters[static_cast<std::size_t>(currentNozzle)]
983 : job.nozzleDiameter;
984 out << "(Nozzle T" << (currentNozzle + 1) << " - dia "
985 << number(dia, opt.decimals) << " mm)\n";
986 if (opt.includeToolChange) out << "M0\n";
987 }
988 out << "G00 " << xy(p, opt) << "\n";
989 out << "G00 Z" << number(opt.zMove, opt.decimals) << "\n";
990 out << "G01 Z" << number(opt.zCut, opt.decimals) << " F"
991 << number(opt.feedZ, opt.feedDecimals) << "\n";
992 const double dwell = job.nozzleDwells.size() == job.points.size()
993 ? job.nozzleDwells[index]
994 : opt.dwellTime;
995 if (dwell > 0.0) out << "G04 P" << number(dwell, opt.feedDecimals) << "\n";
996 out << "G01 Z" << number(opt.zEnd, opt.decimals) << " F"
997 << number(opt.feedZ, opt.feedDecimals) << "\n";
998 }
999 out << "G00 Z" << number(opt.zMove, opt.decimals) << "\n";
1000 if (opt.endXY) out << "G00 " << xy(*opt.endXY, opt) << "\n";
1001 out << "M05\nM02\n";
1002 return out.str();

1006 * @brief Genera el G-code de una herramienta que sigue una lista de recorridos.
1007 *
1008 * Despacha según el postprocesador:
1009 * - láser → `laserPathsGCode`;
1010 * - `Hpgl` → `IN; SP1; PU/PD; PA...` con coordenadas en mm;
1011 * - `LineXyz` → `G00/G01` con `X`, `Y` y `Z` explícitos en cada movimiento;

1014 * En el camino convencional se escribe cabecera con diámetro, avances, Z y
1015 * límites, unidades, `G90/G94`, husillo y cambio de herramienta opcional.

1017 * `zCut != 0` se generan escalones desde 0 hasta `zCut`; si no, una sola

1020 * está a menos de `connectTolerance` del final previo).

1023 * @param opt Parámetros de máquina, unidades y postprocesador.
1024 * @return Programa G-code.

1027 * @note `connectPaths` sólo evita la retracción y el reposicionamiento; no
1028 * agrega ningún movimiento de enlace, por lo que el tramo entre el final
1029 * anterior y el inicio actual se mecaniza en línea recta.
1030 * @warning Con `depthPerPass > 0` cada nivel repite el recorrido completo,
1031 * incluso los ya cortados, lo que alarga el programa pero mantiene la
1032 * carga de fresado constante (mismo criterio que FlatCAM).
