20/// @brief Acumula `b` dentro de `target`, ignorando cajas inválidas.
21void addBounds(BBox& target, const BBox& b) {
22 if (!b.valid) return;
23 if (!target.valid) {
24 target = b;
25 return;
26 }
27 target.xmin = std::min(target.xmin, b.xmin);
28 target.ymin = std::min(target.ymin, b.ymin);
29 target.xmax = std::max(target.xmax, b.xmax);
30 target.ymax = std::max(target.ymax, b.ymax);
31}

33/// @brief Longitud del contorno abierto de `ring` (sin cerrar el último lado).
34double ringLength(const Ring& ring) {
35 double total = 0.0;
36 for (std::size_t i = 1; i < ring.size(); ++i) {
37 total += std::hypot(ring[i].x - ring[i - 1].x, ring[i].y - ring[i - 1].y);
38 }
39 if (ring.size() >= 2) {
40 const Pt& first = ring.front();
41 const Pt& last = ring.back();
42 const double closing = std::hypot(first.x - last.x, first.y - last.y);
43 if (closing > 1e-9) total += closing;
44 }
45 return total;
46}

48/**
49 * @brief Parte un contorno cerrado en tramos abiertos dejando puentes (tabs).
50 *
51 * El contorno se parametriza por longitud de arco `s` y los `count` puentes se
52 * centran en `(k + 0,5) · L / count`, de forma que quedan repartidos
53 * uniformemente y nunca en el vértice inicial. De cada puente se elimina el
54 * intervalo `[centro − ancho/2, centro + ancho/2]` (con ajuste modular si
55 * cruza el origen) y se reconstruye el tramo complementario, insertando los
56 * vértices originales que caen dentro del intervalo para no perder la forma.
57 *
58 * @param closed Anillo cerrado de entrada (puede repetir el primer punto al
59 * final; se elimina la duplicación).
60 * @param width Ancho de cada puente en milímetros.
61 * @param count Cantidad de puentes.
62 * @param totalLength Perímetro aproximado, usado sólo para validar que los
63 * puentes no consuman toda la pieza.
64 * @return Lista de polilíneas abiertas. Si no se puede aplicar puentes
65 * devuelve un único elemento con el anillo original intacto.
66 *
67 * @note Casos de salida temprana: `count <= 0`, `width <= 0`, menos de 4
68 * puntos, longitud no positiva, o `width × count ≥ 90 %` del perímetro.
69 * En todos ellos se devuelve el contorno completo para no destruir la
70 * pieza.
71 * @note La longitud real se recalcula internamente (`L`); `totalLength` sólo
72 * participa del chequeo de saturación.
73 * @warning Si los puentes se solapan (ancho grande respecto de la separación)
74 * el tramo puede quedar degenerado; el límite del 90 % amortigua pero
75 * no elimina el caso cuando `count` es grande y el ancho moderado.
76 */
77std::vector<Ring> splitWithTabs(const Ring& closed, double width, int count, double totalLength,
78 TabSides sides, const std::vector<double>& manualPositions) {
79 std::vector<Ring> out;
80 if ((count <= 0 && manualPositions.empty()) || width <= 0.0 || closed.size() < 4) {
81 out.push_back(closed);
82 return out;
83 }
84 if (totalLength <= 0.0) {
85 out.push_back(closed);
86 return out;
87 }
88
89 Ring pts = closed;
90 if (pts.front().x == pts.back().x && pts.front().y == pts.back().y) pts.pop_back();
91 const std::size_t n = pts.size();
92 if (n < 3) {
93 out.push_back(closed);
94 return out;
95 }
96
97 std::vector<double> s(n + 1, 0.0);
98 for (std::size_t i = 1; i <= n; ++i) {
99 const Pt& a = pts[i - 1];
100 const Pt& b = pts[i % n];
101 s[i] = s[i - 1] + std::hypot(b.x - a.x, b.y - a.y);
102 }
103 const double length = s[n];
104
105 auto pointAt = [&](double t) {
106 t = std::fmod(t, length);
107 if (t < 0.0) t += length;
108 auto it = std::upper_bound(s.begin(), s.end(), t);
109 std::size_t i = static_cast<std::size_t>(std::max<std::ptrdiff_t>(1, it - s.begin()));
110 if (i > n) i = n;
111 const double segLen = s[i] - s[i - 1];
112 const double f = segLen > 0.0 ? (t - s[i - 1]) / segLen : 0.0;
113 const Pt& a = pts[i - 1];
114 const Pt& b = pts[i % n];
115 return Pt{a.x + (b.x - a.x) * f, a.y + (b.y - a.y) * f};
116 };
117
118 std::vector<double> centers;
119 if (!manualPositions.empty()) {
120 centers.reserve(manualPositions.size());
121 for (double fraction : manualPositions)
122 centers.push_back(std::clamp(fraction, 0.0, 1.0) * length);
123 } else {
124 for (int k = 0; k < count; ++k)
125 centers.push_back((k + 0.5) * length / count);
126 if (sides != TabSides::All) {
127 BBox box;
128 for (const Pt& p : pts) {
129 if (!box.valid) box = BBox{p.x, p.y, p.x, p.y};
130 else {
131 box.xmin = std::min(box.xmin, p.x);
132 box.ymin = std::min(box.ymin, p.y);
133 box.xmax = std::max(box.xmax, p.x);
134 box.ymax = std::max(box.ymax, p.y);
135 }
136 }
137 const double tolerance = std::max(0.01, length * 0.001);
138 std::vector<double> filtered;
139 for (double center : centers) {
140 const Pt p = pointAt(center);
141 const bool topBottom = std::fabs(p.y - box.ymax) <= tolerance ||
142 std::fabs(p.y - box.ymin) <= tolerance;
143 const bool leftRight = std::fabs(p.x - box.xmin) <= tolerance ||
144 std::fabs(p.x - box.xmax) <= tolerance;
145 if ((sides == TabSides::TopBottom && topBottom) ||
146 (sides == TabSides::LeftRight && leftRight))
147 filtered.push_back(center);
148 }
149 if (!filtered.empty()) centers = std::move(filtered);
150 }
151 }
152 if (width * static_cast<double>(centers.size()) >= length * 0.9) {
153 out.push_back(closed);
154 return out;
155 }
156
157 struct Span {
158 double start;
159 double end;
160 };
161 std::vector<Span> tabs;
162 tabs.reserve(centers.size());
163 for (double center : centers) {
164 double start = center - width / 2.0;
165 double end = center + width / 2.0;
166 while (start < 0.0) {
167 start += length;
168 end += length;
169 }
170 if (end > length) end -= length;
171 tabs.push_back({start, end});
172 }
173 std::sort(tabs.begin(), tabs.end(), [](const Span& a, const Span& b) { return a.start < b.start; });
174
175 for (std::size_t k = 0; k < tabs.size(); ++k) {
176 const Span& tab = tabs[k];
177 const Span& next = tabs[(k + 1) % tabs.size()];
178 double a = tab.end;
179 double b = next.start;
180 if (b <= a) b += length;
181
182 Ring seg;
183 seg.push_back(pointAt(a));
184 std::vector<std::pair<double, Pt>> candidates;
185 for (std::size_t i = 0; i < n; ++i) {
186 for (double offset : {0.0, length}) {
187 const double t = s[i] + offset;
188 if (t > a && t < b) candidates.emplace_back(t, pts[i]);
189 }
190 }
191 std::sort(candidates.begin(), candidates.end(),
192 [](const auto& p1, const auto& p2) { return p1.first < p2.first; });
193 for (const auto& [t, point] : candidates) {
194 (void)t;
195 seg.push_back(point);
196 }
197 seg.push_back(pointAt(b));
198 if (seg.size() >= 2) out.push_back(std::move(seg));
199 }
200 return out;
201}

203/// @brief Agrega `ring` como recorrido, invirtiendo el sentido si `invert`.
204///
205/// Rechaza geometrías inválidas o vacías y acumula la longitud real del
206/// recorrido en `result.totalLength`.
207void appendPath(CutoutResult& result, const Ring& ring, bool invert) {
208 Ring pts = ring;
209 if (invert) std::reverse(pts.begin(), pts.end());
210 Geom path = Geom::line(pts);
211 if (!path.valid() || path.empty()) return;
212 result.totalLength += path.length();
213 result.paths.push_back(std::move(path));
214}

218/**
219 * @brief Calcula los recorridos de corte de un contorno de placa.
220 *
221 * El desplazamiento del contorno se calcula como:
222 * - corte externo: `margin + Ø/2` (la fresa pasa por fuera, dejando la pieza
223 * al tamaño nominal más el margen);
224 * - corte interno: `margin − Ø/2` (recorte dentro del contorno).
225 *
226 * Luego une el contorno consigo mismo (`unaryUnion`) para eliminar solapes y
227 * multiplicidades, aplica el buffer, recorta contra `options.area` si existe y
228 * extrae los recorridos: primero el contorno exterior partido por puentes y,
229 * si `includeHoles`, también los agujeros internos partidos de la misma forma.
230 *
231 * @param outline Contorno de la placa (sólidos del Gerber de borde).
232 * @param options Herramienta, margen, tipo de corte, puentes, área e inversión.
233 * @return Envolvente desplazada, recorridos del centro de la fresa, longitud
234 * total y límites de la envolvente.
235 *
236 * @note Si `unaryUnion` falla o queda vacía se usa el contorno original, de
237 * modo que el corte sigue siendo posible con geometría problemática.
238 * @note `includeHoles` está desactivado por defecto: los contornos internos
239 * del trazo de borde no deben cortarse salvo pedido explícito.
240 * @note `totalLength` cuenta cada tramo por separado; no incluye los rápidos
241 * entre tramos.
242 * @note `ringLength` suma el lado de cierre cuando el anillo no repite el
243 * primer punto, de modo que el perímetro usado para repartir los puentes
244 * coincide con el real.
245 * @warning Con `external == false` y un margen menor que el radio, el buffer
246 * negativo puede vaciar la geometría y el resultado queda vacío.
247 */
248CutoutResult cutoutGeometry(const Geom& outline, const CutoutOptions& options) {
249 CutoutResult result;
250 if (!outline.valid() || outline.empty()) return result;
251
252 const double radius = options.toolDiameter / 2.0;
253 const double offset = options.external ? options.margin + radius : options.margin - radius;
254
255 Geom merged = outline.unaryUnion();
256 if (!merged.valid() || merged.empty()) merged = outline;
257
258 // GEOS redondea las esquinas por defecto; para una placa hay que conservar
259 // el ángulo recto del contorno usando uniones en inglete.
260 Geom envelope = offset == 0.0
261 ? merged
262 : merged.bufferStyle(offset, std::max(2, options.quadrantSegments),
263 GEOSBUF_CAP_ROUND, GEOSBUF_JOIN_MITRE, 10.0);
264 if (!envelope.valid() || envelope.empty()) return result;
265
266 if (options.area) {
267 envelope = envelope.intersection(Geom::box(options.area->xmin, options.area->ymin,
268 options.area->xmax, options.area->ymax));
269 if (!envelope.valid() || envelope.empty()) return result;
270 }
271 result.envelope = envelope;
272
273 for (const PolygonContours& pc : envelope.polygons()) {
274 const double outerLength = ringLength(pc.outer);
275 for (const Ring& segment : splitWithTabs(pc.outer, options.tabWidth, options.tabCount,
276 outerLength, options.tabSides,
277 options.tabPositions)) {
278 appendPath(result, segment, options.invert);
279 }
280 if (options.includeHoles) {
281 for (const Ring& holeRing : pc.holes) {
282 const double holeLength = ringLength(holeRing);
283 for (const Ring& segment : splitWithTabs(holeRing, options.tabWidth,
284 options.tabCount, holeLength,
285 options.tabSides,
286 options.tabPositions)) {
287 appendPath(result, segment, options.invert);
288 }
289 }
290 }
291 }
292
293 addBounds(result.bounds, envelope.bounds());
294 return result;
295}
