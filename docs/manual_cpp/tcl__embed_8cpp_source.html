37/**
38 * @brief Estado compartido por todos los comandos Tcl registrados.
39 *
40 * Mantiene el último Gerber y Excellon abiertos, sus banderas de validez y el
41 * trabajo activo (`PathJob` o `DrillJob`) que exportará `export_gcode`. Los dos

44struct ShellState {

46 bool hasGerber = false;

52 bool hasDrillJob = false;
53};

55/**
56 * @brief Devuelve un argumento Tcl como cadena, o `nullptr` si falta.
57 * @param objv Vector de objetos del comando.
58 * @param index Posición a leer.
59 * @param count Cantidad de argumentos (`objc`).
60 * @return Puntero interno al texto (válido mientras viva el objeto Tcl).
61 *
62 * @note Es código muerto en la versión actual: ningún comando lo usa y puede
63 * generar un aviso de función no utilizada según el compilador.
64 */
65const char* arg(Tcl_Interp*, Tcl_Obj* const objv[], int index, int count) {
66 return index < count ? Tcl_GetString(objv[index]) : nullptr;
67}

69/**
70 * @brief Lee un argumento numérico Tcl con valor de reserva.
71 *
72 * Usa `Tcl_GetDoubleFromObj`, que puede dejar un mensaje de error en el
73 * intérprete si el objeto no es numérico; en ese caso se devuelve el valor de
74 * reserva y el mensaje residual se sobrescribe al fijar el resultado del
75 * comando.
76 *
77 * @param interp Intérprete Tcl.
78 * @param objv Vector de objetos del comando.
79 * @param index Posición a leer.
80 * @param count Cantidad de argumentos.
81 * @param fallback Valor a devolver si falta el argumento o no es numérico.
82 * @return Número convertido o `fallback`.
83 */
84double numberArg(Tcl_Interp* interp, Tcl_Obj* const objv[], int index, int count, double fallback) {
85 if (index >= count) return fallback;
86 double value = fallback;
87 if (Tcl_GetDoubleFromObj(interp, objv[index], &value) != TCL_OK) return fallback;
88 return value;
89}

91/**
92 * @brief Reemplaza el trabajo de recorridos y desactiva el de taladrado.
93 * @param state Estado a modificar.
94 * @param paths Recorridos producidos por la herramienta.
95 * @param bounds Límites del trabajo.
96 * @param tool Diámetro de herramienta informativo.
97 *
98 * @note Los recorridos con menos de 2 puntos se descartan.
99 */
100void setPaths(ShellState& state, const std::vector<Geom>& paths, const BBox& bounds, double tool) {
101 state.paths = PathJob{};
102 state.paths.bounds = bounds;
103 state.paths.toolDiameter = tool;
104 for (const Geom& g : paths) {
105 Ring ring = g.coords();
106 if (ring.size() >= 2) state.paths.paths.emplace_back(ring.begin(), ring.end());
107 }
108 state.hasPathJob = true;
109 state.hasDrillJob = false;
110}

112/**
113 * @brief Implementa `open_gerber <archivo>`.
114 *
115 * Parsea el Gerber con `GerberParser` y lo deja como objeto activo. Devuelve la
116 * ruta como resultado Tcl.
117 *
118 * @param data Puntero al `ShellState`.
119 * @param interp Intérprete Tcl.
120 * @param objc Cantidad de argumentos.
121 * @param objv Argumentos (`open_gerber`, ruta).
122 * @return `TCL_OK` o `TCL_ERROR` con el mensaje de la excepción.
123 *
124 * @note No reinicia el trabajo de taladrado previo; ver la advertencia de
125 * `runTclScript`.
126 */
127int cmdOpenGerber(ClientData data, Tcl_Interp* interp, int objc, Tcl_Obj* const objv[]) {
128 auto* state = static_cast<ShellState*>(data);
129 if (objc < 2) {
130 Tcl_SetObjResult(interp, Tcl_NewStringObj("usage: open_gerber file", -1));
131 return TCL_ERROR;
132 }
133 try {
134 state->gerber = GerberParser().parseFile(Tcl_GetString(objv[1]));
135 state->hasGerber = true;
136 Tcl_SetObjResult(interp, Tcl_NewStringObj(Tcl_GetString(objv[1]), -1));
137 return TCL_OK;
138 } catch (const std::exception& e) {
139 Tcl_SetResult(interp, const_cast<char*>(e.what()), TCL_VOLATILE);
140 return TCL_ERROR;
141 }
142}

176/**
177 * @brief Implementa `isolate <Ø> <pasadas> <solape%>`.
178 *
179 * Genera el aislamiento del Gerber activo y deja el resultado como `PathJob`.
180 *
181 * @param data Puntero al `ShellState`.
182 * @param interp Intérprete Tcl.
183 * @param objc Cantidad de argumentos.
184 * @param objv Argumentos posicionales opcionales (Ø 0,2; pasadas 1; solape 0).
185 * @return `TCL_OK` con la cantidad de recorridos, o `TCL_ERROR` si no hay
186 * Gerber cargado.
187 *
188 * @warning No captura excepciones de GEOS: si el cálculo falla, la excepción
189 * atraviesa el callback de Tcl (comportamiento indefinido) en lugar de
190 * convertirse en `TCL_ERROR`.
191 */
192int cmdIsolate(ClientData data, Tcl_Interp* interp, int objc, Tcl_Obj* const objv[]) {
193 auto* state = static_cast<ShellState*>(data);
194 if (!state->hasGerber) {
195 Tcl_SetObjResult(interp, Tcl_NewStringObj("no gerber loaded", -1));
196 return TCL_ERROR;
197 }
198 IsolationOptions opt;
199 opt.toolDiameter = numberArg(interp, objv, 1, objc, 0.2);
200 opt.passes = static_cast<int>(numberArg(interp, objv, 2, objc, 1));
201 opt.overlap = numberArg(interp, objv, 3, objc, 0.0) / 100.0;
202 const IsolationResult result = isolateCopper(state->gerber.solid, opt);
203 setPaths(*state, result.paths, result.bounds, opt.toolDiameter);
204 Tcl_SetObjResult(interp, Tcl_NewIntObj(static_cast<int>(result.paths.size())));
205 return TCL_OK;
206}

241 * @brief Implementa `ncc <Ø> <paso%>`.
242 *
243 * Despeja el cobre sobrante usando como área de trabajo el bounding box del

246 *
247 * @param data Puntero al `ShellState`.
248 * @param interp Intérprete Tcl.
249 * @param objc Cantidad de argumentos.
250 * @param objv Argumentos posicionales (Ø 0,5; paso 50 %).
251 * @return `TCL_OK` con la cantidad de recorridos, o `TCL_ERROR` sin Gerber.
252 *

255 * @warning No captura excepciones de GEOS.

257int cmdNcc(ClientData data, Tcl_Interp* interp, int objc, Tcl_Obj* const objv[]) {
258 auto* state = static_cast<ShellState*>(data);
259 if (!state->hasGerber) {
260 Tcl_SetObjResult(interp, Tcl_NewStringObj("no gerber loaded", -1));
261 return TCL_ERROR;
262 }
263 NccOptions opt;
264 opt.toolDiameter = numberArg(interp, objv, 1, objc, 0.5);
265 opt.stepOver = numberArg(interp, objv, 2, objc, 50.0) / 100.0;
266 const BBox b = state->gerber.bounds.expanded(1.0);
267 const Geom area = b.valid ? Geom::box(b.xmin, b.ymin, b.xmax, b.ymax) : Geom();

269 setPaths(*state, result.paths, result.bounds, opt.toolDiameter);
270 Tcl_SetObjResult(interp, Tcl_NewIntObj(static_cast<int>(result.paths.size())));
271 return TCL_OK;

303/**
304 * @brief Implementa `mill <Ø>`: fresado siguiendo los recorridos del Gerber.
305 *
306 * Recorre `gerber.follows` y convierte en `PathJob` los elementos que sean
307 * `LineString` o `LinearRing` de primer nivel.
308 *
309 * @param data Puntero al `ShellState`.
310 * @param interp Intérprete Tcl.
311 * @param objc Cantidad de argumentos.
312 * @param objv Argumento opcional con el diámetro (0,2 por defecto).
313 * @return `TCL_OK` con la cantidad de recorridos, o `TCL_ERROR` sin Gerber.
314 *
315 * @warning A diferencia de `collectLineStrings` de la CLI, aquí no se recurre
316 * en `MultiLineString` ni `GeometryCollection`: esos recorridos se
317 * pierden silenciosamente y el fresado queda incompleto.
318 */
319int cmdMill(ClientData data, Tcl_Interp* interp, int objc, Tcl_Obj* const objv[]) {
320 auto* state = static_cast<ShellState*>(data);
321 if (!state->hasGerber) {
322 Tcl_SetObjResult(interp, Tcl_NewStringObj("no gerber loaded", -1));
323 return TCL_ERROR;
324 }
325 std::vector<std::vector<Pt>> lines;
326 // Recorre colecciones anidadas (MULTILINESTRING, GEOMETRYCOLLECTION) para
327 // no perder recorridos al fresar.
328 std::function<void(const Geom&)> collect = [&](const Geom& g) {
329 if (!g.valid() || g.empty()) return;
330 const int type = g.typeId();
331 if (type == GEOS_LINESTRING || type == GEOS_LINEARRING) {
332 Ring ring = g.coords();
333 if (ring.size() >= 2) lines.emplace_back(ring.begin(), ring.end());
334 return;
335 }
336 if (type == GEOS_MULTILINESTRING || type == GEOS_GEOMETRYCOLLECTION ||
337 type == GEOS_MULTIPOLYGON) {
338 for (int i = 0; i < g.numGeometries(); ++i) collect(g.geometryN(i));
339 }
340 };
341 for (const Geom& g : state->gerber.follows) collect(g);
342 state->paths = PathJob{};
343 state->paths.paths = std::move(lines);
344 state->paths.bounds = state->gerber.bounds;
345 state->paths.toolDiameter = numberArg(interp, objv, 1, objc, 0.2);
346 state->hasPathJob = true;
347 state->hasDrillJob = false;
348 Tcl_SetObjResult(interp, Tcl_NewIntObj(static_cast<int>(state->paths.paths.size())));
349 return TCL_OK;
350}

432/**
433 * @brief Implementa `export_gcode <archivo>`.
434 *
435 * Exporta el trabajo activo: si hay `DrillJob` genera taladrado y, si no, los
436 * recorridos de geometría. Usa `GCodeOptions` por defecto (milímetros,
437 * preprocesador `Default`, Z estándar).
438 *
439 * @param data Puntero al `ShellState`.
440 * @param interp Intérprete Tcl.
441 * @param objc Cantidad de argumentos.
442 * @param objv Argumentos (`export_gcode`, ruta).
443 * @return `TCL_OK`, o `TCL_ERROR` si no hay trabajo, falta la ruta o no se
444 * puede escribir el archivo.
445 *
446 * @warning No hay forma de pasar parámetros de máquina desde Tcl: avances, Z,
447 * husillo y postprocesador quedan siempre en los valores por defecto
448 * de `GCodeOptions`.
449 */
450int cmdExportGcode(ClientData data, Tcl_Interp* interp, int objc, Tcl_Obj* const objv[]) {
451 auto* state = static_cast<ShellState*>(data);
452 if (objc < 2) {
453 Tcl_SetObjResult(interp, Tcl_NewStringObj("usage: export_gcode file", -1));
454 return TCL_ERROR;
455 }
456 GCodeOptions opt;
457 std::string code;
458 if (state->hasDrillJob) code = drillingGCode(state->drills, opt);
459 else if (state->hasPathJob) code = pathsGCode(state->paths, opt);
460 else {
461 Tcl_SetObjResult(interp, Tcl_NewStringObj("nothing to export", -1));
462 return TCL_ERROR;
463 }
464 std::ofstream out(Tcl_GetString(objv[1]), std::ios::binary);
465 if (!out) {
466 Tcl_SetObjResult(interp, Tcl_NewStringObj("cannot write file", -1));
467 return TCL_ERROR;
468 }
469 out << code;
470 return TCL_OK;
471}

495 return TCL_OK;

500/**
501 * @brief Crea el intérprete Tcl, registra los comandos y ejecuta el script.
502 *
503 * Flujo: `Tcl_FindExecutable`, `Tcl_CreateInterp`, `Tcl_Init` (un fallo sólo se
504 * informa como advertencia, porque los comandos del núcleo siguen
505 * disponibles), registro de los doce comandos CAM con `&state` como
506 * `ClientData` y ejecución:
507 * - con `path` vacío o `-`: REPL por `stdin`, evaluando línea por línea;
508 * - en otro caso: `Tcl_EvalFile` del script completo.
509 *
510 * @param path Ruta del script o cadena vacía (o `-`) para modo interactivo.
511 * @return 0 en éxito; 1 si no se puede crear el intérprete o el script falla.
512 *
513 * @note El estado vive en la pila de esta función y el intérprete se destruye
514 * antes de salir, por lo que ningún comando conserva punteros colgantes.
515 * @note En el REPL los errores se informan y el bucle continúa; en modo script
516 * el primer error aborta con código 1.
517 * @warning El REPL evalúa una línea por vez, de modo que una definición
518 * multilínea (`proc f {} {` … `}`) falla por script incompleto.
519 * @warning `Tcl_Init` depende de `TCL_LIBRARY` o del framework detectado; si el
520 * entorno no lo encuentra, `proc`, `if` y `expr` no estarán
521 * disponibles aunque la advertencia sugiera lo contrario.
522 */
523int runTclScript(const std::string& path) {
