42/**
43 * @brief Copia la secuencia de coordenadas de un anillo GEOS a un Ring.
44 * @param ring Anillo o secuencia GEOS de origen.
45 * @return Vértices en orden; vector vacío si no hay secuencia de coordenadas
46 * o si GEOS no puede informar su tamaño.
47 * @note El último punto puede repetir el primero: GEOS cierra los anillos y
48 * esta función no elimina ese duplicado.
49 */
50Ring readRing(const GEOSGeometry* ring) {
51 Ring out;
52 const GEOSCoordSequence* cs = GEOSGeom_getCoordSeq(ring);
53 if (!cs) return out;
54 unsigned int n = 0;
55 if (!GEOSCoordSeq_getSize(cs, &n)) return out;
56 out.reserve(n);
57 for (unsigned int i = 0; i < n; ++i) {
58 double x = 0.0;
59 double y = 0.0;
60 if (GEOSCoordSeq_getXY(cs, i, &x, &y)) out.push_back({x, y});
61 }
62 return out;
63}

65/**
66 * @brief Construye un LinearRing GEOS cerrado a partir de puntos sueltos.
67 *
68 * Copia los puntos y agrega el primero al final si el anillo no venía cerrado.
69 * La propiedad de la secuencia y del anillo pasa al llamador.
70 * @param pts Vértices del contorno (sin repetir el primero).
71 * @return Anillo nuevo (debe liberarse con GEOSGeom_destroy).
72 * @throw std::invalid_argument si hay menos de 3 puntos.
73 * @throw std::runtime_error si GEOS falla al crear la secuencia o el anillo.
74 */
75GEOSGeometry* makeRing(const Ring& pts) {
76 if (pts.size() < 3) throw std::invalid_argument("ring needs at least 3 points");
77 Ring closed = pts;
78 if (closed.front().x != closed.back().x || closed.front().y != closed.back().y) {
79 closed.push_back(closed.front());
80 }
81 GEOSCoordSequence* seq = GEOSCoordSeq_create(static_cast<unsigned int>(closed.size()), 2);
82 if (!seq) throw std::runtime_error("GEOSCoordSeq_create failed");
83 for (std::size_t i = 0; i < closed.size(); ++i) {
84 if (!GEOSCoordSeq_setXY(seq, static_cast<unsigned int>(i), closed[i].x, closed[i].y)) {
85 GEOSCoordSeq_destroy(seq);
86 throw std::runtime_error("GEOSCoordSeq_setXY failed");
87 }
88 }
89 GEOSGeometry* ring = GEOSGeom_createLinearRing(seq);
90 if (!ring) {
91 GEOSCoordSeq_destroy(seq);
92 throw std::runtime_error("GEOSGeom_createLinearRing failed");
93 }
94 return ring;
95}

97/// Empaqueta el functor de transformación para el callback C de GEOS.
98struct TransformData {
99 const std::function<Pt(Pt)>* fn;
100};

102/**
103 * @brief Callback de GEOSGeom_transformXY: aplica el functor punto a punto.
104 *
105 * Modifica las coordenadas in situ y devuelve 1 para que GEOS continúe.
106 * @param x Coordenada X de entrada y salida.
107 * @param y Coordenada Y de entrada y salida.
108 * @param userdata Puntero a TransformData con el std::function a aplicar.
109 * @return Siempre 1; GEOS interpreta otro valor como aborto de la operación.
110 */
111int transformCallback(double* x, double* y, void* userdata) {
112 auto* data = static_cast<TransformData*>(userdata);
113 const Pt p = (*data->fn)({*x, *y});
114 *x = p.x;
115 *y = p.y;
116 return 1;
117}

223/**
224 * @brief Crea un polígono con contorno exterior y agujeros.
225 *
226 * Sólo usa los agujeros con 3 o más puntos; si el exterior tiene menos de 3
227 * devuelve un polígono vacío. Ante cualquier fallo libera los anillos ya
228 * creados antes de propagar la excepción.
229 * @param shell Contorno exterior.
230 * @param holes Agujeros (opcional).
231 * @throw std::runtime_error si GEOS no puede crear el polígono.
232 * @note No valida auto-intersecciones ni la orientación de los anillos.
233 */
234Geom Geom::polygon(const Ring& shell, const std::vector<Ring>& holes) {
235 initGeosRuntime();
236 if (shell.size() < 3) return emptyPolygon();
237 GEOSGeometry* shellRing = makeRing(shell);
238 std::vector<GEOSGeometry*> holeRings;
239 holeRings.reserve(holes.size());
240 try {
241 for (const Ring& h : holes) {
242 if (h.size() >= 3) holeRings.push_back(makeRing(h));
243 }
244 } catch (...) {
245 for (GEOSGeometry* h : holeRings) GEOSGeom_destroy(h);
246 GEOSGeom_destroy(shellRing);
247 throw;
248 }
249 GEOSGeometry* g = GEOSGeom_createPolygon(
250 shellRing, holeRings.empty() ? nullptr : holeRings.data(),
251 static_cast<unsigned int>(holeRings.size()));
252 if (!g) {
253 GEOSGeom_destroy(shellRing);
254 for (GEOSGeometry* h : holeRings) GEOSGeom_destroy(h);
255 throw std::runtime_error("GEOSGeom_createPolygon failed");
256 }
257 return adopt(g);
258}

284/**
285 * @brief Crea un óvalo (obround) de ancho y alto dados.
286 *
287 * Coloca dos círculos de radio min(w,h)/2 separados a lo largo del lado mayor
288 * y toma su envolvente convexa, que es el óvalo exacto.
289 * @param center Centro del óvalo.
290 * @param width Ancho total; si es <= 0 devuelve vacío.
291 * @param height Alto total; si es <= 0 devuelve vacío.
292 * @param quadSegs Segmentos por cuadrante de cada círculo.
293 */
294Geom Geom::obround(const Pt& center, double width, double height, int quadSegs) {
295 if (width <= 0.0 || height <= 0.0) return emptyPolygon();
296 const double r = std::min(width, height) / 2.0;
297 Pt a = center;
298 Pt b = center;
299 if (width > height) {
300 a.x += (width - height) / 2.0;
301 b.x -= (width - height) / 2.0;
302 } else {
303 a.y += (height - width) / 2.0;
304 b.y -= (height - width) / 2.0;
305 }
306 return circle(a, r, quadSegs).unite(circle(b, r, quadSegs)).convexHull();
307}

309/**
310 * @brief Crea un polígono regular inscripto en un círculo.
311 *
312 * Genera los vértices empezando en el ángulo 0 y, si hace falta, rota el
313 * resultado alrededor de @p center.
314 * @param center Centro y eje de rotación.
315 * @param diameter Diámetro del círculo circunscripto.
316 * @param vertices Cantidad de lados (mínimo 3).
317 * @param rotationDeg Rotación final en grados.
318 * @return Polígono; vacío si los parámetros no son utilizables.
319 */
320Geom Geom::regularPolygon(const Pt& center, double diameter, int vertices, double rotationDeg) {
321 if (vertices < 3 || diameter <= 0.0) return emptyPolygon();
322 Ring pts;
323 pts.reserve(static_cast<std::size_t>(vertices));
324 const double r = diameter / 2.0;
325 for (int i = 0; i < vertices; ++i) {
326 const double a = 2.0 * M_PI * static_cast<double>(i) / static_cast<double>(vertices);
327 pts.push_back({center.x + r * std::cos(a), center.y + r * std::sin(a)});
328 }
329 Geom g = polygon(pts);
330 if (rotationDeg != 0.0) g = g.rotateDeg(rotationDeg, center);
331 return g;
332}

350/**
351 * @brief Serializa la geometría a WKT.
352 *
353 * Configura el escritor sin espacios sobrantes y con la precisión pedida, que
354 * es la que determina el tamaño de los proyectos guardados.
355 * @param decimals Dígitos decimales de redondeo.
356 * @return Texto WKT; "POLYGON EMPTY" si el Geom es nulo.
357 * @throw std::runtime_error si el escritor falla.
358 */
359std::string Geom::toWkt(int decimals) const {
360 if (!h_) return "POLYGON EMPTY";
361 initGeosRuntime();
362 GEOSWKTWriter* writer = GEOSWKTWriter_create();
363 if (!writer) throw std::runtime_error("GEOSWKTWriter_create failed");
364 GEOSWKTWriter_setTrim(writer, 1);
365 GEOSWKTWriter_setRoundingPrecision(writer, decimals);
366 char* text = GEOSWKTWriter_write(writer, h_.get());
367 GEOSWKTWriter_destroy(writer);
368 if (!text) throw std::runtime_error("GEOSWKTWriter_write failed");
369 std::string out(text);
370 GEOSFree(text);
371 return out;
372}

374/**
375 * @brief Agrupa varias geometrías en una colección GEOS del tipo pedido.
376 *
377 * Clona cada parte válida, de modo que el resultado no comparte memoria con
378 * las entradas; si no queda ninguna devuelve un polígono vacío.
379 * @param parts Geometrías a agrupar.
380 * @param type Identificador GEOS de colección (por ejemplo GEOS_MULTIPOLYGON).
381 * @throw std::runtime_error si GEOS no puede crear la colección.
382 */
383Geom Geom::collection(const std::vector<Geom>& parts, int type) {
384 std::vector<GEOSGeometry*> raw;
385 raw.reserve(parts.size());
386 for (const Geom& p : parts) {
387 if (p.valid()) raw.push_back(GEOSGeom_clone(p.raw()));
388 }
389 if (raw.empty()) return emptyPolygon();
390 GEOSGeometry* g = GEOSGeom_createCollection(type, raw.data(), static_cast<unsigned int>(raw.size()));
391 if (!g) {
392 for (GEOSGeometry* p : raw) GEOSGeom_destroy(p);
393 throw std::runtime_error("GEOSGeom_createCollection failed");
394 }
395 return adopt(g);
396}

398/**
399 * @brief Unión de muchas geometrías en un solo resultado.
400 *
401 * Descarta nulos y vacíos, devuelve directamente el único elemento si queda
402 * uno solo y, con varios, arma un MultiPolygon y aplica unaryUnion, que es más
403 * eficiente que encadenar uniones binarias.
404 * @param parts Geometrías a unir.
405 * @return Unión; polígono vacío si no hay partes utilizables.
406 * @note Con una sola parte válida se devuelve compartiendo memoria con ella.
407 */
408Geom Geom::unionOf(const std::vector<Geom>& parts) {
409 if (parts.empty()) return emptyPolygon();
410 if (parts.size() == 1) return parts.front();
411 std::vector<Geom> valid;
412 valid.reserve(parts.size());
413 for (const Geom& p : parts) {
414 if (p.valid() && !p.empty()) valid.push_back(p);
415 }
416 if (valid.empty()) return emptyPolygon();
417 if (valid.size() == 1) return valid.front();
418 return Geom::collection(valid, GEOS_MULTIPOLYGON).unaryUnion();
419}

527/**
528 * @brief Descompone la geometría en contornos exteriores con sus huecos.
529 *
530 * Maneja Polygon, MultiPolygon y GeometryCollection (recursivo por parte).
531 * Descarta contornos con menos de 3 puntos y huecos degenerados, de modo que
532 * el resultado sólo contiene polígonos dibujables.
533 * @return Lista de PolygonContours en el orden de las partes.
534 */
535std::vector<PolygonContours> Geom::polygons() const {
536 std::vector<PolygonContours> out;
537 if (!h_ || GEOSisEmpty(h_.get()) == 1) return out;
538
539 switch (GEOSGeomTypeId(h_.get())) {
540 case GEOS_POLYGON: {
541 const GEOSGeometry* ext = GEOSGetExteriorRing(h_.get());
542 if (!ext) break;
543 PolygonContours pc;
544 pc.outer = readRing(ext);
545 if (pc.outer.size() >= 3) {
546 const int n = GEOSGetNumInteriorRings(h_.get());
547 for (int i = 0; i < n; ++i) {
548 const GEOSGeometry* inner = GEOSGetInteriorRingN(h_.get(), i);
549 if (inner) {
550 Ring hole = readRing(inner);
551 if (hole.size() >= 3) pc.holes.push_back(std::move(hole));
552 }
553 }
554 out.push_back(std::move(pc));
555 }
556 break;
557 }
558 case GEOS_MULTIPOLYGON:
559 case GEOS_GEOMETRYCOLLECTION: {
560 const int n = numGeometries();
561 for (int i = 0; i < n; ++i) {
562 const GEOSGeometry* part = GEOSGetGeometryN(h_.get(), i);
563 if (!part) continue;
564 std::vector<PolygonContours> sub = adopt(GEOSGeom_clone(part)).polygons();
565 out.insert(out.end(), sub.begin(), sub.end());
566 }
567 break;
568 }
569 default:
570 break;
571 }
572 return out;
573}

577/**
578 * @brief Aplica un buffer circular uniforme (dilatar o contraer).
579 * @param distance Radio en mm; un valor negativo contrae la geometría.
580 * @param quadSegs Segmentos por cuadrante de la aproximación.
581 * @return Geometría resultante; Geom nulo si el origen es nulo.
582 * @throw std::runtime_error si GEOS falla.

636/**
637 * @brief Par de puntos más cercanos entre dos geometrías.
638 *
639 * Es el dato que usan las herramientas de distancia mínima y de diámetro
640 * óptimo para dibujar la cota sobre el lienzo.
641 * @param other Geometría con la que se mide.
642 * @return {punto en this, punto en other}; ceros si no hay resultado.
643 */
644std::pair<Pt, Pt> Geom::nearestPoints(const Geom& other) const {
645 initGeosRuntime();
646 if (!h_ || !other.h_) return {{0.0, 0.0}, {0.0, 0.0}};
647 GEOSCoordSequence* seq = GEOSNearestPoints(h_.get(), other.h_.get());
648 if (!seq) return {{0.0, 0.0}, {0.0, 0.0}};
649 double x0 = 0.0, y0 = 0.0, x1 = 0.0, y1 = 0.0;
650 GEOSCoordSeq_getXY(seq, 0, &x0, &y0);
651 GEOSCoordSeq_getXY(seq, 1, &x1, &y1);
652 GEOSCoordSeq_destroy(seq);
653 return {{x0, y0}, {x1, y1}};
654}

724/**
725 * @brief Rota la geometría alrededor de un origen.
726 * @param angle Ángulo en grados, sentido antihorario.
727 * @param origin Punto de giro.
728 */
729Geom Geom::rotateDeg(double angle, const Pt& origin) const {
730 const double a = angle * M_PI / 180.0;
731 const double c = std::cos(a);
732 const double s = std::sin(a);
733 return transformXY([c, s, origin](Pt p) {
734 const double dx = p.x - origin.x;
735 const double dy = p.y - origin.y;
736 return Pt{origin.x + dx * c - dy * s, origin.y + dx * s + dy * c};
737 });
738}

740/**
741 * @brief Escala de forma independiente en X e Y respecto de un origen.
742 * @param sx Factor de escala en X.
743 * @param sy Factor de escala en Y.
744 * @param origin Punto fijo de la transformación.
745 */
746Geom Geom::scale(double sx, double sy, const Pt& origin) const {
747 return transformXY([sx, sy, origin](Pt p) {
748 return Pt{origin.x + (p.x - origin.x) * sx, origin.y + (p.y - origin.y) * sy};
749 });
750}
