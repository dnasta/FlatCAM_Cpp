34/// @brief Quita espacios, tabs y saltos de línea de ambos extremos.
35std::string trim(const std::string& s) {
36 const std::size_t first = s.find_first_not_of(" \t\r\n");
37 if (first == std::string::npos) return {};
38 const std::size_t last = s.find_last_not_of(" \t\r\n");
39 return s.substr(first, last - first + 1);
40}

42/**
43 * @brief Divide una línea en tokens separados por espacios o tabs.
44 *
45 * Las comillas dobles agrupan texto con espacios y se eliminan del resultado;
46 * no se admiten comillas escapadas ni llaves. Un token vacío entre comillas
47 * (`""`) no se agrega a la lista.
48 *
49 * @param line Línea ya recortada.
50 * @return Lista de tokens.
51 *
52 * @note Como las comillas se quitan antes de la sustitución de variables, un
53 * argumento entre comillas con `$var` se resuelve igual que sin ellas.
54 */
55std::vector<std::string> splitCommand(const std::string& line) {
56 std::vector<std::string> out;
57 std::string current;
58 bool quoted = false;
59 for (char c : line) {
60 if (c == '"') {
61 quoted = !quoted;
62 continue;
63 }
64 if (!quoted && (c == ' ' || c == '\t')) {
65 if (!current.empty()) {
66 out.push_back(current);
67 current.clear();
68 }
69 } else {
70 current.push_back(c);
71 }
72 }
73 if (!current.empty()) out.push_back(current);
74 return out;
75}

77/**
78 * @brief Sustituye un token que empieza con `$` por el valor de la variable.
79 *
80 * @param token Token original.
81 * @param vars Mapa de variables del script.
82 * @return El valor asociado o una cadena vacía.
83 *
84 * @note El nombre de la variable es todo el token sin el `$`; no se admite
85 * concatenación como `$dir/file.gbr` (se buscaría la variable
86 * `dir/file.gbr`).
87 * @note Una variable inexistente devuelve vacío y no genera error, a diferencia
88 * de Tcl real, que abortaría el script.
89 */
90std::string substitute(const std::string& token, const std::map<std::string, std::string>& vars) {
91 if (token.empty() || token[0] != '$') return token;
92 const auto it = vars.find(token.substr(1));
93 return it == vars.end() ? std::string() : it->second;
94}

96/**
97 * @brief Convierte el token `index` a número, con valor de reserva.
98 * @param tokens Tokens del comando.
99 * @param index Posición a leer.
100 * @param fallback Valor a devolver si falta el token o no es numérico.
101 * @return Número convertido o `fallback`.
102 *
103 * @note Atrapa cualquier excepción de `std::stod`, por lo que un argumento
104 * inválido no interrumpe el script.
105 */
106double toNumber(const std::vector<std::string>& tokens, std::size_t index, double fallback) {
107 if (index >= tokens.size()) return fallback;
108 try {
109 return std::stod(tokens[index]);
110 } catch (...) {
111 return fallback;
112 }
113}

115/**
116 * @brief Estado compartido entre comandos del shell.
117 *
118 * Guarda el último Gerber y Excellon abiertos —con sus banderas de validez— y
119 * el trabajo activo (`PathJob` o `DrillJob`) que consumirá `export_gcode`. Los
120 * dos trabajos son mutuamente excluyentes: abrir un Excellon desactiva el de
121 * recorridos y cualquier operación de geometría desactiva el de taladrado.
122 */
123struct ShellState {
124 GerberResult gerber;
125 bool hasGerber = false;
126 ExcellonResult excellon;
127 bool hasExcellon = false;
128 PathJob paths;
129 DrillJob drills;
130 bool hasPathJob = false;
131 bool hasDrillJob = false;
132};

134/**
135 * @brief Construye el texto del comando `info` con el estado actual.
136 * @param state Estado del shell.
137 * @return Resumen multilínea (una línea por objeto/trabajo cargado).
138 *
139 * @note El texto usa español y unidades del archivo, sin conversión.
140 */
141std::string describe(const ShellState& state) {
142 std::ostringstream os;
143 if (state.hasGerber) {
144 os << "gerber: " << state.gerber.apertures.size() << " aperturas, área "
145 << state.gerber.solid.area() << "\n";
146 }
147 if (state.hasExcellon) {
148 os << "excellon: " << state.excellon.drillCount << " taladros, "
149 << state.excellon.slotCount << " ranuras\n";
150 }
151 if (state.hasPathJob) os << "recorridos: " << state.paths.paths.size() << "\n";
152 if (state.hasDrillJob) os << "taladros en job: " << state.drills.tools.size() << " herramientas\n";
153 return os.str();
154}

156/**
157 * @brief Reemplaza el trabajo de recorridos del estado con geometría nueva.
158 *
159 * Convierte cada `Geom` a su lista de coordenadas y descarta los recorridos con
160 * menos de 2 puntos. Al fijar un `PathJob` desactiva el trabajo de taladrado
161 * para que `export_gcode` no exporte datos obsoletos.
162 *
163 * @param state Estado a modificar.
164 * @param paths Recorridos producidos por la herramienta.
165 * @param bounds Límites del trabajo.
166 * @param tool Diámetro de herramienta informativo.
167 */
168void setPaths(ShellState& state, const std::vector<Geom>& paths, const BBox& bounds, double tool) {
169 state.paths = PathJob{};
170 state.paths.bounds = bounds;
171 state.paths.toolDiameter = tool;
172 for (const Geom& g : paths) {
173 Ring ring = g.coords();
174 if (ring.size() >= 2) state.paths.paths.emplace_back(ring.begin(), ring.end());
175 }
176 state.hasPathJob = true;
177 state.hasDrillJob = false;
178}

182/**
183 * @brief Ejecuta un script Tcl mínimo línea a línea.
184 *
185 * Abre el archivo, recorta cada línea, ignora vacías y las que empiezan con `#`,
186 * tokeniza y despacha el verbo. Los errores de comando se informan por `stderr`
187 * y la ejecución continúa con la línea siguiente, igual que el shell real.
188 *
189 * @param path Ruta del script a ejecutar.
190 * @return 0 al terminar (incluso si hubo errores de comando); 1 si el archivo
191 * no se puede abrir.
192 *
193 * @note Comandos soportados: `set`, `puts`, `open_gerber`, `open_excellon`,
194 * `isolate`, `cutout`, `ncc`, `paint`, `export_gcode`, `info`,
195 * `quit`/`exit`.
196 * @note Los argumentos numéricos de las herramientas son posicionales y
197 * opcionales (ver `toNumber`): `isolate Ø pasadas superposición%`.
198 * @note Los parsers se construyen con opciones por defecto, de modo que este
199 * shell no respeta `--units` ni `--flatcam-compat` de la CLI.
200 * @warning Al abrir un Gerber no se desactiva `hasDrillJob`; si antes se abrió
201 * un Excellon, `export_gcode` seguirá exportando el taladrado viejo
202 * aunque el estado visible muestre el Gerber recién cargado.
203 * @warning No hay modo interactivo: con `path` vacío `std::ifstream` falla y la
204 * función devuelve 1. El REPL sólo existe en la variante con libtcl.
205 */
206int runTclScript(const std::string& path) {
207 std::ifstream in(path);
208 if (!in) {
209 std::cerr << "cannot open script: " << path << "\n";
210 return 1;
211 }
212
213 ShellState state;
214 std::map<std::string, std::string> vars;
215 std::string line;
216
217 while (std::getline(in, line)) {
218 line = trim(line);
219 if (line.empty() || line[0] == '#') continue;
220
221 std::vector<std::string> tokens = splitCommand(line);
222 if (tokens.empty()) continue;
223 const std::string command = tokens[0];
224
225 try {
226 if (command == "set") {
227 if (tokens.size() < 2) continue;
228 std::string value;
229 for (std::size_t i = 2; i < tokens.size(); ++i) {
230 if (i > 2) value += " ";
231 value += substitute(tokens[i], vars);
232 }
233 vars[tokens[1]] = value;
234 } else if (command == "puts") {
235 std::string text;
236 for (std::size_t i = 1; i < tokens.size(); ++i) {
237 if (i > 1) text += " ";
238 text += substitute(tokens[i], vars);
239 }
240 std::cout << text << "\n";
241 } else if (command == "open_gerber") {
242 if (tokens.size() < 2) continue;
243 const std::string file = substitute(tokens[1], vars);
244 state.gerber = GerberParser().parseFile(file);
245 state.hasGerber = true;
246 state.hasPathJob = false;
247 state.hasDrillJob = false;
248 std::cout << "gerber: " << file << " (" << state.gerber.solid.area() << ")\n";
249 } else if (command == "open_excellon") {
250 if (tokens.size() < 2) continue;
251 const std::string file = substitute(tokens[1], vars);
252 state.excellon = ExcellonParser().parseFile(file);
253 state.drills = drillJobFromExcellon(state.excellon);
254 state.hasExcellon = true;
255 state.hasDrillJob = true;
256 state.hasPathJob = false;
257 std::cout << "excellon: " << file << " (" << state.excellon.drillCount << ")\n";
258 } else if (command == "isolate") {
259 if (!state.hasGerber) throw std::runtime_error("no gerber loaded");
260 IsolationOptions opt;
261 opt.toolDiameter = toNumber(tokens, 1, 0.2);
262 opt.passes = static_cast<int>(toNumber(tokens, 2, 1));
263 opt.overlap = toNumber(tokens, 3, 0.0) / 100.0;
264 const IsolationResult result = isolateCopper(state.gerber.solid, opt);
265 setPaths(state, result.paths, result.bounds, opt.toolDiameter);
266 std::cout << "isolate: " << result.paths.size() << " recorridos\n";
267 } else if (command == "cutout") {
268 if (!state.hasGerber) throw std::runtime_error("no gerber loaded");
269 CutoutOptions opt;
270 opt.toolDiameter = toNumber(tokens, 1, 1.0);
271 opt.margin = toNumber(tokens, 2, 0.0);
272 opt.tabCount = static_cast<int>(toNumber(tokens, 3, 0));
273 opt.tabWidth = toNumber(tokens, 4, 0.0);
274 const CutoutResult result = cutoutGeometry(state.gerber.solid, opt);
275 setPaths(state, result.paths, result.bounds, opt.toolDiameter);
276 std::cout << "cutout: " << result.paths.size() << " recorridos\n";
277 } else if (command == "ncc") {
278 if (!state.hasGerber) throw std::runtime_error("no gerber loaded");
279 NccOptions opt;
280 opt.toolDiameter = toNumber(tokens, 1, 0.5);
281 opt.stepOver = toNumber(tokens, 2, 50.0) / 100.0;
282 const BBox b = state.gerber.bounds.expanded(1.0);
283 const Geom area = b.valid ? Geom::box(b.xmin, b.ymin, b.xmax, b.ymax) : Geom();
284 const NccResult result = clearCopper(state.gerber.solid, area, opt);
285 setPaths(state, result.paths, result.bounds, opt.toolDiameter);
286 std::cout << "ncc: " << result.paths.size() << " recorridos\n";
287 } else if (command == "paint") {
288 if (!state.hasGerber) throw std::runtime_error("no gerber loaded");
289 PaintOptions opt;
290 opt.toolDiameter = toNumber(tokens, 1, 0.1);
291 opt.stepOver = toNumber(tokens, 2, 50.0) / 100.0;
292 const PaintResult result = paintArea(state.gerber.solid, opt);
293 setPaths(state, result.paths, result.bounds, opt.toolDiameter);
294 std::cout << "paint: " << result.paths.size() << " recorridos\n";
295 } else if (command == "export_gcode") {
296 if (tokens.size() < 2) continue;
297 const std::string file = substitute(tokens[1], vars);
298 GCodeOptions opt;
299 std::string code;
300 if (state.hasDrillJob) {
301 code = drillingGCode(state.drills, opt);
302 } else if (state.hasPathJob) {
303 code = pathsGCode(state.paths, opt);
304 } else {
305 throw std::runtime_error("nothing to export");
306 }
307 std::ofstream out(file, std::ios::binary);
308 if (!out) throw std::runtime_error("cannot write " + file);
309 out << code;
310 std::cout << "wrote " << file << "\n";
311 } else if (command == "info") {
312 std::cout << describe(state);
313 } else if (command == "quit" || command == "exit") {
314 break;
315 } else {
316 std::cerr << "unknown command: " << command << "\n";
317 }
318 } catch (const std::exception& e) {
319 std::cerr << "error in '" << command << "': " << e.what() << "\n";
320 }
321 }
322 return 0;
323}
