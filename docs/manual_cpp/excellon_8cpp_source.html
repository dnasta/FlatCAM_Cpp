37/// Recorta espacios en ambos extremos.
38std::string_view trim(std::string_view s) {
39 while (!s.empty() && std::isspace(static_cast<unsigned char>(s.front()))) s.remove_prefix(1);
40 while (!s.empty() && std::isspace(static_cast<unsigned char>(s.back()))) s.remove_suffix(1);
41 return s;
42}

44/// Quita los '%' externos y recorta espacios.
45std::string_view inner(std::string_view s) {
46 s = trim(s);
47 if (!s.empty() && s.front() == '%') s.remove_prefix(1);
48 if (!s.empty() && s.back() == '%') s.remove_suffix(1);
49 return trim(s);
50}

84/**
85 * @brief Convierte un campo de coordenada Excellon a unidades del archivo.
86 *
87 * Si el texto trae punto se usa directo. Si no, separa el signo y aplica la
88 * regla del modo de ceros: con 'T' divide por 10^fracDigits; en los demás casos
89 * divide por 10^(dígitos presentes - enteros declarados).
90 * @param ctx Contexto con unidades y formato vigentes.
91 * @param text Campo sin el eje, por ejemplo "0123".
92 * @return Valor con signo en las unidades del archivo.
93 * @note El formato depende de si el archivo es métrico o en pulgadas, porque
94 * cada uno guarda sus propios dígitos enteros y fraccionarios.
95 */
96double parseCoordinate(const Ctx& ctx, std::string_view text) {
97 std::string t(text);
98 if (t.empty()) return 0.0;
99 if (t.find('.') != std::string::npos) return std::strtod(t.c_str(), nullptr);
100
101 const bool negative = t.front() == '-';
102 if (!t.empty() && (t.front() == '+' || t.front() == '-')) t.erase(t.begin());
103 const long long raw = std::llabs(std::strtoll(t.c_str(), nullptr, 10));
104
105 const bool inch = ctx.res.fileUnits == Units::Inch;
106 const int fracDigits = inch ? ctx.res.formatLowerIn : ctx.res.formatLowerMm;
107
108 double v = 0.0;
109 // Con supresión de ceros (LZ o TZ) el valor siempre se escala por los
110 // dígitos fraccionarios declarados. En LZ el texto es el sufijo derecho
111 // del campo, así que usar `len - intDigits` daría un valor 10^fracDigits
112 // veces mayor (por ejemplo "500" en 3.3 sería 500 en vez de 0,5 mm).
113 v = static_cast<double>(raw) / std::pow(10.0, fracDigits);
114 return negative ? -v : v;
115}

117/**
118 * @brief Devuelve la herramienta, creándola con diámetro ficticio si falta.
119 *
120 * Reproduce el criterio de FlatCAM: cuando el archivo no define diámetro se
121 * asigna 1 + (n-1)/100 mm y se avisa, para que la geometría siga siendo
122 * dibujable.
123 */
124ExcellonTool& toolFor(Ctx& ctx, int number) {
125 auto it = ctx.res.tools.find(number);
126 if (it == ctx.res.tools.end()) {
127 ExcellonTool t;
128 t.number = number;
129 // FlatCAM gives a bogus, slightly increasing diameter to diameterless files.
130 t.diameter = 1.0 + (number - 1) / 100.0;
131 t.diameterKnown = false;
132 it = ctx.res.tools.emplace(number, std::move(t)).first;
133 ctx.res.warnings.push_back("tool T" + std::to_string(number) + " has no diameter definition");
134 }
135 return it->second;
136}

138/**
139 * @brief Interpreta una línea de unidades y formato.
140 *
141 * Reconoce METRIC o M71 e INCH o M72, fija 3.3 o 2.4 por defecto, lee TZ o LZ
142 * para el modo de ceros y, si el especificador trae un punto (por ejemplo
143 * "000.0000"), toma de ahí los dígitos enteros y fraccionarios.
144 */
145void parseUnits(Ctx& ctx, std::string_view line) {
146 const std::string u = upper(inner(line));
147 if (startsWith(u, "METRIC") || startsWith(u, "M71")) {
148 ctx.res.fileUnits = Units::Mm;
149 ctx.res.unitsKnown = true;
150 ctx.res.formatUpperMm = 3;
151 ctx.res.formatLowerMm = 3;
152 } else if (startsWith(u, "INCH") || startsWith(u, "M72")) {
153 ctx.res.fileUnits = Units::Inch;
154 ctx.res.unitsKnown = true;
155 ctx.res.formatUpperIn = 2;
156 ctx.res.formatLowerIn = 4;
157 } else {
158 return;
159 }
160
161 const std::size_t comma = u.find(',');
162 if (comma == std::string::npos) return;
163 const std::string spec = u.substr(comma + 1);
164 if (spec.find("TZ") != std::string::npos) ctx.res.zeros = 'T';
165 if (spec.find("LZ") != std::string::npos) ctx.res.zeros = 'L';
166
167 // Explicit decimal format, e.g. "000.000" or "00.0000".
168 const std::size_t dot = spec.find('.');
169 if (dot != std::string::npos) {
170 int intDigits = 0;
171 for (std::size_t i = 0; i < dot; ++i) {
172 if (isdigit(static_cast<unsigned char>(spec[i]))) ++intDigits;
173 }
174 int fracDigits = 0;
175 for (std::size_t i = dot + 1; i < spec.size(); ++i) {
176 if (isdigit(static_cast<unsigned char>(spec[i]))) ++fracDigits;
177 }
178 if (intDigits > 0 && fracDigits > 0) {
179 if (ctx.res.fileUnits == Units::Mm) {
180 ctx.res.formatUpperMm = intDigits;
181 ctx.res.formatLowerMm = fracDigits;
182 } else {
183 ctx.res.formatUpperIn = intDigits;
184 ctx.res.formatLowerIn = fracDigits;
185 }
186 }
187 }
188}

190// T1C0.8 / T01C0.800 / T1
191/**
192 * @brief Parsea una definición o selección de herramienta (TnCn o Tn).
193 *
194 * Con diámetro válido actualiza la herramienta y la deja activa; sin diámetro
195 * sólo la selecciona, creándola como desconocida.
196 */
197void parseToolLine(Ctx& ctx, std::string_view line) {
198 std::size_t i = 1;
199 const std::size_t start = i;
200 while (i < line.size() && isdigit(static_cast<unsigned char>(line[i]))) ++i;
201 if (i == start) return;
202 const int number = std::atoi(std::string(line.substr(start, i - start)).c_str());
203 const std::size_t c = line.find('C', i);
204 if (c == std::string_view::npos) {
205 ctx.tool = number;
206 toolFor(ctx, number);
207 return;
208 }
209 const std::string_view diamText = line.substr(c + 1);
210 std::string d(diamText);
211 const double dia = std::strtod(d.c_str(), nullptr);
212 if (dia <= 0.0) return;
213
214 auto it = ctx.res.tools.find(number);
215 if (it == ctx.res.tools.end()) {
216 ExcellonTool tool;
217 tool.number = number;
218 it = ctx.res.tools.emplace(number, std::move(tool)).first;
219 }
220 it->second.diameter = dia;
221 it->second.diameterKnown = true;
222 ctx.tool = number;
223}

228};

230/**
231 * @brief Extrae los campos X e Y de una línea.
232 *
233 * Acepta signo y punto decimal e ignora el resto de los caracteres (G, M, R...).
234 * @return Campos presentes; cada uno queda como texto sin interpretar.
235 */
236Coords readCoords(std::string_view s) {
237 Coords c;
238 std::size_t i = 0;
239 while (i < s.size()) {
240 const char ch = s[i];
241 if (ch == 'X' || ch == 'Y') {
242 ++i;
243 const std::size_t st = i;
244 if (i < s.size() && (s[i] == '+' || s[i] == '-')) ++i;
245 while (i < s.size() && (isdigit(static_cast<unsigned char>(s[i])) || s[i] == '.')) ++i;
246 if (i > st) {
247 if (ch == 'X') c.x = std::string(s.substr(st, i - st));
248 else c.y = std::string(s.substr(st, i - st));
249 }
250 } else {
251 ++i;
252 }
253 }
254 return c;
255}

282/**
283 * @brief Procesa una línea del archivo Excellon.
284 *
285 * Cubre M48 y M95, el fin de programa, M71 y M72, G90 y G91, METRIC e INCH, la
286 * definición y selección de herramientas, el ruteo M15/M16/M17 con G00 y G01, el
287 * prefijo de repetición R, las ranuras G85 y, por defecto, los taladros con
288 * coordenadas.
289 */
290void handleBodyLine(Ctx& ctx, std::string_view line) {
291 const std::string u = upper(trim(line));
292 if (u.empty()) return;
293 if (u.front() == ';') return;
294 if (startsWith(u, "M48")) {
295 ctx.inHeader = true;
296 return;
297 }
298 if (u == "%" || u == "M95") {
299 ctx.inHeader = false;
300 return;
301 }
302 if (startsWith(u, "M30") || startsWith(u, "M00") || startsWith(u, "M02")) return;
303 if (u.rfind("M71", 0) == 0 || u.rfind("M72", 0) == 0) {
304 parseUnits(ctx, u);
305 return;
306 }
307 if (startsWith(u, "G90")) {
308 ctx.incremental = false;
309 return;
310 }
311 if (startsWith(u, "G91")) {
312 ctx.incremental = true;
313 return;
314 }
315 if (startsWith(u, "METRIC") || startsWith(u, "INCH")) {
316 parseUnits(ctx, u);
317 return;
318 }
319 if (u.front() == 'T' && u.size() > 1 && isdigit(static_cast<unsigned char>(u[1]))) {
320 parseToolLine(ctx, u);
321 return;
322 }
323 if (u.front() == ';') return;
324 if (startsWith(u, "M15") || startsWith(u, "M16") || startsWith(u, "M17")) {
325 if (startsWith(u, "M15")) {
326 ctx.routingDown = true;
327 ctx.routeStart = {ctx.x, ctx.y};
328 } else {
329 ctx.routingDown = false;
330 }
331 return;
332 }
333 if (startsWith(u, "G00") || startsWith(u, "G0") || startsWith(u, "G01") || startsWith(u, "G1")) {
334 const Coords c = readCoords(u);
335 double nx = c.x ? parseCoordinate(ctx, *c.x) : ctx.x;
336 double ny = c.y ? parseCoordinate(ctx, *c.y) : ctx.y;
337 if (ctx.incremental) {
338 if (c.x) nx += ctx.x;
339 if (c.y) ny += ctx.y;
340 }
341 if (ctx.routingDown && (startsWith(u, "G01") || startsWith(u, "G1"))) {
342 addSlot(ctx, ctx.routeStart, {nx, ny});
343 ctx.routeStart = {nx, ny};
344 }
345 ctx.x = nx;
346 ctx.y = ny;
347 return;
348 }
349
350 // Repeat prefix: R3X1.0Y2.0
351 if (u.front() == 'R' && u.size() > 1 && isdigit(static_cast<unsigned char>(u[1]))) {
352 const Coords c = readCoords(u);
353 std::size_t i = 1;
354 while (i < u.size() && isdigit(static_cast<unsigned char>(u[i]))) ++i;
355 ctx.repeat = std::atoi(u.substr(1, i - 1).c_str());
356 ctx.repeatX = c.x ? std::strtod(c.x->c_str(), nullptr) : 0.0;
357 ctx.repeatY = c.y ? std::strtod(c.y->c_str(), nullptr) : 0.0;
358 return;
359 }
360
361 // G85 slot: X..Y..G85X..Y..
362 const std::size_t g85 = u.find("G85");
363 if (g85 != std::string::npos) {
364 const Coords startCoords = readCoords(u.substr(0, g85));
365 const Coords stopCoords = readCoords(u.substr(g85 + 3));
366 double sx = startCoords.x ? parseCoordinate(ctx, *startCoords.x) : ctx.x;
367 double sy = startCoords.y ? parseCoordinate(ctx, *startCoords.y) : ctx.y;
368 double ex = stopCoords.x ? parseCoordinate(ctx, *stopCoords.x) : sx;
369 double ey = stopCoords.y ? parseCoordinate(ctx, *stopCoords.y) : sy;
370 if (ctx.incremental) {
371 if (startCoords.x) sx += ctx.x;
372 if (startCoords.y) sy += ctx.y;
373 if (stopCoords.x) ex += sx;
374 if (stopCoords.y) ey += sy;
375 }
376 addSlot(ctx, {sx, sy}, {ex, ey});
377 ctx.x = ex;
378 ctx.y = ey;
379 return;
380 }
381
382 const Coords c = readCoords(u);
383 if (!c.x && !c.y) return;
384
385 double x = c.x ? parseCoordinate(ctx, *c.x) : ctx.x;
386 double y = c.y ? parseCoordinate(ctx, *c.y) : ctx.y;
387 if (ctx.incremental) {
388 if (c.x) x += ctx.x;
389 if (c.y) y += ctx.y;
390 }
391
392 if (ctx.routingDown) {
393 addSlot(ctx, ctx.routeStart, {x, y});
394 ctx.routeStart = {x, y};
395 } else {
396 addDrill(ctx, x, y);
397 for (int i = 1; i <= ctx.repeat; ++i) {
398 addDrill(ctx, x + i * ctx.repeatX, y + i * ctx.repeatY);
399 }
400 ctx.repeat = 0;
401 }
402 ctx.x = x;
403 ctx.y = y;
404}

406/**
407 * @brief Construye la geometría y aplica la conversión de unidades.
408 *
409 * Por cada herramienta crea un círculo por taladro y una cápsula, o un círculo
410 * si la ranura es degenerada, por segmento; une todo en un sólido, calcula los
411 * límites, convierte a las unidades de la aplicación y reescala diámetros y
412 * geometrías por herramienta.
413 */
414void finalize(Ctx& ctx) {
415 const double factor = conversionFactor(ctx.res.fileUnits, ctx.opt.appUnits);
416 const int quad = std::max(2, ctx.opt.stepsPerCircle / 4);
417
418 for (auto& [number, tool] : ctx.res.tools) {
419 for (const Pt& d : tool.drills) {
420 Geom g = Geom::circle(d, tool.diameter / 2.0, quad);
421 tool.geometry.push_back(g);
422 ctx.res.geometry.push_back(g);
423 }
424 for (const auto& [a, b] : tool.slotSegments) {
425 Geom g = (a.x == b.x && a.y == b.y)
426 ? Geom::circle(a, tool.diameter / 2.0, quad)
427 : Geom::line({{a.x, a.y}, {b.x, b.y}}).buffer(tool.diameter / 2.0, quad);
428 tool.geometry.push_back(g);
429 ctx.res.geometry.push_back(g);
430 }
431 }
432
433 std::vector<Geom> converted;
434 converted.reserve(ctx.res.geometry.size());
435 for (const Geom& g : ctx.res.geometry) {
436 Geom c = factor == 1.0 ? g : g.scale(factor, factor);
437 converted.push_back(c);
438 const BBox b = c.bounds();
439 if (!b.valid) continue;
440 if (!ctx.res.bounds.valid) {
441 ctx.res.bounds = b;
442 } else {
443 ctx.res.bounds.xmin = std::min(ctx.res.bounds.xmin, b.xmin);
444 ctx.res.bounds.ymin = std::min(ctx.res.bounds.ymin, b.ymin);
445 ctx.res.bounds.xmax = std::max(ctx.res.bounds.xmax, b.xmax);
446 ctx.res.bounds.ymax = std::max(ctx.res.bounds.ymax, b.ymax);
447 }
448 }
449 ctx.res.geometry = std::move(converted);
450
451 if (!ctx.res.geometry.empty()) ctx.res.solid = Geom::unionOf(ctx.res.geometry);
452
453 for (auto& [number, tool] : ctx.res.tools) {
454 if (factor != 1.0) {
455 tool.diameter *= factor;
456 for (Geom& g : tool.geometry) g = g.scale(factor, factor);
457 }
458 }
459}

466/**
467 * @brief Parsea un Excellon que ya está en memoria.
468 * @param source Texto completo del archivo.
469 * @return Resultado con herramientas, taladros, ranuras, geometría y avisos.
470 * @note Si el archivo no declara unidades se usan las de la aplicación y el
471 * modo de ceros por defecto de las opciones.
472 */
473ExcellonResult ExcellonParser::parseString(std::string_view source) const {
474 ExcellonResult result;
475 result.zeros = opt_.defaultZeros;
476 result.fileUnits = opt_.appUnits;
477 Ctx ctx(opt_, result);
478
479 std::size_t start = 0;
480 while (start <= source.size()) {
481 const std::size_t end = source.find('\n', start);
482 const std::string_view line = (end == std::string_view::npos)
483 ? source.substr(start)
484 : source.substr(start, end - start);
485 handleBodyLine(ctx, line);
486 if (end == std::string_view::npos) break;
487 start = end + 1;
488 }
489 finalize(ctx);
490 return result;
491}
