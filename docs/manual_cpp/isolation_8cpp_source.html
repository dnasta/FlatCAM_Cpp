2 * @file isolation.cpp
3 * @brief Aislamiento de cobre por offsets sucesivos y rest machining.
4 *
5 * Reproduce el esquema de FlatCAM: el cobre se descompone en polígonos, cada
6 * pasada dilata cada uno con `buffer(offset)` y une los resultados para obtener
7 * la "envolvente" de esa pasada, de la que se extraen exteriores y huecos como
8 * recorridos del centro de la fresa. Incluye además el rest machining
9 * (`isolateRest`), que aísla sólo el cobre inalcanzable por una fresa mayor.
10 */

26/// @brief Acumula `b` dentro de `target`, ignorando cajas inválidas.
27void addBounds(BBox& target, const BBox& b) {
28 if (!b.valid) return;
29 if (!target.valid) {
30 target = b;
31 return;
32 }
33 target.xmin = std::min(target.xmin, b.xmin);
34 target.ymin = std::min(target.ymin, b.ymin);
35 target.xmax = std::max(target.xmax, b.xmax);
36 target.ymax = std::max(target.ymax, b.ymax);
37}

41/**
42 * @brief Calcula los recorridos de aislamiento alrededor del cobre.
43 *
44 * Para cada pasada `i` (0-based) se usa la fórmula de FlatCAM:
45 * `offset = Ø · (2i + 1)/2 − i · solape · Ø`, es decir, la distancia del centro
46 * de la fresa al borde del cobre de la i-ésima pasada. El procedimiento es:
47 *
48 * 1. Descomponer el cobre en polígonos independientes (exterior + huecos).
49 * 2. Por pasada, dilatar cada polígono `offset` mm y unir los resultados en una
50 * sola envolvente; con `avoidIntersections` se omite la pasada de un
51 * polígono cuyo buffer (`offset × 2`) choca con otro polígono distinto.
52 * 3. Extraer de la envolvente el contorno exterior (salvo
53 * `interiorsOnly`) y los huecos (salvo `exteriorsOnly`) como recorridos,
54 * invirtiendo el sentido si `invert` (fresado en concordancia).
55 *
56 * @param copper Cobre de entrada (sólidos de un Gerber).
57 * @param options Diámetro, pasadas, solape, filtros, sentido y rest machining.
58 * @return Recorridos, envolventes por pasada, polígonos no aislados, longitud
59 * total y límites acumulados.
60 *
61 * @note `overlap` se acota a [0, 0.99] y `passes` a un mínimo de 1, de modo que
62 * valores fuera de rango no rompen el cálculo.
63 * @note Las envolventes se guardan en `result.envelopes` para la
64 * previsualización y se acumulan sin unirlas entre pasadas.
65 * @note `notIsolated` sólo se llena si `avoidIntersections` está activo: son
66 * los polígonos que no recibieron ninguna pasada.
67 * @warning Con `exteriorsOnly` y `interiorsOnly` a la vez no se genera ningún
68 * recorrido, porque ambas condiciones de emisión quedan desactivadas.
69 * @warning El chequeo de colisión dilata con `offset × 2`, no con `offset`;
70 * es una aproximación conservadora del comportamiento de FlatCAM y
71 * puede descartar pasadas que en realidad no chocaban.
72 */
73IsolationResult isolateCopper(const Geom& copper, const IsolationOptions& options) {
74 IsolationResult result;
75 if (!copper.valid() || copper.empty()) return result;
76
77 const int passes = std::max(1, options.passes);
78 const int quadSegs = std::max(2, options.quadrantSegments);
79 const double overlap = std::clamp(options.overlap, 0.0, 0.99);
80
81 // Split the copper into individual polygons once; FlatCAM buffers each one
82 // and unions the result for every pass.
83 struct SourcePolygon {
84 Geom geom;
85 };
86 std::vector<SourcePolygon> sources;
87 for (const PolygonContours& pc : copper.polygons()) {
88 Geom poly = Geom::polygon(pc.outer, pc.holes);
89 if (poly.valid() && !poly.empty()) sources.push_back({poly});
90 }
91
92 std::vector<std::size_t> perSourcePasses(sources.size(), 0);
93
94 for (int pass = 0; pass < passes; ++pass) {
95 const double offset = options.toolDiameter * ((2.0 * pass + 1.0) / 2.0) -
96 pass * overlap * options.toolDiameter;
97
98 std::vector<Geom> buffers;
99 buffers.reserve(sources.size());
100 for (std::size_t i = 0; i < sources.size(); ++i) {
101 const SourcePolygon& source = sources[i];
102 if (options.avoidIntersections && offset > 0.0) {
103 Geom check = source.geom.buffer(offset * 2.0, quadSegs);
104 bool collides = false;
105 for (std::size_t j = 0; j < sources.size() && !collides; ++j) {
106 if (i == j) continue;
107 if (check.intersects(sources[j].geom)) collides = true;
108 }
109 if (collides) continue;
110 }
111 Geom buffered = offset == 0.0 ? source.geom : source.geom.buffer(offset, quadSegs);
112 if (buffered.valid() && !buffered.empty()) {
113 buffers.push_back(std::move(buffered));
114 ++perSourcePasses[i];
115 }
116 }
117 if (buffers.empty()) continue;
118
119 Geom envelope = buffers.size() == 1 ? buffers.front() : Geom::unionOf(buffers);
120 if (!envelope.valid() || envelope.empty()) continue;
121 result.envelopes.push_back(envelope);
122
123 for (const PolygonContours& pc : envelope.polygons()) {
124 if (!options.interiorsOnly) {
125 Geom path = Geom::line(oriented(pc.outer, options.invert));
126 if (path.valid() && !path.empty()) {
127 result.totalLength += path.length();
128 result.paths.push_back(std::move(path));
129 }
130 }
131 if (!options.exteriorsOnly) {
132 for (const Ring& hole : pc.holes) {
133 Geom path = Geom::line(oriented(hole, options.invert));
134 if (path.valid() && !path.empty()) {
135 result.totalLength += path.length();
136 result.paths.push_back(std::move(path));
137 }
138 }
139 }
140 }
141 addBounds(result.bounds, envelope.bounds());
142 }
143
144 if (options.avoidIntersections) {
145 for (std::size_t i = 0; i < sources.size(); ++i) {
146 if (perSourcePasses[i] == 0) result.notIsolated.push_back(sources[i].geom);
147 }
148 }
149
150 return result;
151}

153/**
154 * @brief Aísla únicamente el cobre que una herramienta mayor no pudo alcanzar.
155 *
156 * Divide el cobre en polígonos y marca como "resto" aquellos cuyo buffer de
157 * `previousToolDiameter` intersecta algún otro polígono, es decir, los que
158 * están demasiado cerca de otro feature como para que la fresa anterior pase
159 * entre ambos. Si no hay resto devuelve un resultado vacío. En caso contrario
160 * une esos polígonos y ejecuta `isolateCopper` con la herramienta menor,
161 * desactivando `avoidIntersections` para no perder pasadas dentro del resto.
162 *
163 * @param copper Cobre original.
164 * @param previousToolDiameter Diámetro de la fresa usada en la pasada previa;
165 * debe ser > 0.
166 * @param options Opciones de la herramienta menor.
167 * @return Resultado del aislamiento del resto, con `notIsolated` igual a los
168 * polígonos considerados resto.
169 *
170 * @note El resultado devuelto descarta deliberadamente los `notIsolated` del
171 * cálculo interno y los reemplaza por la lista de polígonos de resto.
172 * @note La unión previa (`unionOf`) evita que polígonos vecinos generen
173 * contornos duplicados en la misma pasada.
174 * @note El chequeo usa `buffer(previousToolDiameter)`, igual que
175 * `buf_chek = iso_offset * 2` del `generate_rest_geometry` original de
176 * FlatCAM; se mantiene por paridad funcional.
177 */
178IsolationResult isolateRest(const Geom& copper, double previousToolDiameter,
179 const IsolationOptions& options) {
180 IsolationResult result;
181 if (!copper.valid() || copper.empty() || previousToolDiameter <= 0.0) return result;
182
183 std::vector<Geom> sources;
184 for (const PolygonContours& pc : copper.polygons()) {
185 Geom poly = Geom::polygon(pc.outer, pc.holes);
186 if (poly.valid() && !poly.empty()) sources.push_back(std::move(poly));
187 }
188
189 std::vector<Geom> rest;
190 for (std::size_t i = 0; i < sources.size(); ++i) {
191 Geom check = sources[i].buffer(previousToolDiameter, std::max(2, options.quadrantSegments));
192 bool collides = false;
193 for (std::size_t j = 0; j < sources.size() && !collides; ++j) {
194 if (i == j) continue;
195 if (check.intersects(sources[j])) collides = true;
196 }
197 if (collides) rest.push_back(sources[i]);
198 }
199 if (rest.empty()) return result;
200
201 Geom restGeometry = rest.size() == 1 ? rest.front() : Geom::unionOf(rest);
202 IsolationOptions current = options;
203 current.avoidIntersections = false;
204 result = isolateCopper(restGeometry, current);
205 result.notIsolated = std::move(rest);
206 return result;
207}
