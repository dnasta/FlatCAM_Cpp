16/**
17 * @brief Convierte un texto en la geometría de un QR en coordenadas CAD.
18 *
19 * Elige la corrección de errores desde @p ecc ('L', 'Q' o 'H'; cualquier otro
20 * valor, incluido 'M', cae en MEDIUM), codifica con qrcodegen y recorre las
21 * filas agrupando módulos oscuros consecutivos en cajas de moduleSize. Cada
22 * caja se desplaza por el borde y el eje Y se invierte porque la librería
23 * numera las filas de arriba hacia abajo. Al final une todas las cajas con
24 * Geom::unionOf para entregar un solo polígono.
25 * @param text Contenido a codificar.
26 * @param moduleSize Lado del módulo en mm; si es <= 0 devuelve vacío.
27 * @param borderModules Borde en módulos (se satura a >= 0).
28 * @param originX Coordenada X de la esquina inferior izquierda (mm).
29 * @param originY Ídem en Y (mm).
30 * @param ecc Nivel de corrección: 'L', 'M', 'Q' o 'H'.
31 * @return Vector con un único Geom; vacío si no hay módulos oscuros o si el
32 * tamaño de módulo no es positivo.
33 * @note qrcodegen lanza excepción si el texto no cabe en el nivel pedido; este
34 * módulo no la captura.
35 */
36std::vector<Geom> qrCodeGeometry(const std::string& text, double moduleSize, int borderModules,
37 double originX, double originY, char ecc) {
38 if (moduleSize <= 0.0) return {};
39 qrcodegen::QrCode::Ecc level = qrcodegen::QrCode::Ecc::MEDIUM;
40 switch (ecc) {
41 case 'L': level = qrcodegen::QrCode::Ecc::LOW; break;
42 case 'Q': level = qrcodegen::QrCode::Ecc::QUARTILE; break;
43 case 'H': level = qrcodegen::QrCode::Ecc::HIGH; break;
44 default: level = qrcodegen::QrCode::Ecc::MEDIUM; break;
45 }
46 const qrcodegen::QrCode qr = qrcodegen::QrCode::encodeText(text.c_str(), level);
47 const int size = qr.getSize();
48 const int border = std::max(0, borderModules);
49 const double origin = border * moduleSize;
50
51 std::vector<Geom> runs;
52 runs.reserve(static_cast<std::size_t>(size * size / 3));
53 for (int y = 0; y < size; ++y) {
54 int start = -1;
55 for (int x = 0; x <= size; ++x) {
56 const bool dark = x < size && qr.getModule(x, y);
57 if (dark && start < 0) start = x;
58 if (!dark && start >= 0) {
59 // QR y grows downwards in the library; flip it for CAD.
60 const double y0 = originY + origin + (size - 1 - y) * moduleSize;
61 runs.push_back(Geom::box(originX + origin + start * moduleSize, y0,
62 originX + origin + x * moduleSize, y0 + moduleSize));
63 start = -1;
64 }
65 }
66 }
67 if (runs.empty()) return {};
68 return {Geom::unionOf(runs)};
69}
