22/**
23 * @brief Aplana el resultado de una intersección en una lista de cadenas.
24 *
25 * GEOS devuelve `LineString`, `MultiLineString`, `GeometryCollection` o incluso
26 * `MultiPolygon` según el caso, de modo que la función recorre el árbol y
27 * agrega únicamente las geometrías de tipo línea no vacías.
28 *
29 * @param geom Geometría a recorrer.
30 * @param out Lista donde se acumulan las cadenas encontradas.
31 */
32void collectLines(const Geom& geom, std::vector<Geom>& out) {
33 if (!geom.valid() || geom.empty()) return;
34 const int type = geom.typeId();
35 if (type == GEOS_LINESTRING || type == GEOS_LINEARRING) {
36 out.push_back(geom);
37 return;
38 }
39 if (type == GEOS_MULTILINESTRING || type == GEOS_GEOMETRYCOLLECTION ||
40 type == GEOS_MULTIPOLYGON) {
41 const int n = geom.numGeometries();
42 for (int i = 0; i < n; ++i) collectLines(geom.geometryN(i), out);
43 }
44}

46/// @brief Acumula `b` dentro de `target`, ignorando cajas inválidas.
47void addBounds(BBox& target, const BBox& b) {
48 if (!b.valid) return;
49 if (!target.valid) {
50 target = b;
51 return;
52 }
53 target.xmin = std::min(target.xmin, b.xmin);
54 target.ymin = std::min(target.ymin, b.ymin);
55 target.xmax = std::max(target.xmax, b.xmax);
56 target.ymax = std::max(target.ymax, b.ymax);
57}

61/**
62 * @brief Despeja todo el cobre que no pertenece a la geometría de referencia.
63 *
64 * Construye la región mecanizable como `workArea.difference(keepOut)`, donde
65 * `keepOut = copper.buffer(radius + clearance)` es el cobre protegido con su
66 * distancia de seguridad. Si no se pasa área de trabajo se usa el bounding box
67 * del cobre expandido por `margin + spacing`.
68 *
69 * Con `NccStrategy::Paint` delega en `fc::paintArea` sobre esa región. Con
70 * `NccStrategy::Hatch` (predeterminada) genera líneas paralelas separadas
71 * `stepOver × diámetro`, las rota `angleDeg` alrededor del centro del área de
72 * trabajo, las extiende ±diagonal y las recorta con `intersection(region)`;
73 * cada fragmento de longitud ≥ 1e-6 mm se convierte en un recorrido
74 * independiente.
75 *
76 * @param copper Cobre que debe conservarse.
77 * @param area Área rectangular de trabajo; si está vacía se deriva del cobre.
78 * @param options Estrategia, diámetro, paso, clearance, ángulo, margen y
79 * segmentos por cuadrante.
80 * @return Recorridos, área despejada, longitud total y límites.
81 *
82 * @note `clearedArea` se informa también en modo Paint, aunque en ese caso el
83 * recorte real lo hace `paintArea` con su propio margen.
84 * @note `spacing` se satura a 0,05 mm como mínimo para evitar bucles infinitos.
85 * @warning En modo Hatch el barrido en Y va desde `bounds.ymin − margin` hasta
86 * `bounds.ymax + margin`, pero `bounds` ya deriva de un área que
87 * incluye `margin + spacing`. Las líneas sobrantes se recortan contra
88 * la región, así que el resultado es correcto, aunque se generan y
89 * recortan segmentos de más (trabajo redundante).
90 * @note El barrido usa offsets perpendiculares a la dirección del hatch, de
91 * modo que la separación efectiva es `spacing` para cualquier ángulo.
92 * @note `lineCount` se acumula sólo con fines de depuración y luego se
93 * descarta con `(void)lineCount`; no se informa al llamador.
94 */
95NccResult clearCopper(const Geom& copper, const Geom& area, const NccOptions& options) {
96 NccResult result;
97 if (!copper.valid() || copper.empty()) return result;
98
99 const double radius = options.toolDiameter / 2.0;
100 const double clearance = std::max(0.0, options.clearance);
101 const double spacing = std::max(0.05, options.stepOver * options.toolDiameter);
102
103 Geom workArea = area;
104 if (!workArea.valid() || workArea.empty()) {
105 const BBox b = copper.bounds();
106 if (!b.valid) return result;
107 const double m = options.margin + spacing;
108 workArea = Geom::box(b.xmin - m, b.ymin - m, b.xmax + m, b.ymax + m);
109 }
110
111 Geom keepOut = copper.buffer(radius + clearance, std::max(2, options.quadrantSegments));
112 Geom region = workArea.difference(keepOut);
113 if (!region.valid() || region.empty()) return result;
114 result.clearedArea = region;
115
116 if (options.strategy == NccStrategy::Paint) {
117 PaintOptions paint;
118 paint.toolDiameter = options.toolDiameter;
119 paint.stepOver = options.stepOver;
120 paint.quadrantSegments = options.quadrantSegments;
121 const PaintResult painted = paintArea(region, paint);
122 result.paths = painted.paths;
123 result.totalLength = painted.totalLength;
124 result.bounds = painted.bounds;
125 return result;
126 }
127
128 const BBox bounds = workArea.bounds();
129 if (!bounds.valid) return result;
130
131 const double cx = (bounds.xmin + bounds.xmax) / 2.0;
132 const double cy = (bounds.ymin + bounds.ymax) / 2.0;
133 const double diag = std::hypot(bounds.width(), bounds.height()) + 4.0 * spacing;
134
135 const double angle = options.angleDeg * M_PI / 180.0;
136 const double c = std::cos(angle);
137 const double s = std::sin(angle);
138
139 int lineCount = 0;
140 // Barre offsets perpendiculares a la dirección del hatch. Usar el eje Y
141 // sin rotar dejaba separaciones efectivas mayores y esquinas sin cubrir
142 // con ángulos de 45°.
143 const double half = diag / 2.0;
144 const int steps = std::max(1, static_cast<int>(std::ceil((2.0 * half) / spacing)));
145 for (int k = 0; k <= steps; ++k) {
146 const double offset = -half + k * spacing;
147 const Pt mid{cx - s * offset, cy + c * offset};
148 const Pt p0{mid.x - c * half, mid.y - s * half};
149 const Pt p1{mid.x + c * half, mid.y + s * half};
150
151 Geom line = Geom::line({p0, p1});
152 if (!line.valid() || line.empty()) continue;
153 Geom clipped = line.intersection(region);
154 std::vector<Geom> pieces;
155 collectLines(clipped, pieces);
156 for (Geom& piece : pieces) {
157 const double len = piece.length();
158 if (len < 1e-6) continue;
159 result.totalLength += len;
160 addBounds(result.bounds, piece.bounds());
161 result.paths.push_back(std::move(piece));
162 ++lineCount;
163 }
164 }
165 (void)lineCount;
166
167 return result;
168}
