20/**
21 * @brief Calcula los centros de una malla de puntos que no invade el cobre.
22 *
23 * El algoritmo es puramente raster: recorre `area` de izquierda a derecha y de
24 * abajo hacia arriba con paso `spacing` (mínimo 0,1 mm), empezando en
25 * `ymin/xmin + margin + spacing/2` para que el primer punto quede a media
26 * separación del borde útil. Para cada candidato construye un círculo de
27 * `dotDiameter/2` y lo descarta si intersecta el "keep-out", que es el cobre
28 * original dilatado por `dotDiameter/2 + clearance` con 8 segmentos por
29 * cuadrante.
30 *
31 * @param copper Cobre existente que debe evitarse (puede ser inválido).
32 * @param area Rectángulo donde se siembra la grilla.
33 * @param options Diámetro del punto, separación, distancia de seguridad y
34 * margen interior.
35 * @return Centros `Pt` válidos, en orden de barrido (fila por fila).
36 *
37 * @note Si `area` no es válida devuelve un vector vacío; si `copper` no es
38 * válido no hay keep-out y se devuelven todos los puntos de la malla.
39 * @note La separación se satura a 0,1 mm para evitar bucles infinitos con
40 * valores nulos o negativos.
41 * @warning El costo es O(filas × columnas) con una intersección GEOS por punto;
42 * no se usa índice espacial, por lo que áreas grandes con muchos
43 * puntos pueden ser lentas.
44 * @warning Si `margin` supera la mitad del ancho o alto del área, los bucles
45 * no se ejecutan y el resultado queda vacío sin aviso.
46 */
47std::vector<Pt> thievingPoints(const Geom& copper, const BBox& area,
48 const ThievingOptions& options) {
49 std::vector<Pt> points;
50 if (!area.valid) return points;
51
52 const double radius = options.dotDiameter / 2.0;
53 const double spacing = std::max(0.1, options.spacing);
54 const Geom keepOut = copper.valid() ? copper.buffer(radius + options.clearance, 8) : Geom();
55
56 const double margin = options.margin;
57 const double x0 = area.xmin + margin + spacing / 2.0;
58 const double y0 = area.ymin + margin + spacing / 2.0;
59 const int nx = x0 <= area.xmax - margin
60 ? static_cast<int>(std::floor((area.xmax - margin - x0) / spacing)) + 1
61 : 0;
62 const int ny = y0 <= area.ymax - margin
63 ? static_cast<int>(std::floor((area.ymax - margin - y0) / spacing)) + 1
64 : 0;
65 for (int iy = 0; iy < ny; ++iy) {
66 const double y = y0 + iy * spacing;
67 for (int ix = 0; ix < nx; ++ix) {
68 const double x = x0 + ix * spacing;
69 const Geom dot = Geom::circle({x, y}, radius, 8);
70 if (keepOut.valid() && keepOut.intersects(dot)) continue;
71 points.push_back({x, y});
72 }
73 }
74 return points;
75}
