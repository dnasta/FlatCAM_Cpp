20/**
21 * @brief Carga y evalúa el script indicado en la línea de comandos.
22 *
23 * Flujo: crea `QCoreApplication`, valida que haya un argumento, lee el archivo
24 * completo como UTF-8, crea el motor, instala la API y evalúa el código usando
25 * la ruta como nombre de archivo (para que los errores informen la ubicación).
26 *
27 * @param argc Debe ser al menos 2 (programa + script).
28 * @param argv Ruta del script en `argv[1]`.
29 * @return 0 si el script se evalúa sin error; 1 si el archivo no se puede abrir
30 * o la evaluación devuelve un error; 2 si faltan argumentos.
31 *
32 * @note No llama a `engine.installExtensions(ConsoleExtension)`: `console.log`
33 * no está disponible, pero `print()` sí, porque lo define
34 * `installScriptApi`.
35 * @note Los errores de script se escriben por `stderr` y detienen el proceso;
36 * no se ejecuta ningún `catch` de JavaScript que el script no haya
37 * definido.
38 * @warning `api` se construye después de `engine`, de modo que se destruye
39 * antes. Como `newQObject` aplica `JavaScriptOwnership`, el motor
40 * podría intentar borrarlo en su destructor; ver la advertencia de
41 * `installScriptApi`.
42 */
43int main(int argc, char** argv) {
44 QCoreApplication app(argc, argv);
45 if (argc < 2) {
46 std::cerr << "usage: flatcam-js <script.js>\n";
47 return 2;
48 }
49
50 QFile file(QString::fromLocal8Bit(argv[1]));
51 if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
52 std::cerr << "cannot open script\n";
53 return 1;
54 }
55 const QString code = QString::fromUtf8(file.readAll());
56
57 QJSEngine engine;
58 fc::ScriptApi api;
59 fc::installScriptApi(engine, &api);
60
61 const QJSValue result = engine.evaluate(code, QString::fromLocal8Bit(argv[1]));
62 if (result.isError()) {
63 std::cerr << "script error: " << result.toString().toStdString() << "\n";
64 return 1;
65 }
66 return 0;
67}
