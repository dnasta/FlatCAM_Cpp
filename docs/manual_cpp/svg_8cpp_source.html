19/**
20 * @brief Escribe un SVG con los sólidos rellenos y los recorridos como líneas.
21 *
22 * Si @p bounds no es válido, calcula la unión de las cajas de los sólidos; si
23 * tampoco así hay caja, no escribe nada. Ancho y alto se acotan a 1e-6 para
24 * evitar un viewBox degenerado. Los contornos con menos de 3 puntos se
25 * descartan y los huecos se emiten como subcaminos del mismo path con
26 * fill-rule="evenodd". El grupo exterior aplica translate(0, ymin+ymax) y
27 * scale(1,-1) para pasar de Y hacia arriba a Y hacia abajo.
28 * @param path Archivo de destino.
29 * @param solids Polígonos rellenos (cobre).
30 * @param follows Recorridos abiertos; se usa g.coords() de cada uno.
31 * @param bounds Caja a usar como viewBox; se calcula si no es válida.
32 * @throw std::runtime_error si no se puede abrir el archivo.
33 */
34void exportSvg(const std::string& path, const std::vector<Geom>& solids,
35 const std::vector<Geom>& follows, const BBox& bounds) {
36 BBox b = bounds;
37 if (!b.valid) {
38 for (const Geom& g : solids) {
39 const BBox gb = g.bounds();
40 if (!gb.valid) continue;
41 if (!b.valid) b = gb;
42 else {
43 b.xmin = std::min(b.xmin, gb.xmin);
44 b.ymin = std::min(b.ymin, gb.ymin);
45 b.xmax = std::max(b.xmax, gb.xmax);
46 b.ymax = std::max(b.ymax, gb.ymax);
47 }
48 }
49 }
50 if (!b.valid) return;
51
52 const double w = std::max(1e-6, b.width());
53 const double h = std::max(1e-6, b.height());
54
55 std::ostringstream out;
56 out << "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n";
57 out << "<svg xmlns=\"http://www.w3.org/2000/svg\" width=\"" << w << "mm\" height=\"" << h
58 << "mm\" viewBox=\"" << b.xmin << " " << b.ymin << " " << w << " " << h << "\">\n";
59 out << "<g transform=\"translate(0 " << (b.ymin + b.ymax) << ") scale(1 -1)\">\n";
60
61 out << "<g fill=\"#c8952e\" fill-rule=\"evenodd\" stroke=\"none\">\n";
62 for (const Geom& g : solids) {
63 for (const PolygonContours& pc : g.polygons()) {
64 if (pc.outer.size() < 3) continue;
65 out << "<path d=\"M";
66 for (std::size_t i = 0; i < pc.outer.size(); ++i) {
67 out << (i ? " L" : " ") << pc.outer[i].x << " " << pc.outer[i].y;
68 }
69 out << " Z";
70 for (const Ring& hole : pc.holes) {
71 if (hole.size() < 3) continue;
72 out << " M";
73 for (std::size_t i = 0; i < hole.size(); ++i) {
74 out << (i ? " L" : " ") << hole[i].x << " " << hole[i].y;
75 }
76 out << " Z";
77 }
78 out << "\"/>\n";
79 }
80 }
81 out << "</g>\n";
82
83 out << "<g fill=\"none\" stroke=\"#4a90d9\" stroke-width=\"0.1\">\n";
84 for (const Geom& g : follows) {
85 if (!g.valid()) continue;
86 const Ring pts = g.coords();
87 if (pts.size() < 2) continue;
88 out << "<polyline points=\"";
89 for (const Pt& p : pts) out << p.x << "," << p.y << " ";
90 out << "\"/>\n";
91 }
92 out << "</g>\n</g>\n</svg>\n";
93
94 std::ofstream file(path, std::ios::binary);
95 if (!file) throw std::runtime_error("cannot write SVG: " + path);
96 file << out.str();
97}
