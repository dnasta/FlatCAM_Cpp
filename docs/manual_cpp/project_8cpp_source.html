49/**
50 * @brief Detecta la firma xz (FD 37 7A 58 5A 00) al inicio del archivo.
51 * @param path Archivo a inspeccionar.
52 * @return true si el proyecto está comprimido con lzma/xz.
53 */
54bool isXz(const std::string& path) {
55 std::ifstream in(path, std::ios::binary);
56 if (!in) return false;
57 unsigned char magic[6] = {};
58 in.read(reinterpret_cast<char*>(magic), sizeof(magic));
59 return in.gcount() == 6 && magic[0] == 0xFD && magic[1] == '7' && magic[2] == 'z' &&
60 magic[3] == 'X' && magic[4] == 'Z' && magic[5] == 0;
61}

63#ifdef FC_HAVE_LZMA
64// Streaming decompressor: lets the SAX parser walk a multi-gigabyte .FlatPrj
65// without ever materialising it on disk or in memory.
66/**
67 * @brief streambuf que descomprime un .xz en streaming.
68 *
69 * Lee el archivo por bloques de 1 MiB, alimenta lzma_stream y expone el texto
70 * descomprimido a través de underflow(), de modo que nlohmann puede parsear
71 * proyectos de varios GiB con memoria acotada. Acepta streams concatenados
72 * (LZMA_CONCATENATED).
73 */
74class LzmaStreamBuf : public std::streambuf {
75public:
76 /**
77 * @brief Abre el archivo y prepara el decodificador lzma.
78 * @param path Ruta del .FlatPrj comprimido.
79 * @throw std::runtime_error si el archivo no se abre o lzma no inicializa.
80 */
81 explicit LzmaStreamBuf(const std::string& path) : in_(path, std::ios::binary) {
82 if (!in_) throw std::runtime_error("cannot open project: " + path);
83 stream_ = LZMA_STREAM_INIT;
84 if (lzma_stream_decoder(&stream_, UINT64_MAX, LZMA_CONCATENATED) != LZMA_OK)
85 throw std::runtime_error("lzma decoder initialisation failed");
86 setg(buffer_, buffer_, buffer_);
87 }
88
89 /// Libera el estado interno de liblzma.
90 ~LzmaStreamBuf() override { lzma_end(&stream_); }
91
92protected:
93 /**
94 * @brief Produce el siguiente bloque de bytes descomprimidos.
95 *
96 * Refleja la mecánica clásica de streambuf: si el área de lectura tiene
97 * datos los devuelve; si no, bombea entrada al decodificador hasta obtener
98 * al menos un byte, EOF o error. Al agotarse la entrada se usa LZMA_FINISH
99 * para cerrar el stream.
100 * @return Siguiente carácter, o traits_type::eof() al terminar.
101 * @throw std::runtime_error si lzma devuelve un código distinto de OK.
102 */
103 int_type underflow() override {
104 if (gptr() < egptr()) return traits_type::to_int_type(*gptr());
105
106 while (true) {
107 stream_.next_out = reinterpret_cast<uint8_t*>(buffer_);
108 stream_.avail_out = sizeof(buffer_);
109
110 if (stream_.avail_in == 0 && !inputEof_) {
111 in_.read(input_, sizeof(input_));
112 stream_.next_in = reinterpret_cast<const uint8_t*>(input_);
113 stream_.avail_in = static_cast<std::size_t>(in_.gcount());
114 if (in_.gcount() == 0) inputEof_ = true;
115 }
116
117 const lzma_ret ret = lzma_code(&stream_, inputEof_ ? LZMA_FINISH : LZMA_RUN);
118 const std::size_t produced = sizeof(buffer_) - stream_.avail_out;
119 if (produced > 0) {
120 setg(buffer_, buffer_, buffer_ + produced);
121 return traits_type::to_int_type(*gptr());
122 }
123 if (ret == LZMA_STREAM_END) return traits_type::eof();
124 if (ret != LZMA_OK)
125 throw std::runtime_error("lzma decompression failed (code " +
126 std::to_string(ret) + ")");
127 }
128 }
129
130private:
131 std::ifstream in_;
132 lzma_stream stream_{};
133 bool inputEof_ = false;
134 char input_[1 << 20];
135 char buffer_[1 << 20];
136};

139/**
140 * @brief Abre el proyecto como istream, descomprimiendo si hace falta.
141 * @param path Ruta del proyecto.
142 * @param[out] compressed Se marca si el archivo venía comprimido en xz.
143 * @return Stream listo para SAX o para lectura completa.
144 * @throw std::runtime_error si el archivo no se puede abrir, o si es xz y
145 * esta compilación no define FC_HAVE_LZMA.
146 */
147std::unique_ptr<std::istream> openProjectStream(const std::string& path, bool& compressed) {
148 compressed = isXz(path);
149 if (!compressed) {
150 auto stream = std::make_unique<std::ifstream>(path, std::ios::binary);
151 if (!*stream) throw std::runtime_error("cannot open project: " + path);
152 return stream;
153 }
154#ifdef FC_HAVE_LZMA
155 return std::make_unique<std::istream>(new LzmaStreamBuf(path));
156#else
157 throw std::runtime_error(
158 "this project is lzma-compressed and this build has no liblzma support");
159#endif
160}

162/**
163 * @brief Extrae geometrías de un campo solid_geometry o follow_geometry.
164 *
165 * Acepta las tres formas que aparecen en los proyectos FlatCAM: un string WKT
166 * suelto, un objeto con "__inst__" (Shply) y un "__inst__" que es a su vez una
167 * lista anidada, que se recorre recursivamente. Los elementos no reconocidos
168 * se omiten.
169 * @param value Valor JSON del campo.
170 * @return Geometrías parseadas, en el orden encontrado.
171 * @throw std::runtime_error si algún WKT es inválido (lo lanza Geom::fromWkt).
172 */
173std::vector<Geom> readGeometryField(const json& value) {
174 std::vector<Geom> out;
175
176 std::function<void(const json&)> readOne = [&out, &readOne](const json& item) {
177 if (item.is_string()) {
178 out.push_back(Geom::fromWkt(item.get<std::string>()));
179 return;
180 }
181 if (!item.is_object()) return;
182 const auto inst = item.find("__inst__");
183 if (inst == item.end()) return;
184 if (inst->is_string()) {
185 out.push_back(Geom::fromWkt(inst->get<std::string>()));
186 } else if (inst->is_array()) {
187 for (const json& sub : *inst) readOne(sub);
188 }
189 };
190
191 if (value.is_array()) {
192 for (const json& item : value) readOne(item);
193 } else {
194 readOne(value);
195 }
196 return out;
197}

199/// Envuelve una geometría en el objeto Shply con WKT de 9 decimales.
200json writeGeom(const Geom& geom) {
201 return json{{"__class__", "Shply"}, {"__inst__", geom.toWkt(9)}};

290 * @brief Handler SAX que materializa un único objeto del proyecto.

307 ProjectObject object;
308 std::vector<Geom> solids;
309 std::vector<Geom> follows;
310
311 bool null() override { return true; }
312 bool boolean(bool) override { return true; }
313 bool number_integer(number_integer_t) override { return true; }
314 bool number_unsigned(number_unsigned_t) override { return true; }
315 bool number_float(number_float_t, const string_t&) override { return true; }
316 bool binary(binary_t&) override { return true; }
317
318 bool string(string_t& value) override {
319 if (collecting) {
320 if (depth == 3 && currentKey == "kind") object.kind = value;
321 else if (depth == 3 && currentKey == "units") object.units = unitsFromString(value);
322 else if (depth == 3 && currentKey == "source_file") object.sourceFile = value;
323 else if (depth == 3 && currentKey == "text") object.text = value;
324 else if (depth == 3 && currentKey == "properties") object.properties = value;
325 else if (currentKey == "name") object.name = value;

345 mode = GeomMode::Solid;

347 } else if (collecting && mode == GeomMode::None && currentKey == "follow_geometry") {
348 mode = GeomMode::Follow;

352 }
353
354 bool key(string_t& value) override {
355 currentKey = value;
356 return true;
357 }
358
359 bool end_object() override {
360 if (mode != GeomMode::None && depth == geomDepth) mode = GeomMode::None;
361 if (inObjects && depth == 3) {
362 if (index == target) {
363 object.solids = std::move(solids);
364 object.follows = std::move(follows);
365 throw StopParsing{};
366 }
367 ++index;
368 }
369 --depth;
370 return true;
371 }
372
373 bool start_array(std::size_t) override {
374 ++depth;
375 if (depth == 2 && currentKey == "objs") inObjects = true;
376 else if (collecting && mode == GeomMode::None && currentKey == "solid_geometry") {
377 mode = GeomMode::Solid;
378 geomDepth = depth;
379 } else if (collecting && mode == GeomMode::None && currentKey == "follow_geometry") {
380 mode = GeomMode::Follow;
381 geomDepth = depth;
382 }
383 return true;
384 }
385
386 bool end_array() override {
387 if (mode != GeomMode::None && depth == geomDepth) mode = GeomMode::None;
388 if (inObjects && depth == 2) inObjects = false;
389 --depth;
390 return true;
391 }
392
393 bool parse_error(std::size_t, const std::string&, const nlohmann::detail::exception&) override {
394 return false;
395 }

400/**
401 * @brief Carga un proyecto completo en memoria.
402 *
403 * Abre el stream (descomprimiendo si es xz), lee por bloques controlando el
404 * tope kMaxLoadBytes y parsea el JSON completo. Luego recorre objs tomando la
405 * clase (por defecto "gerber"), el nombre (options.name o "<kind>_<i>"), las
406 * unidades, source_file, text, properties y ambas colecciones de geometría.
407 * @param path Ruta del .FlatPrj.
408 * @return Proyecto con todos sus objetos.
409 * @throw std::runtime_error si el archivo no se abre, supera el tope de
410 * memoria, el JSON es inválido o falta el array "objs".
411 * @note Pensado para proyectos chicos; los grandes deben usar
412 * inspectProject, forEachProjectObject o loadProjectObject.
413 */
414Project loadProject(const std::string& path) {
415 bool compressed = false;
416 std::unique_ptr<std::istream> in = openProjectStream(path, compressed);
417
418 std::string content;
419 char chunk[1 << 20];
420 while (in->good()) {
421 in->read(chunk, sizeof(chunk));
422 content.append(chunk, static_cast<std::size_t>(in->gcount()));
423 if (content.size() > kMaxLoadBytes) {
424 throw std::runtime_error(
425 "project is too large to load in memory; use "
426 "'flatcam-cli project --info' to inspect it");
427 }
428 }
429
430 json doc;
431 try {
432 doc = json::parse(content);
433 } catch (const json::exception& e) {
434 throw std::runtime_error(std::string("invalid project JSON: ") + e.what());
435 }
436
437 Project project;
438 if (doc.contains("version") && doc["version"].is_string())
439 project.version = doc["version"].get<std::string>();
440
441 if (doc.contains("project_options") && doc["project_options"].is_string())
442 project.options = doc["project_options"].get<std::string>();
443 if (!doc.contains("objs") || !doc["objs"].is_array())
444 throw std::runtime_error("project has no 'objs' array");
445
446 int index = 0;
447 for (const json& obj : doc["objs"]) {
448 if (!obj.is_object()) continue;
449 ProjectObject out;
450 if (obj.contains("kind") && obj["kind"].is_string())
451 out.kind = obj["kind"].get<std::string>();
452 else
453 out.kind = "gerber";
454
455 if (obj.contains("options") && obj["options"].is_object()) {
456 const json& options = obj["options"];
457 if (options.contains("name") && options["name"].is_string())
458 out.name = options["name"].get<std::string>();
459 }
460 if (out.name.empty()) out.name = out.kind + "_" + std::to_string(index);
461
462 if (obj.contains("units") && obj["units"].is_string())
463 out.units = unitsFromString(obj["units"].get<std::string>());
464
465 if (obj.contains("source_file") && obj["source_file"].is_string())
466 out.sourceFile = obj["source_file"].get<std::string>();
467 if (obj.contains("text") && obj["text"].is_string())
468 out.text = obj["text"].get<std::string>();
469 if (obj.contains("properties") && obj["properties"].is_string())
470 out.properties = obj["properties"].get<std::string>();
471
472 if (obj.contains("solid_geometry")) out.solids = readGeometryField(obj["solid_geometry"]);
473 if (obj.contains("follow_geometry")) out.follows = readGeometryField(obj["follow_geometry"]);
474
475 project.objects.push_back(std::move(out));
476 ++index;
477 }
478 return project;
479}

481/**
482 * @brief Escribe un proyecto en JSON legible, indentado con 2 espacios.
483 *
484 * Guarda versión, project_options y, por objeto, clase, unidades, archivo
485 * fuente, texto, propiedades, nombre y las geometrías en WKT. Los caracteres
486 * inválidos de UTF-8 se reemplazan para que el archivo siempre sea válido.
487 * @param path Destino.
488 * @param project Datos a serializar.
489 * @throw std::runtime_error si no se puede abrir el archivo.
490 * @note No comprime: el .xz lo produce otra herramienta.
491 */
492void saveProject(const std::string& path, const Project& project) {
493 json doc;
494 doc["version"] = project.version;
495 doc["options"] = json::object();
496 doc["project_options"] = project.options;
497 doc["objs"] = json::array();
498
499 for (const ProjectObject& obj : project.objects) {
500 json item;
501 item["kind"] = obj.kind;
502 item["units"] = unitsToString(obj.units);
503 item["source_file"] = obj.sourceFile;
504 item["text"] = obj.text;
505 item["properties"] = obj.properties;
506 item["options"] = json{{"name", obj.name}, {"plot", true}};
507 item["solid_geometry"] = writeGeometryField(obj.solids);
508 item["follow_geometry"] = writeGeometryField(obj.follows);
509 doc["objs"].push_back(std::move(item));
510 }
511
512 std::ofstream out(path, std::ios::binary);
513 if (!out) throw std::runtime_error("cannot write project: " + path);
514 out << doc.dump(2, ' ', false, json::error_handler_t::replace);
515}

517/**
518 * @brief Carga un solo objeto del proyecto sin leer el resto.
519 *
520 * Usa ObjectSax y corta la descompresión en cuanto el objeto está completo; en
521 * proyectos de cientos de MiB esto reduce el tiempo a fracciones de segundo.
522 * @param path Ruta del proyecto.
523 * @param index Índice del objeto dentro de "objs".
524 * @return Objeto con metadatos y geometrías; la clase y el nombre se completan
525 * con valores por defecto si faltaban.
526 * @throw std::runtime_error si el índice no existe o el archivo no se abre.
527 */
528ProjectObject loadProjectObject(const std::string& path, std::size_t index) {
529 bool compressed = false;
530 std::unique_ptr<std::istream> in = openProjectStream(path, compressed);
531
532 ObjectSax handler;
533 handler.target = index;
534 try {
535 json::sax_parse(*in, &handler);
536 } catch (const StopParsing&) {
537 // Expected: we stop as soon as the requested object is complete.
538 }
539
540 if (handler.object.kind.empty() && handler.object.solids.empty() &&
541 handler.object.follows.empty()) {
542 throw std::runtime_error("project object " + std::to_string(index) + " not found");
543 }
544 if (handler.object.kind.empty()) handler.object.kind = "object";
545 if (handler.object.name.empty())
546 handler.object.name = handler.object.kind + "_" + std::to_string(index);
547 return handler.object;
548}

622 * el resumen (suma de áreas y unión de cajas de los sólidos) y llama al
623 * visitante.
624 * @return false si el visitante pidió detener el recorrido.
625 */
626 bool finishObject() {
627 if (object.kind.empty()) object.kind = "object";
628 if (object.name.empty())
629 object.name = object.kind + "_" + std::to_string(summaries.size());
630 object.solids = std::move(solids);
631 object.follows = std::move(follows);
632
633 ProjectObjectSummary summary;
634 summary.kind = object.kind;
635 summary.name = object.name;
636 summary.units = object.units;
637 summary.solids = object.solids.size();
638 summary.follows = object.follows.size();
639 for (const Geom& g : object.solids) {
640 summary.area += g.area();
641 const BBox b = g.bounds();
642 if (!b.valid) continue;
643 if (!summary.bounds.valid) {
644 summary.bounds = b;
645 } else {
646 summary.bounds.xmin = std::min(summary.bounds.xmin, b.xmin);
647 summary.bounds.ymin = std::min(summary.bounds.ymin, b.ymin);
648 summary.bounds.xmax = std::max(summary.bounds.xmax, b.xmax);
649 summary.bounds.ymax = std::max(summary.bounds.ymax, b.ymax);
650 }
651 }
652 summaries.push_back(summary);
653 const bool keepGoing = visitor ? visitor(std::move(object)) : true;
654 object = ProjectObject{};
655 return keepGoing;
656 }
657
658 bool end_object() override {
659 if (mode != GeomMode::None && depth == geomDepth) mode = GeomMode::None;
660 if (inObjects && depth == 3) {
661 collecting = false;
662 if (!finishObject()) throw StopParsing{};
663 }
664 --depth;
665 return true;
666 }
667
668 bool start_array(std::size_t) override {
669 ++depth;
670 if (depth == 2 && currentKey == "objs") inObjects = true;
671 else if (collecting && mode == GeomMode::None && currentKey == "solid_geometry") {
672 mode = GeomMode::Solid;
673 geomDepth = depth;
674 } else if (collecting && mode == GeomMode::None && currentKey == "follow_geometry") {
675 mode = GeomMode::Follow;
676 geomDepth = depth;
677 }
678 return true;
679 }
680
681 bool end_array() override {
682 if (mode != GeomMode::None && depth == geomDepth) mode = GeomMode::None;
683 if (inObjects && depth == 2) inObjects = false;
684 --depth;
685 return true;
686 }

689 return false;
690 }
