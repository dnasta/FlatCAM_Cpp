44/**
45 * @brief Convierte polígonos GEOS en una única ruta Qt con agujeros.
46 *
47 * Recorre todas las geometrías y todos sus `PolygonContours`. Por cada contorno
48 * agrega el anillo exterior y luego cada agujero como subcamino independiente,
49 * cerrando ambos con `closeSubpath()`. La ruta se crea con regla de relleno
50 * **par-impar** (`Qt::OddEvenFill`): así un agujero contenido en el exterior
51 * queda vacío sin depender de la orientación de los anillos.
52 *
53 * @param geoms Sólidos a dibujar; puede estar vacío.
54 * @return Ruta lista para `QPainter::drawPath`, o ruta vacía si ningún anillo
55 * tiene al menos 3 puntos.
56 * @note Las coordenadas se copian sin transformar (unidades de mundo, mm por
57 * convención) y se descartan los contornos degenerados. No se aplica
58 * grosor de línea: el trazo es cosmético y lo decide el `QPen` del
59 * llamador.
60 */
61QPainterPath Canvas::buildPath(const std::vector<fc::Geom>& geoms) {
62 QPainterPath path;
63 path.setFillRule(Qt::OddEvenFill);
64 for (const fc::Geom& g : geoms) {
65 for (const fc::PolygonContours& poly : g.polygons()) {
66 if (poly.outer.size() < 3) continue;
67 QPolygonF outer;
68 outer.reserve(static_cast<int>(poly.outer.size()));
69 for (const fc::Pt& p : poly.outer) outer.append(QPointF(p.x, p.y));
70 path.addPolygon(outer);
71 path.closeSubpath();
72 for (const fc::Ring& hole : poly.holes) {
73 if (hole.size() < 3) continue;
74 QPolygonF h;
75 h.reserve(static_cast<int>(hole.size()));
76 for (const fc::Pt& p : hole) h.append(QPointF(p.x, p.y));
77 path.addPolygon(h);
78 path.closeSubpath();
79 }
80 }
81 }
82 return path;
83}

85/**
86 * @brief Agrega a una ruta los tramos de una geometría de tipo línea.
87 *
88 * Resuelve de forma recursiva `LINESTRING`/`LINEARRING` y las colecciones
89 * (`MULTILINESTRING`, `GEOMETRYCOLLECTION`, `MULTIPOLYGON`) recorriendo sus
90 * componentes con `geometryN()`. Para una línea usa `moveTo()` en el primer
91 * vértice y `lineTo()` en los restantes; no cierra el anillo porque un
92 * `LINEARRING` de GEOS ya repite el primer punto al final.
93 *
94 * @param path Ruta destino; se modifica en sitio.
95 * @param g Geometría de origen.
96 * @note Las geometrías sin componentes de línea (polígonos sueltos) se ignoran:
97 * los contornos rellenos se dibujan con buildPath, no aquí.
98 * @warning Sólo se soportan segmentos rectos; los arcos ya llegan aplanados
99 * desde el parser, así que no se pierde información práctica.
100 */
101void Canvas::appendLinePath(QPainterPath& path, const fc::Geom& g) {
102 if (!g.valid() || g.empty()) return;
103 const int type = g.typeId();
104 if (type == GEOS_LINESTRING || type == GEOS_LINEARRING) {
105 const fc::Ring pts = g.coords();
106 if (pts.size() < 2) return;
107 path.moveTo(pts.front().x, pts.front().y);
108 for (std::size_t i = 1; i < pts.size(); ++i) path.lineTo(pts[i].x, pts[i].y);
109 return;
110 }
111 if (type == GEOS_MULTILINESTRING || type == GEOS_GEOMETRYCOLLECTION ||
112 type == GEOS_MULTIPOLYGON) {
113 const int n = g.numGeometries();
114 for (int i = 0; i < n; ++i) appendLinePath(path, g.geometryN(i));
115 }
116}

161/**
162 * @brief Reemplaza todo el contenido del visor por una única capa.
163 *
164 * Descarta las capas existentes y los límites previos, construye la capa con
165 * las rutas de sólidos y recorridos, conserva la geometría GEOS y calcula la
166 * caja envolvente global: primero con los sólidos y, si ninguno aporta límites
167 * válidos, con los recorridos (caso típico de una capa sólo de G-code). Al
168 * final ajusta la vista y repinta.
169 *
170 * @param solids Polígonos de cobre.
171 * @param follows Recorridos abiertos.
172 * @note `hasData_` sólo queda activo si la caja resultante es válida; si no,
173 * el visor sigue mostrando el mensaje de bienvenida.
174 * @warning No limpia `toolpathPath_` (la previsualización CAM); para eso está
175 * clearData(). Tampoco reinicia `nextId_`, por lo que los
176 * identificadores siguen creciendo entre cargas.
177 */
178void Canvas::setGeometryData(const std::vector<fc::Geom>& solids,
179 const std::vector<fc::Geom>& follows) {
180 layers_.clear();
181 bounds_ = fc::BBox();
182 hasData_ = false;
183
184 Layer layer;
185 layer.id = nextId_++;
186 layer.solid = buildPath(solids);
187 layer.follow = buildLinePath(follows);
188 layer.solids = solids;
189 layer.follows = follows;
190 layers_.push_back(std::move(layer));
191
192 for (const fc::Geom& g : solids) {
193 const fc::BBox gb = g.bounds();
194 if (!gb.valid) continue;
195 if (!bounds_.valid) {
196 bounds_ = gb;
197 } else {
198 bounds_.xmin = std::min(bounds_.xmin, gb.xmin);
199 bounds_.ymin = std::min(bounds_.ymin, gb.ymin);
200 bounds_.xmax = std::max(bounds_.xmax, gb.xmax);
201 bounds_.ymax = std::max(bounds_.ymax, gb.ymax);
202 }
203 }
204 if (!bounds_.valid) {
205 for (const fc::Geom& g : follows) {
206 const fc::BBox gb = g.bounds();
207 if (!gb.valid) continue;
208 if (!bounds_.valid) bounds_ = gb;
209 else {
210 bounds_.xmin = std::min(bounds_.xmin, gb.xmin);
211 bounds_.ymin = std::min(bounds_.ymin, gb.ymin);
212 bounds_.xmax = std::max(bounds_.xmax, gb.xmax);
213 bounds_.ymax = std::max(bounds_.ymax, gb.ymax);
214 }
215 }
216 }
217 hasData_ = bounds_.valid;
218 fit();
219 update();
220}

237/**
238 * @brief Agrega una capa nueva durante una carga por lotes.
239 *
240 * Construye las rutas Qt de sólidos y recorridos, guarda la geometría GEOS,
241 * extiende la caja global con los límites válidos de cada geometría y
242 * devuelve el identificador asignado. La capa se agrega al final de la lista,
243 * es decir, encima de las anteriores.
244 *
245 * @param solids Polígonos de la capa.
246 * @param follows Recorridos de la capa.
247 * @param keepGeometry `true` conserva la geometría GEOS para edición,
248 * exportación y undo; `false` la descarta para ahorrar memoria en
249 * proyectos grandes (sólo queda el dibujo).
250 * @return Identificador de la capa recién creada.
251 * @note No repinta ni ajusta la vista: eso lo hace endGeometry().
252 */
253int Canvas::appendGeometry(const std::vector<fc::Geom>& solids,
254 const std::vector<fc::Geom>& follows, bool keepGeometry) {
255 Layer layer;
256 layer.id = nextId_++;
257 layer.solid = buildPath(solids);
258 layer.follow = buildLinePath(follows);
259 if (keepGeometry) {
260 layer.solids = solids;
261 layer.follows = follows;
262 }
263 layers_.push_back(std::move(layer));
264
265 /// Extiende la caja global con los límites válidos de una geometría.
266 auto grow = [this](const fc::BBox& gb) {
267 if (!gb.valid) return;
268 if (!bounds_.valid) {
269 bounds_ = gb;
270 } else {
271 bounds_.xmin = std::min(bounds_.xmin, gb.xmin);
272 bounds_.ymin = std::min(bounds_.ymin, gb.ymin);
273 bounds_.xmax = std::max(bounds_.xmax, gb.xmax);
274 bounds_.ymax = std::max(bounds_.ymax, gb.ymax);
275 }
276 };
277 for (const fc::Geom& g : solids) grow(g.bounds());
278 for (const fc::Geom& g : follows) grow(g.bounds());
279 hasData_ = bounds_.valid;
280 return layer.id;
281}

307/**
308 * @brief Reordena las capas según una lista de identificadores.
309 *
310 * Primero copia las capas en el orden pedido (ignorando los ids que no
311 * existen) y después agrega al final, conservando su orden relativo, las capas
312 * que no aparecían en la lista. Así nunca se pierde ninguna capa.
313 *
314 * @param ids Orden deseado, de abajo hacia arriba.
315 * @note El coste es O(n·m) con n capas y m ids, aceptable para decenas de
316 * capas. Los identificadores repetidos se ignoran y no se recalcula la
317 * caja (las capas no cambian de posición).
318 */
319void Canvas::setLayerOrder(const std::vector<int>& ids) {
320 std::vector<Layer> ordered;
321 ordered.reserve(layers_.size());
322 std::vector<int> seen;
323 for (int id : ids) {
324 if (std::find(seen.begin(), seen.end(), id) != seen.end()) continue;
325 for (const Layer& layer : layers_) {
326 if (layer.id == id) {
327 ordered.push_back(layer);
328 seen.push_back(id);
329 break;
330 }
331 }
332 }
333 for (const Layer& layer : layers_) {
334 const bool already = std::any_of(ordered.begin(), ordered.end(),
335 [&layer](const Layer& l) { return l.id == layer.id; });
336 if (!already) ordered.push_back(layer);
337 }
338 layers_ = std::move(ordered);
339 update();
340}

400/**
401 * @brief Calcula los límites de una capa a partir de sus rutas.
402 *
403 * Une el rectángulo envolvente de los sólidos con el de los recorridos (los
404 * recorridos pueden sobresalir del cobre, por ejemplo en un aislamiento) y
405 * convierte el resultado a `fc::BBox`.
406 *
407 * @param id Identificador de la capa.
408 * @return Caja válida en coordenadas de mundo, o caja inválida si la capa no
409 * existe o ambas rutas están vacías.
410 * @note Es más barato que recorrer la geometría GEOS y sirve para hacer zoom a
411 * un objeto; el lado superior/izquierdo de `QRectF` coincide con el
412 * mínimo de cada eje.
413 */
414fc::BBox Canvas::layerBounds(int id) const {
415 fc::BBox b;
416 const Layer* layer = findLayer(id);
417 if (!layer) return b;
418 QRectF rect = layer->solid.boundingRect();
419 const QRectF followRect = layer->follow.boundingRect();
420 if (!followRect.isEmpty()) rect = rect.isEmpty() ? followRect : rect.united(followRect);
421 if (rect.isEmpty()) return b;
422 b.xmin = rect.left();
423 b.ymin = rect.top();
424 b.xmax = rect.right();
425 b.ymax = rect.bottom();
426 b.valid = true;
427 return b;
428}

531 * @brief Recalcula la caja envolvente de todas las capas.
532 *
533 * Parte de una caja inválida y la extiende con la unión de las rutas de sólidos
534 * y recorridos de cada capa, incluidas las ocultas. Actualiza `hasData_` según

536 *
537 * @note No ajusta la vista ni repinta: eso queda a cargo del llamador, para
538 * permitir ediciones encadenadas sin saltos de zoom.

540void Canvas::recomputeBounds() {
541 bounds_ = fc::BBox();
542 for (const Layer& layer : layers_) {
543 QRectF rect = layer.solid.boundingRect();
544 const QRectF followRect = layer.follow.boundingRect();
545 if (!followRect.isEmpty()) rect = rect.isEmpty() ? followRect : rect.united(followRect);
546 if (rect.isEmpty()) continue;
547 fc::BBox b{rect.left(), rect.top(), rect.right(), rect.bottom()};
548 if (!bounds_.valid) {
549 bounds_ = b;
550 } else {
551 bounds_.xmin = std::min(bounds_.xmin, b.xmin);
552 bounds_.ymin = std::min(bounds_.ymin, b.ymin);
553 bounds_.xmax = std::max(bounds_.xmax, b.xmax);
554 bounds_.ymax = std::max(bounds_.ymax, b.ymax);
555 }
556 }
557 hasData_ = bounds_.valid;
558}

560/**
561 * @brief Combina las capas visibles en una capa nueva ("film" interno).
562 *
563 * Une los sólidos con `QPainterPath::united()` y concatena los recorridos con
564 * `addPath()`. El resultado se agrega encima de las demás capas; las originales
565 * no se ocultan ni se borran.
566 *
567 * @return Id de la capa combinada, o -1 si no hay nada visible con contenido.
568 * @note También concatena `solids`/`follows` de las capas visibles. Las capas
569 * cargadas en modo streaming (sin geometría) sólo aportan su dibujo, por
570 * lo que no se pueden transformar después.
571 * @note La unión de rutas es una operación costosa con mucho cobre; en placas
572 * grandes puede tardar segundos.
573 */
574int Canvas::mergeVisibleLayers() {
575 QPainterPath mergedSolid;
576 QPainterPath mergedFollow;
577 std::vector<fc::Geom> mergedSolids;
578 std::vector<fc::Geom> mergedFollows;
579 for (const Layer& layer : layers_) {
580 if (!layer.visible) continue;
581 mergedSolid = mergedSolid.united(layer.solid);
582 mergedFollow.addPath(layer.follow);
583 mergedSolids.insert(mergedSolids.end(), layer.solids.begin(), layer.solids.end());
584 mergedFollows.insert(mergedFollows.end(), layer.follows.begin(), layer.follows.end());
585 }
586 if (mergedSolid.isEmpty() && mergedFollow.isEmpty()) return -1;
587
588 Layer layer;
589 layer.id = nextId_++;
590 layer.solid = mergedSolid;
591 layer.follow = mergedFollow;
592 layer.solids = std::move(mergedSolids);
593 layer.follows = std::move(mergedFollows);
594 layers_.push_back(std::move(layer));
595 recomputeBounds();
596 update();
597 return layer.id;
598}

668/**
669 * @brief Aplica una transformación afín a la geometría de una capa.
670 *
671 * Recorre sólidos y recorridos y transforma cada vértice con
672 * `fc::Geom::transformXY`, usando el `QTransform` recibido como función de
673 * mapeo; después reconstruye las rutas Qt y repinta. Al trabajar sobre GEOS, la
674 * transformación también queda reflejada en las exportaciones.
675 *
676 * @param id Capa a transformar.
677 * @param transform Mover/rotar/escalar/sesgar en coordenadas de mundo.
678 * @note Recalcula límites y `hasData_`; el ajuste de vista y `exportSvg()`
679 * reflejan la transformación de inmediato.
680 */
681void Canvas::transformLayer(int id, const QTransform& transform) {
682 Layer* layer = findLayer(id);
683 if (!layer) return;
684 /// Mapea cada vértice de cada geometría con la transformación dada.
685 auto apply = [&transform](const std::vector<fc::Geom>& input) {
686 std::vector<fc::Geom> output;
687 output.reserve(input.size());
688 for (const fc::Geom& g : input) {
689 output.push_back(g.transformXY([&transform](fc::Pt p) {
690 const QPointF q = transform.map(QPointF(p.x, p.y));
691 return fc::Pt{q.x(), q.y()};
692 }));
693 }
694 return output;
695 };
696 layer->solids = apply(layer->solids);
697 layer->follows = apply(layer->follows);
698 layer->solid = buildPath(layer->solids);
699 layer->follow = buildLinePath(layer->follows);
700 recomputeBounds();
701 update();
702}

704/**
705 * @brief Exporta las capas visibles a un SVG (film de fabricación).
706 *
707 * Calcula un alto en píxeles proporcional a la caja (ancho fijo de 1400 px),
708 * define el `viewBox` en unidades de mundo y aplica una matriz que invierte el
709 * eje Y (`[1 0 0 -1 0 ymin+ymax]`) para que el SVG, cuyo origen está arriba,
710 * coincida con el mundo CAM, cuyo origen está abajo. Dibuja el cobre relleno y
711 * los recorridos como líneas.
712 *
713 * @param path Ruta de destino del archivo `.svg`.
714 * @return `true` si se escribió; `false` si no hay datos o si
715 * `QPainter::begin()` no pudo abrir el generador (ruta inválida, sin
716 * permisos, etc.).
717 * @warning Usa colores fijos (cobre `200,149,46`, recorridos `74,144,217`) e
718 * ignora `setColors`. No incluye rejilla, ejes, área de trabajo,
719 * trayectorias ni HUD, y no comprueba el resultado final de la
720 * escritura.
721 */
722bool Canvas::exportSvg(const QString& path) const {
723 if (!hasData_) return false;
724
725 const double w = std::max(1e-6, bounds_.width());
726 const double h = std::max(1e-6, bounds_.height());
727 const int pixelWidth = 1400;
728 const int pixelHeight = std::max(1, static_cast<int>(std::lround(pixelWidth * h / w)));
729
730 QSvgGenerator generator;
731 generator.setFileName(path);
732 generator.setSize(QSize(pixelWidth, pixelHeight));
733 generator.setViewBox(QRectF(bounds_.xmin, bounds_.ymin, w, h));
734 generator.setTitle(QStringLiteral("FlatCAM C++"));
735 generator.setDescription(QStringLiteral("Film export"));
736
737 QPainter painter;
738 if (!painter.begin(&generator)) return false;
739 painter.setRenderHint(QPainter::Antialiasing, true);
740 painter.setTransform(QTransform(1, 0, 0, -1, 0, bounds_.ymin + bounds_.ymax));
741
742 for (const Layer& layer : layers_) {
743 if (!layer.visible) continue;
744 painter.setBrush(QColor(200, 149, 46));
745 painter.setPen(QPen(QColor(240, 196, 106), 0.0));
746 painter.drawPath(layer.solid);
747
748 painter.setBrush(Qt::NoBrush);
749 painter.setPen(QPen(QColor(74, 144, 217), 0.0));
750 painter.drawPath(layer.follow);
751 }
752 painter.end();
753 return true;
754}

772/**
773 * @brief Ajusta escala y desplazamiento para encuadrar todo el contenido.
774 *
775 * Reserva un margen fijo de 40 px, calcula la escala como el menor factor que
776 * cabe en ancho y alto, descarta valores no finitos o no positivos (caja
777 * degenerada o widget diminuto, que caen a escala 1.0) y centra la caja en el
778 * widget invirtiendo el eje Y.
779 *
780 * @note No hace nada si `hasData_` es falso. No repinta: el llamador decide
781 * cuándo actualizar.
782 */
783void Canvas::fit() {
784 if (!hasData_) return;
785 const double w = std::max(1, width());
786 const double h = std::max(1, height());
787 const double bw = std::max(bounds_.width(), 1e-6);
788 const double bh = std::max(bounds_.height(), 1e-6);
789 scale_ = std::min((w - 40.0) / bw, (h - 40.0) / bh);
790 if (!std::isfinite(scale_) || scale_ <= 0.0) scale_ = 1.0;
791 const double cx = (bounds_.xmin + bounds_.xmax) / 2.0;
792 const double cy = (bounds_.ymin + bounds_.ymax) / 2.0;
793 offset_ = QPointF(w / 2.0 - cx * scale_, h / 2.0 + cy * scale_);
794}

796/// @brief Ajusta la vista incluyendo el origen de máquina cuando está cerca.
797void Canvas::fitWithOrigin() {
798 if (!hasData_) return;
799 fc::BBox view = bounds_;
800 const double diag = std::hypot(bounds_.width(), bounds_.height());
801 const double nx = std::clamp(0.0, bounds_.xmin, bounds_.xmax);
802 const double ny = std::clamp(0.0, bounds_.ymin, bounds_.ymax);
803 const double distance = std::hypot(nx, ny);
804 const bool inside = 0.0 >= bounds_.xmin && 0.0 <= bounds_.xmax &&
805 0.0 >= bounds_.ymin && 0.0 <= bounds_.ymax;
806 if (!inside && distance <= std::max(10.0, diag * 6.0)) {
807 view.xmin = std::min(view.xmin, 0.0);
808 view.ymin = std::min(view.ymin, 0.0);
809 view.xmax = std::max(view.xmax, 0.0);
810 view.ymax = std::max(view.ymax, 0.0);
811 }
812 const double pad = std::max(0.5, diag * 0.04);
813 view = view.expanded(pad);
814 const double w = std::max(1, width());
815 const double h = std::max(1, height());
816 scale_ = std::min((w - 40.0) / std::max(view.width(), 1e-6),
817 (h - 40.0) / std::max(view.height(), 1e-6));
818 if (!std::isfinite(scale_) || scale_ <= 0.0) scale_ = 1.0;
819 const double cx = (view.xmin + view.xmax) / 2.0;
820 const double cy = (view.ymin + view.ymax) / 2.0;
821 offset_ = QPointF(w / 2.0 - cx * scale_, h / 2.0 + cy * scale_);
822 update();
823}

945/**
946 * @brief Dibuja el lienzo completo.
947 *
948 * Orden de pintado: fondo, mensaje de bienvenida si no hay datos, rejilla
949 * adaptativa (paso automático que mantiene entre 40 y 200 px por celda),
950 * ejes en `displayOrigin_`, área de trabajo punteada, capas de abajo hacia
951 * arriba (multicolor por primitiva o cobre uniforme más recorridos),
952 * trayectorias CAM y, ya en coordenadas de dispositivo, el HUD con la posición
953 * del cursor en mm.
954 *
955 * @param event Evento de pintado (no se usa; se repinta todo).
956 * @note La transformación de mundo es `[scale 0 0 -scale ox oy]`, es decir, Y
957 * crece hacia arriba como en CAM. En modo multicolor se reconstruye la
958 * ruta de cada sólido en cada repintado: es correcto pero costoso con
959 * miles de primitivas.
960 * @warning No hay recorte por región visible (sólo el que aplica Qt); con zoom
961 * extremo el coste de dibujar todas las rutas sigue siendo el mismo.
962 */
963void Canvas::paintEvent(QPaintEvent*) {
964 QPainter p(this);
965 p.fillRect(rect(), background_);
966 if (!hasData_) {
967 p.setPen(QColor(120, 130, 140));
968 p.drawText(rect(), Qt::AlignCenter, tr("Abra un archivo Gerber, Excellon o proyecto"));
969 return;
970 }
971
972 QTransform t(scale_, 0.0, 0.0, -scale_, offset_.x(), offset_.y());
973 p.setTransform(t);
974
975 const fc::BBox view{(0.0 - offset_.x()) / scale_,
976 (offset_.y() - height()) / scale_,
977 (width() - offset_.x()) / scale_,
978 offset_.y() / scale_};
979 double step = gridStep_ > 0.0 ? gridStep_ : 10.0;
980 if (gridStep_ <= 0.0) {
981 while (step * scale_ < 40.0) step *= 10.0;
982 while (step * scale_ > 200.0) step /= 10.0;
983 }
984 if (gridVisible_) {
985 p.setPen(QPen(gridColor_, 0.0));
986 const double x0 = std::floor(view.xmin / step) * step;
987 const double y0 = std::floor(view.ymin / step) * step;
988 for (double x = x0; x <= view.xmax + step; x += step)
989 p.drawLine(QPointF(x, view.ymin - step), QPointF(x, view.ymax + step));
990 for (double y = y0; y <= view.ymax + step; y += step)
991 p.drawLine(QPointF(view.xmin - step, y), QPointF(view.xmax + step, y));
992 }
993
994 // Ejes de máquina X=0 / Y=0 resaltados. Son la referencia desde la que
995 // arranca el cabezal, distinta del origen de visualización configurable.
996 const bool darkCanvas = background_.lightness() < 128;
997 const QColor axisX = darkCanvas ? QColor(226, 108, 117, 230) : QColor(190, 40, 55, 230);
998 const QColor axisY = darkCanvas ? QColor(96, 200, 136, 230) : QColor(20, 130, 70, 230);
999 const double axisWidth = 2.0 / std::max(1.0, scale_);
1000 if (view.ymin <= 0.0 && view.ymax >= 0.0) {
1001 p.setPen(QPen(axisX, axisWidth));
1002 p.drawLine(QPointF(view.xmin, 0.0), QPointF(view.xmax, 0.0));
1003 }
1004 if (view.xmin <= 0.0 && view.xmax >= 0.0) {
1005 p.setPen(QPen(axisY, axisWidth));
1006 p.drawLine(QPointF(0.0, view.ymin), QPointF(0.0, view.ymax));
1007 }
1008
1009 if (std::abs(displayOrigin_.x) > 1e-9 || std::abs(displayOrigin_.y) > 1e-9) {
1010 p.setPen(QPen(QColor(120, 130, 140, 150), 0.0, Qt::DashLine));
1011 p.drawLine(QPointF(bounds_.xmin, displayOrigin_.y),
1012 QPointF(bounds_.xmax, displayOrigin_.y));
1013 p.drawLine(QPointF(displayOrigin_.x, bounds_.ymin),
1014 QPointF(displayOrigin_.x, bounds_.ymax));
1015 }
1016
1017 if (workspace_.valid) {
1018 QPen pen(QColor(120, 90, 200), 0.0, Qt::DashLine);
1019 p.setPen(pen);
1020 p.setBrush(Qt::NoBrush);
1021 p.drawRect(QRectF(workspace_.xmin, workspace_.ymin, workspace_.width(),
1022 workspace_.height()));
1023 }
1024
1025 for (const Layer& layer : layers_) {
1026 if (!layer.visible) continue;
1027 if (layer.multicolor) {
1028 for (std::size_t i = 0; i < layer.solids.size(); ++i) {
1029 QColor color = QColor::fromHsv(static_cast<int>((i * 47) % 360), 180, 220);
1030 color.setAlpha(190);
1031 p.setBrush(layer.filled ? QBrush(color) : Qt::NoBrush);
1032 p.setPen(QPen(color.lighter(125), 0.0));
1033 p.drawPath(buildPath({layer.solids[i]}));
1034 }
1035 } else {
1036 p.setBrush(layer.filled ? QBrush(copperFill_) : Qt::NoBrush);
1037 p.setPen(QPen(copperEdge_, 0.0));
1038 p.drawPath(layer.solid);
1039 }
1040
1041 p.setBrush(Qt::NoBrush);
1042 QPen followPen(followColor_, layer.width);
1043 followPen.setCapStyle(Qt::RoundCap);
1044 followPen.setJoinStyle(Qt::RoundJoin);
1045 p.setPen(followPen);
1046 p.drawPath(layer.follow);
1047 }
1048
1049 p.setBrush(Qt::NoBrush);
1050 p.setPen(QPen(toolpathColor_, 0.0));
1051 p.drawPath(toolpathPath_);
1052
1053 p.setTransform(QTransform());
1054 p.setRenderHint(QPainter::TextAntialiasing, true);
1055 QFont small = p.font();
1056 small.setPointSizeF(9.0);
1057 p.setFont(small);
1058 const auto toScreen = [this](double x, double y) {
1059 return QPointF(x * scale_ + offset_.x(), -y * scale_ + offset_.y());
1060 };
1061 const QRect viewport = rect();
1062 const auto inside = [&viewport](const QPointF& p) {
1063 return viewport.adjusted(-2, -2, 2, 2).contains(p.toPoint());
1064 };
1065 if (!markers_.empty()) {
1066 const QColor markerColor = darkCanvas ? QColor(255, 190, 90, 240)
1067 : QColor(170, 95, 15, 240);
1068 p.setPen(QPen(markerColor, 1.5));
1069 p.setBrush(Qt::NoBrush);
1070 for (const fc::Pt& marker : markers_) {
1071 const QPointF s = toScreen(marker.x, marker.y);
1072 if (!inside(s)) continue;
1073 p.drawEllipse(s, 4.0, 4.0);
1074 p.drawLine(s + QPointF(-7, 0), s + QPointF(7, 0));
1075 p.drawLine(s + QPointF(0, -7), s + QPointF(0, 7));
1076 }
1077 }
1078
1079 // Rótulos de los ejes y marcas de coordenadas.
1080 const QPointF originScreen = toScreen(0.0, 0.0);
1081 const QColor originColor = darkCanvas ? QColor(235, 238, 242, 230)
1082 : QColor(20, 24, 30, 230);
1083 if (inside(originScreen)) {
1084 p.setPen(QPen(originColor, 1.2));
1085 p.setBrush(Qt::NoBrush);
1086 p.drawEllipse(originScreen, 5.0, 5.0);
1087 p.drawLine(originScreen + QPointF(-8, 0), originScreen + QPointF(8, 0));
1088 p.drawLine(originScreen + QPointF(0, -8), originScreen + QPointF(0, 8));
1089 p.setPen(originColor);
1090 p.drawText(originScreen + QPointF(8, -7), QStringLiteral("0,0"));
1091 }
1092 if (inside(toScreen(0.0, 0.0))) {
1093 p.setPen(axisX);
1094 p.drawText(viewport.adjusted(8, 0, -8, -8), Qt::AlignRight | Qt::AlignBottom,
1095 QStringLiteral("Y=0"));
1096 p.setPen(axisY);
1097 p.drawText(viewport.adjusted(8, 8, -8, -8), Qt::AlignLeft | Qt::AlignTop,
1098 QStringLiteral("X=0"));
1099 }
1100 if (step * scale_ >= 28.0) {
1101 const QColor tickColor = darkCanvas ? QColor(160, 168, 178, 190)
1102 : QColor(85, 95, 105, 200);
1103 p.setPen(tickColor);
1104 const double x0 = std::floor(view.xmin / step) * step;
1105 for (double x = x0; x <= view.xmax + step && view.ymin <= 0.0 && view.ymax >= 0.0;
1106 x += step) {
1107 if (std::abs(x) < step * 0.01) continue;
1108 const QPointF s = toScreen(x, 0.0);
1109 if (!inside(s)) continue;
1110 p.drawLine(s + QPointF(0, -3), s + QPointF(0, 3));
1111 p.drawText(s + QPointF(3, -4), QString::number(x, 'f', step < 1.0 ? 2 : 0));
1112 }
1113 const double y0 = std::floor(view.ymin / step) * step;
1114 for (double y = y0; y <= view.ymax + step && view.xmin <= 0.0 && view.xmax >= 0.0;
1115 y += step) {
1116 if (std::abs(y) < step * 0.01) continue;
1117 const QPointF s = toScreen(0.0, y);
1118 if (!inside(s)) continue;
1119 p.drawLine(s + QPointF(-3, 0), s + QPointF(3, 0));
1120 p.drawText(s + QPointF(5, 3), QString::number(y, 'f', step < 1.0 ? 2 : 0));
1121 }
1122 }
1123
1124 // Recorrido desde el origen de máquina hasta la PCB y medidas del área.
1125 if (bounds_.valid) {
1126 const double nx = std::clamp(0.0, bounds_.xmin, bounds_.xmax);
1127 const double ny = std::clamp(0.0, bounds_.ymin, bounds_.ymax);
1128 const QPointF target = toScreen(nx, ny);
