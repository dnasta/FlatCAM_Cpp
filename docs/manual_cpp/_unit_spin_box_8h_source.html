13/**
14 * @brief Campo numérico con unidad de longitud conmutable.
15 *
16 * Internamente guarda el valor en milímetros. Cuando el sufijo es exactamente
17 * `" mm"` muestra el valor en la unidad activa (mm o pulgadas) y convierte los
18 * límites, el paso y los decimales. Ángulos, RPM, avances y porcentajes se
19 * comportan como un `QDoubleSpinBox` normal.
20 *
21 * Un registro estático permite actualizar todos los campos abiertos al cambiar
22 * de unidad sin reiniciar la aplicación.
23 */
24class UnitSpinBox : public QDoubleSpinBox {

90 /// Aplica la unidad activa a los valores crudos en mm.
91 void applyConversion() {
92 if (!length_) {
93 QDoubleSpinBox::setDecimals(rawDecimals_);
94 QDoubleSpinBox::setRange(rawMin_, rawMax_);
95 QDoubleSpinBox::setSingleStep(rawStep_);
96 QDoubleSpinBox::setValue(rawValue_);
97 QDoubleSpinBox::setSuffix(rawSuffix_);
98 return;
99 }
100 const double factor = displayInch_ ? 25.4 : 1.0;
101 QDoubleSpinBox::setDecimals(displayInch_ ? rawDecimals_ + 4 : rawDecimals_);
102 QDoubleSpinBox::setRange(rawMin_ / factor, rawMax_ / factor);
103 QDoubleSpinBox::setSingleStep(rawStep_ / factor);
104 QDoubleSpinBox::setValue(rawValue_ / factor);
105 QDoubleSpinBox::setSuffix(displayInch_ ? " in" : " mm");
106 }
