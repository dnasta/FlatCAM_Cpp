15/**
16 * @brief Campo numérico con unidad de longitud conmutable.
17 *
18 * Internamente guarda el valor en milímetros. Cuando el sufijo es exactamente
19 * `" mm"` muestra el valor en la unidad activa (mm o pulgadas) y convierte los
20 * límites, el paso y los decimales. Ángulos, RPM, avances y porcentajes se
21 * comportan como un `QDoubleSpinBox` normal.
22 *
23 * Un registro estático permite actualizar todos los campos abiertos al cambiar
24 * de unidad sin reiniciar la aplicación.
25 */
26class UnitSpinBox : public QDoubleSpinBox {
27public:
28 /// @brief Crea el campo y lo registra para el cambio de unidad.
29 explicit UnitSpinBox(QWidget* parent = nullptr) : QDoubleSpinBox(parent) {
30 // El programa usa punto decimal en el G-code y en las tablas; forzar
31 // la locale C evita que el mismo valor se muestre con coma.
32 setLocale(QLocale::c());
33 instances().push_back(this);
34 applyConversion();
35 }
36
37 /// @brief Quita el campo del registro.
38 ~UnitSpinBox() override {
39 auto& list = instances();
40 list.erase(std::remove(list.begin(), list.end(), this), list.end());
41 }

121 /// Aplica la unidad activa a los valores crudos en mm.
122 void applyConversion() {
123 if (!length_) {
124 QDoubleSpinBox::setDecimals(rawDecimals_);
125 QDoubleSpinBox::setRange(rawMin_, rawMax_);
126 QDoubleSpinBox::setSingleStep(rawStep_);
127 QDoubleSpinBox::setValue(rawValue_);
128 QDoubleSpinBox::setSuffix(rawSuffix_);
129 return;
130 }
131 const double factor = displayInch_ ? 25.4 : 1.0;
132 QDoubleSpinBox::setDecimals(displayInch_ ? rawDecimals_ + 4 : rawDecimals_);
133 QDoubleSpinBox::setRange(rawMin_ / factor, rawMax_ / factor);
134 QDoubleSpinBox::setSingleStep(rawStep_ / factor);
135 QDoubleSpinBox::setValue(rawValue_ / factor);
136 QDoubleSpinBox::setSuffix(displayInch_ ? " in" : " mm");
137 }
138
139 double rawMin_ = 0.0;

142 double rawStep_ = 1.0;
143 int rawDecimals_ = 2;
144 QString rawSuffix_;
145 bool length_ = false;
146 static inline bool displayInch_ = false;
