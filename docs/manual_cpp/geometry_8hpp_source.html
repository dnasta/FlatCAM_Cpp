1/**
2 * @file geometry.hpp
3 * @brief Núcleo geométrico inmutable sobre la API C de GEOS.
4 *
5 * FlatCAM C++ usa GEOS para todas las operaciones 2D: uniones, diferencias,
6 * intersecciones, buffers, simplificación, reparación y transformaciones.
7 * Este archivo define los tipos básicos (`Pt`, `Ring`, `BBox`,
8 * `PolygonContours`) y dos envoltorios RAII:
9 *
10 * - `fc::Geom`: geometría inmutable con copia barata (puntero compartido).
11 * - `fc::SpatialIndex`: índice STRtree para consultas por caja envolvente.
12 *
13 * GEOS no es reentrante en su API clásica; `initGeosRuntime()` se llama de
14 * forma automática desde las fábricas de `Geom`.
15 */

28/**
29 * @brief Punto bidimensional en milímetros.
30 *
31 * El eje X crece hacia la derecha y el eje Y hacia arriba, igual que en
32 * Gerber/Excellon y en el canvas.
33 */
34struct Pt {
35 double x = 0.0; ///< Coordenada X en milímetros.
36 double y = 0.0; ///< Coordenada Y en milímetros.
37};
38
39/// Secuencia de puntos que describe un anillo, una polilínea o un contorno.
40using Ring = std::vector<Pt>;
41
42/**
43 * @brief Caja envolvente alineada a los ejes.
44 *
45 * `valid` distingue una caja vacía (sin geometría) de una caja real en el
46 * origen. Las operaciones de expansión y contención respetan esa bandera.
47 */
48struct BBox {
49 double xmin = 0.0; ///< Límite inferior en X.
50 double ymin = 0.0; ///< Límite inferior en Y.
51 double xmax = 0.0; ///< Límite superior en X.
52 double ymax = 0.0; ///< Límite superior en Y.
53 bool valid = false; ///< `true` si la caja contiene geometría.
54
55 BBox() = default;
56
57 /**
58 * @brief Construye una caja válida a partir de sus límites.
59 * @param x0 Límite inferior X.
60 * @param y0 Límite inferior Y.
61 * @param x1 Límite superior X.
62 * @param y1 Límite superior Y.
63 */
64 BBox(double x0, double y0, double x1, double y1)
65 : xmin(x0), ymin(y0), xmax(x1), ymax(y1), valid(true) {}
66
67 /// @return Ancho de la caja (puede ser negativo si está mal construida).
68 double width() const { return xmax - xmin; }
69
70 /// @return Alto de la caja (puede ser negativo si está mal construida).
71 double height() const { return ymax - ymin; }
72
73 /**
74 * @brief Indica si un punto cae dentro de la caja.
75 * @param p Punto a comprobar.
76 * @return `true` si la caja es válida y el punto está dentro o en el borde.
77 */
78 bool contains(const Pt& p) const {
79 return valid && p.x >= xmin && p.x <= xmax && p.y >= ymin && p.y <= ymax;
80 }
81
82 /**
83 * @brief Expande la caja un margen uniforme.
84 * @param margin Distancia en milímetros (puede ser negativa).
85 * @return Nueva caja expandida; la original si no era válida.
86 */
87 BBox expanded(double margin) const {
88 if (!valid) return *this;
89 return BBox{xmin - margin, ymin - margin, xmax + margin, ymax + margin};
90 }
91};

104/**

107 * Las copias comparten el mismo `GEOSGeometry` mediante `shared_ptr`; por eso

322/**
323 * @brief Índice espacial envolvente de `GEOSSTRtree`.
324 *
325 * Inserta geometrías con un identificador entero y permite consultar qué
326 * identificadores intersectan una caja. Es útil para acelerar la detección de
327 * colisiones y la selección de objetos.
328 *
329 * El índice no es copiable y debe construirse (`build()`) antes de consultar.
330 */
331class SpatialIndex {
332public:
333 /// Crea un índice vacío.
334 SpatialIndex();
335
336 /// Libera el STRtree de GEOS.
337 ~SpatialIndex();
338
339 SpatialIndex(const SpatialIndex&) = delete;
340 SpatialIndex& operator=(const SpatialIndex&) = delete;
341
342 /**
343 * @brief Inserta una geometría con un identificador.
344 * @param g Geometría a indexar (se conserva una copia compartida).
345 * @param id Identificador devuelto en las consultas.
346 */
347 void insert(const Geom& g, std::uint32_t id);
348
349 /// Construye el árbol; obligatorio antes de `query()`.
350 void build();
351
352 /**
353 * @brief Consulta los identificadores que intersectan una caja.
354 * @param box Caja de búsqueda.
355 * @return Identificadores en orden no especificado.
356 */
357 std::vector<std::uint32_t> query(const BBox& box) const;
358
359private:
360 GEOSSTRtree* tree_ = nullptr; ///< Árbol propiedad de esta clase.
361 std::vector<Geom> keep_; ///< Mantiene vivas las geometrías indexadas.
362 std::vector<std::uint32_t> ids_; ///< Identificadores paralelos a `keep_`.
363 bool built_ = false; ///< Evita reconstruir o consultar antes de tiempo.
364};
