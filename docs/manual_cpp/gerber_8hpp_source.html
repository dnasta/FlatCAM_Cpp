1/**
2 * @file gerber.hpp
3 * @brief Parser de Gerber RS-274X (capas de PCB).
4 *
5 * El parser lee el archivo completo o una cadena, interpreta el formato
6 * (`FS`), las unidades (`MO`/`G70`/`G71`), aperturas estándar y macros,
7 * trazos, arcos, regiones, polaridad y step & repeat. La salida contiene:
8 *
9 * - sólidos (polígonos rellenados, ya en unidades de la aplicación),
10 * - recorridos (trazos y puntos),
11 * - aperturas y macros para el editor,
12 * - avisos de elementos no soportados.
13 *
14 * La geometría se construye con `fc::Geom` (GEOS) y se mantiene en milímetros,
15 * salvo la copia por apertura que se guarda en unidades del archivo para
16 * editarla.
17 */

31/**
32 * @brief Opciones de importación de Gerber.
33 */
34struct GerberOptions {
35 Units appUnits = Units::Mm; ///< Unidad de salida de la geometría.
36 int stepsPerCircle = 64; ///< Segmentos por círculo (FlatCAM: `gerber_circle_steps`).
37 bool simplification = false; ///< Simplifica la geometría al terminar.
38 double simplifyTolerance = 0.005; ///< Tolerancia de simplificación en mm.
39 /// Reproduce el bug de FlatCAM 2024.4 al trazar con apertura rectangular:
40 /// usa una caja entre los dos primeros puntos en vez de un barrido
41 /// Minkowski. Se activa sólo para comparar geometría con el original.
42 bool legacyRectStroke = false;
43};

45/**
46 * @brief Definición de una apertura Gerber y las primitivas dibujadas con ella.
47 */
48struct GerberAperture {
49 /// Tipo de apertura.
50 enum class Kind {
51 Circle, ///< Círculo (`C`).
52 Rectangle, ///< Rectángulo (`R`).
53 Obround, ///< Óvalo (`O`).
54 Polygon, ///< Polígono regular (`P`).
55 Macro, ///< Macro de apertura (`AM`).
56 Region ///< Región `G36/G37` sin apertura explícita.
57 };
58
59 Kind kind = Kind::Circle; ///< Tipo.
60 double size = 0.0; ///< Tamaño principal (diámetro o lado).
61 double width = 0.0; ///< Ancho para rectángulos/obrounds.
62 double height = 0.0; ///< Alto para rectángulos/obrounds.
63 double diameter = 0.0; ///< Diámetro para círculos/polígonos.
64 int vertices = 0; ///< Vértices para polígonos.
65 double rotation = 0.0; ///< Rotación en grados.
66 bool hasRotation = false; ///< `true` si el archivo especificó rotación.
67 std::string macroName; ///< Nombre de la macro (si es `Macro`).
68 std::vector<double> modifiers; ///< Parámetros de la apertura/macro.
69 std::vector<Geom> geometry; ///< Primitivas dibujadas con esta apertura (unidades del archivo).
70};

80/**
81 * @brief Resultado completo de parsear un Gerber.
82 *
83 * `solids` y `follows` están en unidades de la aplicación (mm). El campo
84 * `solid` es la unión de todos los sólidos y se usa para área y límites.
85 */
86struct GerberResult {
87 Units fileUnits = Units::Inch; ///< Unidades declaradas por el archivo.
88 int intDigits = 3; ///< Dígitos enteros del formato `FS`.
89 int fracDigits = 4; ///< Dígitos fraccionarios del formato `FS`.
90 char zeros = 'L'; ///< Supresión de ceros (`L`, `T`, `D`).
91 bool incremental = false; ///< `true` si el archivo usa coordenadas incrementales.
92
93 std::map<std::string, GerberAperture> apertures; ///< Aperturas por código `Dnn`.
94 std::map<std::string, GerberMacro> macros; ///< Macros `AM` por nombre.
95
96 std::vector<Geom> solids; ///< Polígonos oscuros, en unidades de la app.
97 std::vector<Geom> follows; ///< Trazos y puntos, en unidades de la app.
98 Geom solid; ///< Unión de todos los sólidos (cobre).
99 BBox bounds; ///< Límites de la geometría.
100
101 std::size_t flashes = 0; ///< Cantidad de flashes `D03`.
102 std::size_t segments = 0; ///< Cantidad de trazos.
103 std::size_t regions = 0; ///< Cantidad de regiones `G36/G37`.
104 std::vector<std::string> warnings; ///< Avisos no fatales.
105
106 /// @return `true` si no se generó cobre.
107 bool empty() const { return solid.empty(); }
108};

131 GerberResult parseFile(const std::string& path) const;

139 GerberResult parseString(std::string_view source) const;

141private:
