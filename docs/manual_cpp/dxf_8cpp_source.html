19/**
20 * @brief Escribe una polilínea DXF (LWPOLYLINE) en el buffer de salida.
21 *
22 * Emite los pares grupo mínimos: código 8 (capa "0"), 90 (cantidad de
23 * vértices), 70 (bandera de cierre) y los pares 10/20 por cada vértice.
24 * @param out Flujo de texto donde se acumula el documento DXF.
25 * @param pts Vértices en milímetros; con menos de 2 no se escribe nada.
26 * @param closed true para marcar la polilínea como cerrada (código 70 = 1).
27 */
28void writePolyline(std::ostringstream& out, const Ring& pts, bool closed) {
29 if (pts.size() < 2) return;
30 out << "0\nLWPOLYLINE\n8\n0\n90\n" << pts.size() << "\n70\n" << (closed ? 1 : 0) << "\n";
31 for (const Pt& p : pts) {
32 out << "10\n" << p.x << "\n20\n" << p.y << "\n";
33 }
34}

38/**
39 * @brief Exporta sólidos y recorridos a un archivo DXF completo.
40 *
41 * Arma en memoria las secciones HEADER ($ACADVER = AC1015, $INSUNITS = 4) y
42 * ENTITIES. Los sólidos se escriben como polilíneas cerradas (el contorno de
43 * cada polígono y uno por hueco); los recorridos se escriben abiertos,
44 * aceptando LineString/LinearRing simples o colecciones, de las que se toman
45 * sólo las partes lineales.
46 * @param path Ruta de destino; se sobrescribe si ya existe.
47 * @param solids Polígonos de cobre (con huecos) a exportar.
48 * @param follows Contornos abiertos (aislamiento, fresado) a exportar.
49 * @throw std::runtime_error si el archivo no se puede abrir para escritura.
50 */
51void exportDxf(const std::string& path, const std::vector<Geom>& solids,
52 const std::vector<Geom>& follows) {
53 std::ostringstream out;
54 out << "0\nSECTION\n2\nHEADER\n";
55 out << "9\n$ACADVER\n1\nAC1015\n";
56 out << "9\n$INSUNITS\n70\n4\n";
57 out << "0\nENDSEC\n";
58 out << "0\nSECTION\n2\nENTITIES\n";
59
60 for (const Geom& g : solids) {
61 for (const PolygonContours& pc : g.polygons()) {
62 writePolyline(out, pc.outer, true);
63 for (const Ring& hole : pc.holes) writePolyline(out, hole, true);
64 }
65 }
66 for (const Geom& g : follows) {
67 if (!g.valid() || g.empty()) continue;
68 if (g.typeId() == GEOS_LINESTRING || g.typeId() == GEOS_LINEARRING) {
69 writePolyline(out, g.coords(), false);
70 } else {
71 for (int i = 0; i < g.numGeometries(); ++i) {
72 const Geom part = g.geometryN(i);
73 if (part.typeId() == GEOS_LINESTRING || part.typeId() == GEOS_LINEARRING)
74 writePolyline(out, part.coords(), false);
75 }
76 }
77 }
78
79 out << "0\nENDSEC\n0\nEOF\n";
80
81 std::ofstream file(path, std::ios::binary);
82 if (!file) throw std::runtime_error("cannot write DXF: " + path);
83 file << out.str();
84}
