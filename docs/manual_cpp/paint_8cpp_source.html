20/// @brief Acumula `b` dentro de `target`, ignorando cajas inválidas.
21void addBounds(BBox& target, const BBox& b) {
22 if (!b.valid) return;
23 if (!target.valid) {
24 target = b;
25 return;
26 }
27 target.xmin = std::min(target.xmin, b.xmin);
28 target.ymin = std::min(target.ymin, b.ymin);
29 target.xmax = std::max(target.xmax, b.xmax);
30 target.ymax = std::max(target.ymax, b.ymax);
31}

41/**
42 * @brief Rellena una región con recorridos concéntricos paralelos al contorno.
43 *
44 * Pasos del algoritmo:
45 * 1. Se expande la región con `buffer(margin)` si el margen es distinto de 0.
46 * 2. Se contrae un radio de herramienta para que el centro de la fresa quede
47 * siempre dentro del área original (compensación de radio).
48 * 3. En cada iteración se emiten el contorno exterior y —si
49 * `options.includeHoles`— los huecos de cada polígono, y luego se contrae
50 * la región otro `paso`.
51 * 4. El bucle corta cuando la región queda vacía, cuando el área deja de
52 * disminuir (delta menor que 1e-9 relativo al área inicial, señal de
53 * colapso numérico) o al alcanzar 10000 iteraciones.
54 *
55 * @param area Región a pintar (normalmente el cobre de un Gerber).
56 * @param options Diámetro de herramienta, superposición, margen, sentido y
57 * si se pintan los huecos.
58 * @return Recorridos del centro de la fresa, longitud total y límites.
59 *
60 * @note El paso efectivo es `stepOver × toolDiameter` con un mínimo de
61 * 0,01 mm; `stepOver` es una fracción (0,5 = 50 % de superposición).
62 * @note `result.bounds` corresponde a la región con margen, no al conjunto de
63 * recorridos, por lo que puede ser algo mayor que el área realmente
64 * mecanizada.
65 * @warning Si el diámetro es mayor que la región, el buffer negativo la vacía
66 * y el resultado queda sin recorridos (no es un error).
67 */
68PaintResult paintArea(const Geom& area, const PaintOptions& options) {
69 PaintResult result;
70 if (!area.valid() || area.empty()) return result;
71
72 const double radius = std::max(0.0, options.toolDiameter / 2.0);
73 const double step = std::max(0.01, options.stepOver * options.toolDiameter);
74 const int quad = std::max(2, options.quadrantSegments);
75
76 Geom region = options.margin != 0.0 ? area.buffer(options.margin, quad) : area;
77 if (!region.valid() || region.empty()) return result;
78 addBounds(result.bounds, region.bounds());
79
80 // The tool centre must stay one radius inside the region.
81 Geom current = radius > 0.0 ? region.buffer(-radius, quad) : region;
82 if (!current.valid() || current.empty()) return result;
83
84 const double initialArea = current.area();
85 for (int iteration = 0; iteration < 10000; ++iteration) {
86 if (!current.valid() || current.empty()) break;
87
88 for (const PolygonContours& pc : current.polygons()) {
89 Geom outer = Geom::line(oriented(pc.outer, options.invert));
90 if (outer.valid() && !outer.empty()) {
91 result.totalLength += outer.length();
92 result.paths.push_back(std::move(outer));
93 }
94 if (options.includeHoles) {
95 for (const Ring& hole : pc.holes) {
96 Geom path = Geom::line(oriented(hole, options.invert));
97 if (path.valid() && !path.empty()) {
98 result.totalLength += path.length();
99 result.paths.push_back(std::move(path));
100 }
101 }
102 }
103 }
104
105 Geom next = current.buffer(-step, quad);
106 if (!next.valid() || next.empty()) break;
107 if (std::fabs(next.area() - current.area()) < 1e-9 * std::max(1.0, initialArea)) break;
108 current = std::move(next);
109 }
110
111 return result;
112}
