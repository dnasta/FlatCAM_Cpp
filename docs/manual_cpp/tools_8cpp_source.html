24/**
25 * @brief Une un vector de geometrías ignorando nulos y vacíos.
26 * @return Unión; Geom nulo si no queda ninguna parte utilizable.
27 */
28Geom unionAll(const std::vector<Geom>& parts) {
29 std::vector<Geom> valid;
30 valid.reserve(parts.size());
31 for (const Geom& g : parts) {
32 if (g.valid() && !g.empty()) valid.push_back(g);
33 }
34 if (valid.empty()) return Geom();
35 if (valid.size() == 1) return valid.front();
36 return Geom::unionOf(valid);
37}

41/**
42 * @brief Calcula el diámetro de fresa más grande que todavía aísla el cobre.
43 *
44 * Modela cada isla de cobre y cada agujero como features independientes y
45 * busca la menor distancia entre features que no pertenezcan al mismo dueño
46 * (una isla y su propio agujero se tocan por definición y no limitan la fresa).
47 * El diámetro devuelto es el 98 % de esa separación, para que la fresa no roce
48 * ambas paredes.
49 * @param copper Geometría del cobre (polígonos con huecos).
50 * @param maxDiameter Tope superior y valor de retorno si no hay pares válidos.
51 * @return Estructura con diámetro, separación, los dos puntos de la cota y la
52 * bandera valid.
53 * @note Es una búsqueda cuadrática sobre features; aceptable para una capa
54 * típica, pero costosa en cobres con cientos de islas.
55 */
56OptimalTool optimalIsolationDiameter(const Geom& copper, double maxDiameter) {
57 OptimalTool result;
58 result.diameter = maxDiameter;
59 if (!copper.valid() || copper.empty()) return result;
60
61 // Each outer polygon is one copper island. Holes inside an island are
62 // copper-free, so the web between two holes also limits the tool.
63 struct Feature {
64 Geom geom;
65 int owner = -1;
66 bool hole = false;
67 };
68 std::vector<Feature> features;
69 int owner = 0;
70 for (const PolygonContours& pc : copper.polygons()) {
71 if (pc.outer.size() < 3) continue;
72 features.push_back({Geom::polygon(pc.outer, pc.holes), owner, false});
73 for (const Ring& hole : pc.holes) {
74 if (hole.size() >= 3) features.push_back({Geom::polygon(hole), owner, true});
75 }
76 ++owner;
77 }
78
79 double best = maxDiameter;
80 bool found = false;
81 for (std::size_t i = 0; i < features.size(); ++i) {
82 for (std::size_t j = i + 1; j < features.size(); ++j) {
83 const Feature& a = features[i];
84 const Feature& b = features[j];
85 // A hole touches its own island boundary; that distance is zero
86 // and says nothing about the tool width.
87 if (a.hole && b.hole && a.owner == b.owner) continue;
88 if (a.hole != b.hole && a.owner == b.owner) continue;
89 if (!a.hole && !b.hole && a.owner == b.owner) continue;
90 const double d = a.geom.distance(b.geom);
91 if (d > 1e-7 && (!found || d < best)) {
92 best = d;
93 found = true;
94 const std::pair<Pt, Pt> pts = a.geom.nearestPoints(b.geom);
95 result.pointA = pts.first;
96 result.pointB = pts.second;
97 }
98 }
99 }
100
101 result.valid = found;
102 result.gap = found ? best : maxDiameter;
103 // A tiny safety factor: a tool exactly as wide as the gap rubs both walls.
104 result.diameter = found ? best * 0.98 : maxDiameter;
105 return result;
106}

108/**
109 * @brief Lista las separaciones de cobre que limitan el diámetro de fresa.
110 *
111 * Hace el mismo relevamiento de features que optimalIsolationDiameter pero
112 * conserva todos los pares con separación menor que maxDiameter, los ordena de
113 * menor a mayor y descarta los que caen a menos de minSeparation de otro par ya
114 * elegido, para no dibujar cotas superpuestas.
115 * @param copper Cobre a analizar.
116 * @param maxDiameter Las separaciones mayores o iguales se descartan.
117 * @param limit Cantidad máxima de pares devueltos.
118 * @param minSeparation Distancia mínima entre cotas; 0 desactiva el filtro.
119 * @return Pares ordenados por separación creciente.
120 */
121std::vector<OptimalTool> optimalIsolationGaps(const Geom& copper, double maxDiameter,
122 std::size_t limit, double minSeparation) {
123 std::vector<OptimalTool> gaps;
124 if (!copper.valid() || copper.empty()) return gaps;
125 struct Feature {
126 Geom geom;
127 int owner = -1;
128 bool hole = false;
129 };
130 std::vector<Feature> features;
131 int owner = 0;
132 for (const PolygonContours& pc : copper.polygons()) {
133 if (pc.outer.size() < 3) continue;
134 features.push_back({Geom::polygon(pc.outer, pc.holes), owner, false});
135 for (const Ring& hole : pc.holes) {
136 if (hole.size() >= 3) features.push_back({Geom::polygon(hole), owner, true});
137 }
138 ++owner;
139 }
140 for (std::size_t i = 0; i < features.size(); ++i) {
141 for (std::size_t j = i + 1; j < features.size(); ++j) {
142 const Feature& a = features[i];
143 const Feature& b = features[j];
144 if (a.hole && b.hole && a.owner == b.owner) continue;
145 if (a.hole != b.hole && a.owner == b.owner) continue;
146 if (!a.hole && !b.hole && a.owner == b.owner) continue;
147 const double d = a.geom.distance(b.geom);
148 if (d <= 1e-7 || d >= maxDiameter) continue;
149 OptimalTool gap;
150 gap.gap = d;
151 gap.diameter = d * 0.98;
152 gap.valid = true;
153 const std::pair<Pt, Pt> pts = a.geom.nearestPoints(b.geom);
154 gap.pointA = pts.first;
155 gap.pointB = pts.second;
156 gaps.push_back(gap);
157 }
158 }
159 std::sort(gaps.begin(), gaps.end(),
160 [](const OptimalTool& a, const OptimalTool& b) { return a.gap < b.gap; });
161 if (minSeparation > 0.0) {
162 std::vector<OptimalTool> filtered;
163 for (const OptimalTool& gap : gaps) {
164 bool near = false;
165 for (const OptimalTool& kept : filtered) {
166 if (std::hypot(gap.pointA.x - kept.pointA.x, gap.pointA.y - kept.pointA.y) <
167 minSeparation) {
168 near = true;
169 break;
170 }
171 }
172 if (!near) filtered.push_back(gap);
173 }
174 gaps = std::move(filtered);
175 }
176 if (gaps.size() > limit) gaps.resize(limit);
177 return gaps;
178}

180/**
181 * @brief Distancia mínima entre dos conjuntos de geometrías (dos capas).
182 *
183 * Une cada conjunto por separado para que las partes de una misma capa no
184 * cuenten como obstáculo y mide entre los dos resultados.
185 * @param a Primera capa.
186 * @param b Segunda capa.
187 * @param[out] nearestA Punto más cercano en la capa A (puede ser nullptr).
188 * @param[out] nearestB Punto más cercano en la capa B (puede ser nullptr).
189 * @return Distancia en mm, o -1 si alguna capa queda vacía o inválida.
190 */
191double layersDistance(const std::vector<Geom>& a, const std::vector<Geom>& b, Pt* nearestA,
192 Pt* nearestB) {
193 const Geom ua = unionAll(a);
194 const Geom ub = unionAll(b);
195 if (!ua.valid() || !ub.valid() || ua.empty() || ub.empty()) return -1.0;
196 const std::pair<Pt, Pt> pts = ua.nearestPoints(ub);
197 if (nearestA) *nearestA = pts.first;
198 if (nearestB) *nearestB = pts.second;
199 return ua.distance(ub);
200}

202/**
203 * @brief Deduce taladros a partir de los flashes circulares de un Gerber.
204 *
205 * Recorre las aperturas circulares, descarta las que superan maxDiameter y
206 * examina su geometría: sólo acepta piezas aproximadamente cuadradas en su
207 * caja (un flash redondo) y de tamaño coherente con el diámetro, con lo que
208 * quedan afuera los trazos, que son cápsulas alargadas. Deduplica por
209 * diámetro y centro redondeados a micrón y agrupa los taladros en herramientas
210 * por diámetro.
211 * @param gerber Resultado del parser Gerber con aperturas y geometría.
212 * @param maxDiameter Diámetro máximo a considerar; los pads grandes no son
213 * taladros.
214 * @param appUnits Unidades con las que se informa el Excellon resultante.
215 * @param stepsPerCircle Segmentos por círculo de la geometría generada.
216 * @return ExcellonResult con herramientas, taladros, geometría y límites.
217 * @note Es una heurística geométrica: los pads cuadrados o rectangulares se
218 * descartan por la comprobación de caja, no por leer atributos.
219 */
220ExcellonResult extractDrills(const GerberResult& gerber, double maxDiameter, Units appUnits,
221 int stepsPerCircle) {
222 ExcellonResult result;
223 result.fileUnits = appUnits;
224 result.unitsKnown = true;
225
226 struct Hit {
227 double diameter = 0.0;
228 Pt center;
229 };
230 std::vector<Hit> hits;
231 std::set<std::tuple<long long, long long, long long>> seen;
232
233 for (const auto& [code, aperture] : gerber.apertures) {
234 if (aperture.kind != GerberAperture::Kind::Circle) continue;
235 double diameter = aperture.size > 0.0 ? aperture.size : aperture.diameter;
236 if (diameter <= 1e-6) continue;
237 if (diameter > maxDiameter + 1e-9) continue;
238
239 for (const Geom& g : aperture.geometry) {
240 if (!g.valid() || g.empty()) continue;
241 const BBox b = g.bounds();
242 if (!b.valid) continue;
243 const double w = b.width();
244 const double h = b.height();
245 if (w <= 1e-6 || h <= 1e-6) continue;
246 // Flashes are round/square; strokes are long capsules or traces.
247 if (std::fabs(w - h) > std::max(0.03, diameter * 0.25)) continue;
248 if (w > diameter * 1.6 || h > diameter * 1.6) continue;
249
250 const long long kd = std::llround(diameter * 1000.0);
251 const long long kx = std::llround(((b.xmin + b.xmax) * 0.5) * 1000.0);
252 const long long ky = std::llround(((b.ymin + b.ymax) * 0.5) * 1000.0);
253 if (!seen.insert({kd, kx, ky}).second) continue;
254 hits.push_back({diameter, {(b.xmin + b.xmax) * 0.5, (b.ymin + b.ymax) * 0.5}});
255 }
256 }
257
258 std::map<long long, int> toolForDiameter;
259 int nextTool = 1;
260 for (const Hit& hit : hits) {
261 const long long key = std::llround(hit.diameter * 1000.0);
262 int number = 0;
263 const auto it = toolForDiameter.find(key);
264 if (it == toolForDiameter.end()) {
265 number = nextTool++;
266 toolForDiameter[key] = number;
267 ExcellonTool tool;
268 tool.number = number;
269 tool.diameter = hit.diameter;
270 tool.diameterKnown = true;
271 result.tools[number] = tool;
272 } else {
273 number = it->second;
274 }
275 ExcellonTool& tool = result.tools[number];
276 tool.drills.push_back(hit.center);
277 const Geom circle = Geom::circle(hit.center, hit.diameter / 2.0, stepsPerCircle);
278 tool.geometry.push_back(circle);
279 result.geometry.push_back(circle);
280 ++result.drillCount;
281 }
282
283 if (!result.geometry.empty()) result.solid = Geom::unionOf(result.geometry);
284 for (const Geom& g : result.geometry) {
285 const BBox b = g.bounds();
286 if (!b.valid) continue;
287 if (!result.bounds.valid) {
288 result.bounds = b;
289 } else {
290 result.bounds.xmin = std::min(result.bounds.xmin, b.xmin);
291 result.bounds.ymin = std::min(result.bounds.ymin, b.ymin);
292 result.bounds.xmax = std::max(result.bounds.xmax, b.xmax);
293 result.bounds.ymax = std::max(result.bounds.ymax, b.ymax);
294 }
295 }
296 if (hits.empty()) result.warnings.push_back("No circular flashes found below the diameter limit.");
297 return result;
298}

300/**
301 * @brief Resta los taladros al cobre (punch) para obtener el negativo.
302 *
303 * Aplica primero un buffer de holgura a los taladros y luego los descuenta del
304 * sólido del Gerber.
305 * @param gerber Gerber con el cobre; se usa gerber.solid.
306 * @param drills Excellon con los taladros a restar.
307 * @param margin Holgura en mm; si es <= 0 se resta el taladro exacto.
308 * @return Cobre perforado, o el cobre original si no hay taladros.
309 */
310Geom punchGerber(const GerberResult& gerber, const ExcellonResult& drills, double margin) {
311 if (!gerber.solid.valid() || gerber.solid.empty()) return Geom();
312 if (!drills.solid.valid() || drills.solid.empty()) return gerber.solid;
313 const Geom holes = margin > 0.0 ? drills.solid.buffer(margin, 16) : drills.solid;
314 return gerber.solid.difference(holes);
315}

317/**
318 * @brief Reescribe el diámetro de una apertura circular en el texto Gerber.
319 *
320 * Busca %ADD<code>C,<diámetro>[X<hueco>]*% y reemplaza el diámetro, respetando
321 * el segundo valor X (hueco) si estaba presente. No toca otras aperturas ni
322 * los comandos de dibujo.
323 * @param gerber Texto Gerber completo.
324 * @param code Código de apertura, por ejemplo "10".
325 * @param diameter Nuevo diámetro en las unidades del archivo.
326 * @param decimals Decimales con que se escribe el valor.
327 * @param[out] changed Cantidad de reemplazos (puede ser nullptr).
328 * @return Texto modificado, o el original si no hubo coincidencias.
329 */
330std::string editCircleAperture(std::string gerber, const std::string& code, double diameter,
331 int decimals, int* changed) {
332 if (changed) *changed = 0;
333 char buffer[64];
334 std::snprintf(buffer, sizeof(buffer), "%.*f", decimals, diameter);
335 const std::regex pattern("%ADD" + code + "C,([0-9.]+)(X[0-9.]+)?\\*%");
336 const std::string replacement = "%ADD" + code + "C," + buffer + "$2*%";
337 int count = 0;
338 std::string out;
339 out.reserve(gerber.size());
340 auto begin = std::sregex_iterator(gerber.begin(), gerber.end(), pattern);
341 auto end = std::sregex_iterator();
342 std::size_t last = 0;
343 for (auto it = begin; it != end; ++it) {
344 const std::smatch& match = *it;
345 out.append(gerber, last, static_cast<std::size_t>(match.position()) - last);
346 out += "%ADD" + code + "C," + buffer + (match[2].matched ? match[2].str() : "") + "*%";
347 last = static_cast<std::size_t>(match.position() + match.length());
348 ++count;
349 }
350 out.append(gerber, last, std::string::npos);
351 if (changed) *changed = count;
352 return count > 0 ? out : gerber;
353}

355/**
356 * @brief Cambia el tamaño de una apertura circular, rectangular u obround.
357 *
358 * Intenta primero la variante circular; si no hay coincidencias prueba R y O,
359 * escalando ambos lados para conservar la relación de aspecto con el lado
360 * mayor como referencia, y por último P (polígono), donde cambia el diámetro y
361 * conserva vértices y rotación.
362 * @param gerber Texto Gerber completo.
363 * @param code Código de apertura.
364 * @param size Tamaño objetivo en las unidades del archivo.
365 * @param decimals Decimales de salida.
366 * @param[out] changed Cantidad de aperturas modificadas.
367 * @return Texto modificado, o el original si no hubo coincidencias.
368 */
369std::string editApertureSize(std::string gerber, const std::string& code, double size,
370 int decimals, int* changed) {
371 if (changed) *changed = 0;
372 int count = 0;
373 std::string out = editCircleAperture(gerber, code, size, decimals, &count);
374 if (count > 0) {
375 if (changed) *changed = count;
376 return out;
377 }
378
379 auto replaceMatches = [&](const std::regex& pattern,
380 const std::function<std::string(const std::smatch&)>& make) {
381 std::string result;
382 result.reserve(gerber.size());
383 std::size_t last = 0;
384 int local = 0;
385 for (auto it = std::sregex_iterator(gerber.begin(), gerber.end(), pattern);
386 it != std::sregex_iterator(); ++it) {
387 const std::smatch& match = *it;
388 result.append(gerber, last, static_cast<std::size_t>(match.position()) - last);
389 result += make(match);
390 last = static_cast<std::size_t>(match.position() + match.length());
391 ++local;
392 }
393 result.append(gerber, last, std::string::npos);
394 if (local > 0) {
395 gerber = std::move(result);
396 count += local;
397 }
398 };
399
400 char buffer[64];
401 const std::regex rect("%ADD" + code + "([RO]),([0-9.]+)X([0-9.]+)\\*%");
402 replaceMatches(rect, [&](const std::smatch& match) {
403 const double w = std::stod(match[2].str());
404 const double h = std::stod(match[3].str());
405 const double old = std::max(w, h);
406 const double factor = old > 0.0 ? size / old : 1.0;
407 std::snprintf(buffer, sizeof(buffer), "%%ADD%s%s,%.*fX%.*f*%%", code.c_str(),
408 match[1].str().c_str(), decimals, w * factor, decimals, h * factor);
409 return std::string(buffer);
410 });
411 const std::regex polygon("%ADD" + code + "P,([0-9.]+)X(\\d+)(X[0-9.]+)?\\*%");
412 replaceMatches(polygon, [&](const std::smatch& match) {
413 std::snprintf(buffer, sizeof(buffer), "%%ADD%sP,%.*fX%s%s*%%", code.c_str(), decimals,
414 size, match[2].str().c_str(),
415 match[3].matched ? match[3].str().c_str() : "");
416 return std::string(buffer);
417 });
418
419 if (changed) *changed = count;
420 return count > 0 ? gerber : out;
421}

423/**
424 * @brief Genera las marcas en L de las cuatro esquinas de una caja.
425 *
426 * Cada esquina aporta dos segmentos de largo @p length. Con outside en false
427 * las marcas quedan hacia adentro de la caja, desplazadas margin; con outside
428 * en true se dibujan hacia afuera e invertidas, para usarlas como referencia de
429 * alineación fuera del contorno.
430 * @param b Caja de referencia; si no es válida devuelve vacío.
431 * @param length Largo de cada brazo en mm.
432 * @param margin Separación al borde en mm.
433 * @param outside true para marcas exteriores.
434 * @return Hasta 8 LineString, dos por esquina.
435 */
436std::vector<Geom> cornerMarks(const BBox& b, double length, double margin, bool outside) {
437 std::vector<Geom> out;
438 if (!b.valid || length <= 0.0) return out;
439 const double k = outside ? -margin : margin;
440 const double x0 = b.xmin + k;
441 const double y0 = b.ymin + k;
442 const double x1 = b.xmax - k;
443 const double y1 = b.ymax - k;
444 const double dx = outside ? -length : length;
445 const double dy = outside ? -length : length;
446
447 auto add = [&out](Pt a, Pt c) { out.push_back(Geom::line({a, c})); };
448 // Bottom-left
449 add({x0, y0}, {x0 + dx, y0});
450 add({x0, y0}, {x0, y0 + dy});
451 // Bottom-right
452 add({x1, y0}, {x1 - dx, y0});
453 add({x1, y0}, {x1, y0 + dy});
454 // Top-left
455 add({x0, y1}, {x0 + dx, y1});
456 add({x0, y1}, {x0, y1 - dy});
457 // Top-right
458 add({x1, y1}, {x1 - dx, y1});
459 add({x1, y1}, {x1, y1 - dy});
460 return out;
461}
