25/**
26 * @brief Perfil de postprocesador (firmware/controlador).
27 *
28 * Cambia cabecera, comandos de husillo/láser y si se emite `M6`.
29 */
30enum class GCodeFlavor {
31 Default, // MACH3
32 DefaultNoM6, // MACH3 sin M6
33 Grbl, // GRBL 1.1
34 GrblNoM6, // GRBL 1.1 sin M6
35 GrblLaser, // GRBL láser (M03/M05)
36 Marlin, // Marlin
37 MarlinLaserFan, // Marlin láser en pin ventilador (M106/M107)
38 MarlinLaserSpindle, // Marlin láser en pin spindle (M3/M5)
39 Repetier, // Repetier
40 Berta, // Berta CNC
41 Isel, // ISEL CNC
42 IselIcp, // ISEL ICP CNC
43 Roland, // Roland MDX-20
44 ToolchangeManual, // MACH3 cambio manual
45 ToolchangeProbe, // MACH3 con sonda
46 ZLaser, // láser controlado por Z
47 Hpgl, // plotter HPGL
48 LineXyz, // G-code con Z en cada movimiento lineal
49 ToolchangeCustom // toolchange definido por el usuario
50};

52/**
53 * @brief Parámetros de generación de un programa G-code.
54 */
55struct GCodeOptions {
56 Units units = Units::Mm; ///< Unidades del programa (`G20`/`G21`).
57 int decimals = 4; ///< Decimales de coordenadas.
58 int feedDecimals = 2; ///< Decimales de avances.
59
60 double toolDiameter = 0.0; ///< Diámetro informativo para comentarios/exportación.
61 double feedXY = 200.0; ///< Avance horizontal en mm/min.
62 double feedZ = 60.0; ///< Avance de penetración en mm/min.
63 double zMove = 2.0; ///< Altura de viaje.
64 double zCut = -1.5; ///< Profundidad final de corte.
65 double zStart = 5.0; ///< Z inicial del programa.
66 double zEnd = 5.0; ///< Z final del programa.
67 double depthPerPass = 0.0; ///< Profundidad por pasada; 0 = una sola bajada.
68
69 double spindleSpeed = 1000.0; ///< Velocidad del husillo (RPM).
70 bool spindleCw = true; ///< `true` para `M03`, `false` para `M04`.
71
72 bool includeToolChange = true; ///< Emite cambios de herramienta.
73 double zToolchange = 10.0; ///< Z segura para cambiar herramienta.
74 std::optional<Pt> toolchangeXY; ///< Posición XY opcional para el cambio.
75 std::optional<Pt> endXY; ///< Posición XY final opcional.
76
77 bool connectPaths = false; ///< Une recorridos cercanos sin retraer.
78 double connectTolerance = 0.5; ///< Distancia máxima para conectar (mm).
79 double dwellTime = 0.0; ///< Pausa tras bajar (segundos).
80 double laserPower = 300.0; ///< Potencia láser (`S`).
81 double zFocus = 0.0; ///< Z de foco en modo láser.
82 GCodeFlavor flavor = GCodeFlavor::Default; ///< Postprocesador.
83 /// Plantilla para `ToolchangeCustom`; admite `{tool}`, `{diameter}`,
84 /// `{z}`, `{x}` e `{y}`.
85 std::string customToolchange;
86};

89 * @brief Una herramienta de taladrado con sus agujeros y overrides.
90 */
91struct DrillTool {
92 int number = 1; ///< Número de herramienta.
93 double diameter = 0.8; ///< Diámetro de la broca.
94 std::vector<Pt> holes; ///< Centros de taladros.
95 std::vector<std::pair<Pt, Pt>> slotSegments; ///< Ranuras.
96 std::optional<double> zCut; ///< Override de Z de corte.
97 std::optional<double> feedZ; ///< Override de avance Z.
98 std::optional<double> feedXY; ///< Override de avance XY (ranuras).
99 std::optional<double> spindleSpeed; ///< Override de RPM.
100 std::optional<double> zMove; ///< Override de Z de viaje.
101 std::optional<double> dwell; ///< Override de pausa.
102 double offsetX = 0.0; ///< Desplazamiento X de todos los taladros.
103 double offsetY = 0.0; ///< Desplazamiento Y de todos los taladros.
104};
