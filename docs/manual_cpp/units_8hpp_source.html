16/// Unidad de trabajo de un archivo o de la aplicación.
17enum class Units {
18 Mm, ///< Milímetros.
19 Inch ///< Pulgadas.
20};
21
22/// Factor exacto de conversión pulgadas → milímetros.
23constexpr double kInchToMm = 25.4;
24
25/**
26 * @brief Factor multiplicativo para pasar de una unidad a otra.
27 * @param from Unidad de origen.
28 * @param to Unidad de destino.
29 * @return 1, 25.4 o 1/25.4 según corresponda.
30 */
31inline double conversionFactor(Units from, Units to) {
32 if (from == to) return 1.0;
33 return from == Units::Inch ? kInchToMm : 1.0 / kInchToMm;
34}
35
36/**
37 * @brief Convierte una longitud entre unidades.
38 * @param value Valor numérico.
39 * @param from Unidad de origen.
40 * @param to Unidad de destino.
41 * @return Valor convertido.
42 */
43inline double convert(double value, Units from, Units to) {
44 return value * conversionFactor(from, to);
45}

47/// @return Etiqueta corta de la unidad (`"MM"` o `"IN"`).
48inline std::string toString(Units u) { return u == Units::Mm ? "MM" : "IN"; }

50/**
51 * @brief Interpreta una coordenada Gerber con supresión de ceros.
52 *
53 * Reproduce `parse_gerber_number()` de FlatCAM:
54 * - `zeros == 'L'` o `'D'`: se omiten ceros a la izquierda; el número se
55 * escala por `10^-fracDigits`.
56 * - `zeros == 'T'`: se omiten ceros a la derecha; se rellenan los dígitos
57 * faltantes antes de escalar.
58 *
59 * @param text Texto de la coordenada (puede llevar signo).
60 * @param intDigits Dígitos enteros declarados en `FS`.
61 * @param fracDigits Dígitos fraccionarios declarados en `FS`.
62 * @param zeros Modo de supresión (`'L'`, `'T'` o `'D'`).
63 * @return Coordenada en las unidades del archivo.
64 */
65double parseGerberNumber(std::string_view text, int intDigits, int fracDigits, char zeros);
