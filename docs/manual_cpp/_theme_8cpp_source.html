112/// @brief Sustituye los marcadores `@color` de la plantilla QSS.
113QString expand(QString sheet, const Colors& c) {
114 const std::pair<const char*, QString> tokens[] = {
115 {"@window", c.window}, {"@base", c.base},
116 {"@alternate", c.alternate}, {"@text", c.text},
117 {"@muted", c.muted}, {"@disabled", c.disabled},
118 {"@borderStrong", c.borderStrong},
119 {"@border", c.border}, {"@buttonDisabled", c.buttonDisabled},
120 {"@button", c.button}, {"@hover", c.hover},
121 {"@pressed", c.pressed}, {"@header", c.header},
122 {"@accentHover", c.accentHover},
123 {"@accentSoft", c.accentSoft}, {"@accentText", c.accentText},
124 {"@accent", c.accent}, {"@scrollHover", c.scrollHover},
125 {"@scroll", c.scroll}, {"@tooltipBg", c.tooltipBg},
126 {"@tooltipText", c.tooltipText},
127 {"@checkIcon", c.checkIcon},
128 {"@chevronIcon", c.chevronIcon},
129 };
130 for (const auto& [token, value] : tokens)
131 sheet.replace(QString::fromLatin1(token), value);
132 return sheet;
133}

435QString themeKey(Theme theme) {
436 switch (theme) {
437 case Theme::ModernLight: return QStringLiteral("modern-light");
438 case Theme::ModernDark: return QStringLiteral("modern-dark");
439 case Theme::SystemLight: return QStringLiteral("system-light");
440 case Theme::SystemDark: return QStringLiteral("system-dark");
441 }
442 return QStringLiteral("modern-light");
443}

445Theme themeFromKey(const QString& key, Theme fallback) {
446 if (key == QLatin1String("modern-light")) return Theme::ModernLight;
447 if (key == QLatin1String("modern-dark")) return Theme::ModernDark;
448 if (key == QLatin1String("system-light")) return Theme::SystemLight;
449 if (key == QLatin1String("system-dark")) return Theme::SystemDark;
450 return fallback;
451}

453Theme themeFromSettings() {
454 QSettings settings;
455 const QString key = settings.value("preferences/theme").toString();
456 if (!key.isEmpty()) return themeFromKey(key);
457 // Compatibilidad con versiones anteriores que sólo guardaban el booleano.
458 return settings.value("preferences/darkTheme", false).toBool() ? Theme::ModernDark
459 : Theme::ModernLight;

462void apply(QApplication& app, Theme theme) {
463 g_dark = isDark(theme);
464
465 // La apariencia nativa (diálogos de archivo incluidos) sigue la variante
466 // elegida, aunque el sistema operativo esté configurado al revés.
467#if QT_VERSION >= QT_VERSION_CHECK(6, 8, 0)
468 QGuiApplication::styleHints()->setColorScheme(g_dark ? Qt::ColorScheme::Dark
469 : Qt::ColorScheme::Light);
470#endif
471
472 if (theme == Theme::ModernLight || theme == Theme::ModernDark) {
473 if (QStyle* style = QStyleFactory::create("Fusion")) app.setStyle(style);
474 app.setPalette(modernPalette(colorsFor(g_dark)));
475 app.setStyleSheet(styleSheet(g_dark));
476 return;
477 }
478
479 QStyle* style = QStyleFactory::create(platformStyleName());
480 if (!style) style = QStyleFactory::create("Fusion");
481 if (style) app.setStyle(style);
482 const QPalette native = app.style()->standardPalette();
483 app.setPalette(g_dark ? systemDarkPalette(native) : native);
484 app.setStyleSheet(QString::fromUtf8(kSystemTemplate));
485}
