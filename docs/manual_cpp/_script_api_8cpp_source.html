34/**
35 * @brief Aplana geometría de líneas en una lista de polilíneas.
36 *
37 * Convierte `LineString`/`LinearRing` a `std::vector<Pt>` y recurre sobre
38 * `MultiLineString` y `GeometryCollection`.
39 *
40 * @param geom Geometría a recorrer.
41 * @param out Lista de polilíneas resultante.
42 *
43 * @note Se usa en `mill` para transformar los recorridos "follow" del Gerber.
44 * @note Ignora polígonos, geometrías vacías e inválidas.
45 */
46void collectLines(const Geom& geom, std::vector<std::vector<Pt>>& out) {
47 if (!geom.valid() || geom.empty()) return;
48 const int type = geom.typeId();
49 if (type == GEOS_LINESTRING || type == GEOS_LINEARRING) {
50 Ring ring = geom.coords();
51 if (ring.size() >= 2) out.emplace_back(ring.begin(), ring.end());
52 return;
53 }
54 if (type == GEOS_MULTILINESTRING || type == GEOS_GEOMETRYCOLLECTION) {
55 for (int i = 0; i < geom.numGeometries(); ++i) collectLines(geom.geometryN(i), out);
56 }
57}

127/**
128 * @brief Calcula el aislamiento del Gerber activo y lo publica como capa.
129 *
130 * Convierte el resultado de `isolateCopper` en un `PathJob` (para una eventual
131 * exportación) y entrega los recorridos al `LayerSink` con nombre `isolate`.
132 *
133 * @param toolDiameter Diámetro de fresa en mm.
134 * @param passes Cantidad de pasadas.
135 * @param overlapPercent Solape entre pasadas en porcentaje (se divide por 100).
136 * @return Cantidad de recorridos, o -1 si no hay Gerber cargado.
137 *
138 * @note Descarta los recorridos con menos de 2 puntos.
139 * @warning No actualiza `error_` en éxito, por lo que `lastError()` puede
140 * conservar un mensaje de una llamada anterior.
141 */
142int ScriptApi::isolate(double toolDiameter, int passes, double overlapPercent) {
143 error_.clear();
144 if (!hasGerber_) {
145 error_ = "no gerber loaded";
146 return -1;
147 }
148 IsolationOptions opt;
149 opt.toolDiameter = toolDiameter;
150 opt.passes = passes;
151 opt.overlap = overlapPercent / 100.0;
152 const IsolationResult result = isolateCopper(gerber_.solid, opt);
153 paths_ = PathJob{};
154 paths_.bounds = result.bounds;
155 paths_.toolDiameter = toolDiameter;
156 for (const Geom& g : result.paths) {
157 Ring ring = g.coords();
158 if (ring.size() >= 2) paths_.paths.emplace_back(ring.begin(), ring.end());
159 }
160 hasPathJob_ = true;
161 hasDrillJob_ = false;
162 if (sink_) sink_("isolate", "herramienta", {}, result.paths);
163 return static_cast<int>(result.paths.size());
164}

277/**
278 * @brief Prepara fresado siguiendo los recorridos "follow" del Gerber activo.
279 *
280 * Aplana `gerber_.follows` con `collectLines`, arma el `PathJob` y entrega las
281 * líneas reconstruidas (`Geom::line`) al `LayerSink` con nombre `mill`.
282 *
283 * @param toolDiameter Diámetro de fresa en mm.
284 * @return Cantidad de recorridos, o -1 sin Gerber.
285 *
286 * @note A diferencia de la CLI, no aplica optimización de orden; el orden es el
287 * del archivo Gerber.
288 */
289int ScriptApi::mill(double toolDiameter) {
290 error_.clear();
291 if (!hasGerber_) {
292 error_ = "no gerber loaded";
293 return -1;
294 }
295 std::vector<std::vector<Pt>> lines;
296 for (const Geom& g : gerber_.follows) collectLines(g, lines);
297 paths_ = PathJob{};
298 paths_.paths = std::move(lines);
299 paths_.bounds = gerber_.bounds;
300 paths_.toolDiameter = toolDiameter;
301 hasPathJob_ = true;
302 hasDrillJob_ = false;
303 std::vector<Geom> linesGeom;
304 for (const auto& line : paths_.paths) linesGeom.push_back(Geom::line(line));
305 if (sink_) sink_("mill", "herramienta", {}, linesGeom);
306 return static_cast<int>(paths_.paths.size());
307}

336/**
337 * @brief Resta los taladros del Excellon activo al cobre del Gerber activo.
338 *
339 * Genera el "punch" con `fc::punchGerber` y reemplaza la geometría del Gerber
340 * en memoria (`solid`, `solids` y `bounds`) por el resultado, que además se
341 * publica en el `LayerSink` con nombre `punch`.
342 *
343 * @param margin Holgura alrededor de cada taladro.
344 * @return Cantidad de geometrías del resultado, o -1 si falta el Gerber, el
345 * Excellon o el resultado queda vacío.
346 *
347 * @note Los recorridos "follow" originales se descartan porque pueden cruzar
348 * los taladros recién restados; el resultado queda como sólidos.
349 * @warning `punchGerber` devuelve un único `Geom` que suele ser un
350 * `MultiPolygon`; si la implementación devolviera otra colección, el
351 * conteo de `numGeometries()` no equivale a la cantidad de piezas
352 * independientes.
353 */
354int ScriptApi::punch(double margin) {
355 error_.clear();
356 if (!hasGerber_ || !hasExcellon_) {
357 error_ = "need gerber and excellon";
358 return -1;
359 }
360 const Geom punched = fc::punchGerber(gerber_, excellon_, margin);
361 if (!punched.valid() || punched.empty()) {
362 error_ = "empty result";
363 return -1;
364 }
365 gerber_.solid = punched;
366 gerber_.solids = {punched};
367 gerber_.follows.clear();
368 gerber_.bounds = punched.bounds();
369 if (sink_) sink_("punch", "gerber", {punched}, {});
370 return punched.numGeometries();
371}

373/**
374 * @brief Paneliza el objeto activo rotándolo y repitiéndolo en una grilla.
375 *
376 * Rota cada sólido `angle` grados alrededor del centro del bounding box
377 * original, calcula la caja del conjunto rotado y luego copia la pieza en
378 * `rows × cols` posiciones, separadas por `ancho + gapX` y `alto + gapY`,
379 * alineando la primera celda con el origen original. Entrega todas las copias
380 * al `LayerSink` con nombre `panel`.
381 *
382 * @param rows Filas de la grilla (≥ 1).
383 * @param cols Columnas de la grilla (≥ 1).
384 * @param gapX Separación horizontal extra.
385 * @param gapY Separación vertical extra.
386 * @param angle Rotación en grados.
387 * @return Cantidad de geometrías generadas, o -1 si falta el Gerber, las filas
388 * o columnas no son positivas, o la caja resultante no es válida.
389 *
390 * @note `rows < 1` o `cols < 1` informan `no gerber loaded`, mensaje engañoso
391 * que no describe el problema real.
392 * @note Sólo se panelizan los sólidos; los recorridos "follow" no se replican.
393 * @note La función no modifica `gerber_`: el panel existe únicamente en la capa
394 * entregada al `LayerSink`.
395 */
396int ScriptApi::panelize(int rows, int cols, double gapX, double gapY, double angle) {
397 error_.clear();
398 if (!hasGerber_ || rows < 1 || cols < 1) {
399 error_ = "no gerber loaded";
400 return -1;
401 }
402 const BBox b = gerber_.bounds;
403 if (!b.valid) return -1;
404 const Pt center{(b.xmin + b.xmax) * 0.5, (b.ymin + b.ymax) * 0.5};
405 std::vector<Geom> rotated;
406 BBox rb;
407 for (const Geom& g : gerber_.solids) {
408 const Geom r = g.rotateDeg(angle, center);
409 rotated.push_back(r);
410 const BBox gb = r.bounds();
411 if (!gb.valid) continue;
412 if (!rb.valid) rb = gb;
413 else {
414 rb.xmin = std::min(rb.xmin, gb.xmin);
415 rb.ymin = std::min(rb.ymin, gb.ymin);
416 rb.xmax = std::max(rb.xmax, gb.xmax);
417 rb.ymax = std::max(rb.ymax, gb.ymax);
418 }
419 }
420 if (!rb.valid) return -1;
421 const double dx = rb.width() + gapX;
422 const double dy = rb.height() + gapY;
423 std::vector<Geom> out;
424 for (int r = 0; r < rows; ++r) {
425 for (int c = 0; c < cols; ++c) {
426 const double ox = b.xmin - rb.xmin + c * dx;
427 const double oy = b.ymin - rb.ymin + r * dy;
428 for (const Geom& g : rotated) out.push_back(g.translate(ox, oy));
429 }
430 }
431 if (sink_) sink_("panel", "gerber", out, {});
432 return static_cast<int>(out.size());
433}

507/**
508 * @brief Comprueba la separación entre el cobre del Gerber y los taladros.
509 *
510 * Calcula la distancia mínima entre `gerber_.solid` y `excellon_.solid` y la
511 * compara con el valor pedido.
512 *
513 * @param clearance Separación mínima exigida en mm.
514 * @return 1 si hay violación (distancia menor que `clearance`), 0 si cumple, y

521int ScriptApi::rulesCheck(double clearance) {
522 error_.clear();
523 if (!hasGerber_ || !hasExcellon_) {
524 error_ = "need gerber and excellon";
525 return -1;
526 }
527 const double d = gerber_.solid.distance(excellon_.solid);
528 return d < clearance ? 1 : 0;

531/**
532 * @brief Exporta el trabajo activo a un archivo G-code.
533 *
534 * Si hay `DrillJob` genera taladrado; si no, los recorridos de geometría. Usa
535 * `GCodeOptions` por defecto (milímetros, preprocesador `Default`).
536 *
537 * @param path Archivo de destino.
538 * @return `true` si se escribió; `false` con `lastError()` si no hay trabajo,
539 * no se puede abrir el archivo o el generador lanza una excepción.
540 *
541 * @note La escritura usa `QIODevice::Text`, de modo que en Windows el texto
542 * puede convertirse a CRLF (los firmwares suelen aceptarlo).
543 * @warning No hay parámetros de máquina expuestos: avances, Z, husillo y
544 * postprocesador no se pueden configurar desde el script.
545 * @warning `QFile::write(code.c_str())` depende del terminador nulo y no pasa la
546 * longitud; es correcto para G-code sin bytes nulos, pero no es la
547 * forma recomendada de escribir datos binarios.
548 */
549bool ScriptApi::exportGcode(const QString& path) {
550 error_.clear();
551 try {
552 GCodeOptions opt;
553 std::string code;
554 if (hasDrillJob_) code = drillingGCode(drills_, opt);
555 else if (hasPathJob_) code = pathsGCode(paths_, opt);
556 else {
557 error_ = "nothing to export";
558 return false;
559 }
560 QFile file(path);
561 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
562 error_ = "cannot write " + path;
563 return false;
564 }
565 file.write(code.c_str());
566 return true;
567 } catch (const std::exception& e) {
568 error_ = QString::fromUtf8(e.what());
569 return false;
570 }
571}

603/**
604 * @brief Instala la API en un motor JavaScript como objeto global `flatcam`.
605 *
606 * Registra `api` con `QJSEngine::newQObject` y define la función global
607 * `print(s)`, que delega en `flatcam.log(String(s))`.
608 *
609 * @param engine Motor JavaScript destino.
610 * @param api Instancia de la API; el llamador conserva su propiedad.
611 *
612 * @note Se fija `CppOwnership` antes de exponer el objeto: la instancia vive en
613 * la pila del llamador y el motor no debe destruirla.
614 * @note La función `print` se evalúa una sola vez; si el script la redefine,
615 * prevalece la del script.
616 */
617void installScriptApi(QJSEngine& engine, ScriptApi* api) {
618 QJSEngine::setObjectOwnership(api, QJSEngine::CppOwnership);
619 engine.globalObject().setProperty("flatcam", engine.newQObject(api));
620 engine.evaluate("function print(s) { flatcam.log(String(s)); }");
621}
