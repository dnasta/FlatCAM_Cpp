17/// Contexto compartido entre GEOSSTRtree_query y el callback C.
18struct QueryContext {
19 std::vector<std::uint32_t>* out;
20};

22/**
23 * @brief Callback C invocado por GEOS por cada elemento que cae en la ventana.
24 *
25 * GEOS devuelve el puntero tal como se insertó; como en la inserción se guardó
26 * el id casteado a puntero, acá se reconstruye vía intptr_t antes de añadirlo
27 * a la lista de resultados.
28 * @param item Puntero opaco registrado en la inserción (el id).
29 * @param userdata Puntero a QueryContext con el vector de salida.
30 */
31void queryCallback(void* item, void* userdata) {
32 auto* ctx = static_cast<QueryContext*>(userdata);
33 ctx->out->push_back(static_cast<std::uint32_t>(reinterpret_cast<std::intptr_t>(item)));
34}

55/**
56 * @brief Inserta una geometría asociándole un identificador.
57 *
58 * Guarda una copia en keep_ (dueña de la memoria), registra el id y entrega a
59 * GEOS el puntero crudo junto con el id casteado a void*. Las geometrías
60 * inválidas se ignoran en silencio.
61 * @param g Geometría a indexar.
62 * @param id Identificador que devolverá query().
63 * @note El patrón previsto es insertar todo y recién entonces consultar:
64 * insertar después de build() exige reconstruir el árbol.
65 */
66void SpatialIndex::insert(const Geom& g, std::uint32_t id) {
67 if (!g.valid()) return;
68 keep_.push_back(g);
69 ids_.push_back(id);
70 GEOSSTRtree_insert(tree_, g.raw(), reinterpret_cast<void*>(static_cast<std::intptr_t>(id)));
71}

84/**
85 * @brief Devuelve los ids cuyas geometrías intersectan la caja dada.
86 *
87 * Fuerza build() si aún no se hizo, crea un rectángulo GEOS con la ventana y
88 * delega en GEOSSTRtree_query. La comparación es sólo de envolventes: si se
89 * necesita intersección exacta, el llamador debe verificarla después.
90 * @param box Ventana de búsqueda; si no es válida devuelve lista vacía.
91 * @return Ids en el orden en que GEOS los recorre (sin orden garantizado).
92 */
93std::vector<std::uint32_t> SpatialIndex::query(const BBox& box) const {
94 std::vector<std::uint32_t> out;
95 if (!box.valid) return out;
96 if (!built_) const_cast<SpatialIndex*>(this)->build();
97 GEOSGeometry* envelope = GEOSGeom_createRectangle(box.xmin, box.ymin, box.xmax, box.ymax);
98 if (!envelope) return out;
99 QueryContext ctx{&out};
100 GEOSSTRtree_query(tree_, envelope, queryCallback, &ctx);
101 GEOSGeom_destroy(envelope);
102 return out;
103}
